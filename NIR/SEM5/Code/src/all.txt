
#pragma once

enum class RoadCategory {
  Pedestrian,

  Car1Lane,
  Car2Lanes,
  Car3Lanes,
  Car4Lanes,
  Car5PlusLanes,
  CarUnknownLanes,

  Motorway,

  Unknown
};

#include "roadclassifier.hpp"

#include <QJsonObject>
#include <QJsonValue>

int RoadClassifier::getLanes(const QJsonObject &tags) const {
  QJsonValue value = tags.value("lanes");

  bool ok = false;

  int lanes = value.toString().toInt(&ok);

  if (!ok && value.isDouble()) {
    lanes = value.toInt();
    ok = true;
  }

  if (!ok || lanes <= 0)
    return 0;

  return lanes;
}

RoadCategory RoadClassifier::classifyFeature(const OSMFeature &feature,
                                             int &lanes) const {
  const QJsonObject &tags = feature.getTags();

  QString highway = tags.value("highway").toString();

  lanes = getLanes(tags);

  if (highway.isEmpty())
    return RoadCategory::Unknown;

  // Motorways
  if (highway == "motorway" || highway == "motorway_link") {
    return RoadCategory::Motorway;
  }

  // Pedestrian roads
  if (highway == "footway" || highway == "pedestrian" || highway == "path" ||
      highway == "steps" || highway == "bridleway") {
    return RoadCategory::Pedestrian;
  }

  // Other recognized road types
  static const QStringList carRoads = {
      "trunk",        "trunk_link",     "primary",       "primary_link",
      "secondary",    "secondary_link", "tertiary",      "tertiary_link",
      "unclassified", "residential",    "living_street", "service",
      "road"};

  if (!carRoads.contains(highway))
    return RoadCategory::Unknown;

  if (lanes == 0)
    return RoadCategory::CarUnknownLanes;

  if (lanes == 1)
    return RoadCategory::Car1Lane;

  if (lanes == 2)
    return RoadCategory::Car2Lanes;

  if (lanes == 3)
    return RoadCategory::Car3Lanes;

  if (lanes == 4)
    return RoadCategory::Car4Lanes;

  return RoadCategory::Car5PlusLanes;
}

QVector<RoadClassifier::ClassifiedRoad>
RoadClassifier::classify(const QVector<OSMFeature> &features) const {
  QVector<ClassifiedRoad> result;

  result.reserve(features.size());

  for (const OSMFeature &feature : features) {
    int lanes = 0;

    RoadCategory category = classifyFeature(feature, lanes);

    if (category == RoadCategory::Unknown)
      continue;

    ClassifiedRoad road;
    road.feature = feature;
    road.category = category;
    road.lanes = lanes;

    result.append(road);
  }

  return result;
}

#include "roadgraphbuilder.hpp"

#include <QJsonObject>
#include <QJsonValue>
#include <QtMath>

RoadGraphBuilder::RoadGraphBuilder() = default;

RoadGraphBuilder::RoadGraphBuilder(const Settings &settings)
    : settings(settings) {}

int RoadGraphBuilder::getLayer(const QJsonObject &tags) const {
  bool ok = false;

  int layer = tags.value("layer").toString().toInt(&ok);

  if (!ok)
    layer = static_cast<int>(tags.value("layer").toDouble(0));

  return layer;
}

// -----------------------------------------------------------------------------
// Distance (Haversine)
// -----------------------------------------------------------------------------

double RoadGraphBuilder::calculateDistance(const GeoPoint &a,
                                           const GeoPoint &b) const {
  constexpr double EarthRadius = 6371000.0;

  double lat1 = qDegreesToRadians(a.lat);
  double lat2 = qDegreesToRadians(b.lat);

  double dLat = lat2 - lat1;
  double dLon = qDegreesToRadians(b.lon - a.lon);

  double h = qPow(qSin(dLat / 2.0), 2) +
             qCos(lat1) * qCos(lat2) * qPow(qSin(dLon / 2.0), 2);

  double c = 2.0 * qAtan2(qSqrt(h), qSqrt(1.0 - h));

  return EarthRadius * c;
}

// -----------------------------------------------------------------------------
// Node creation
// -----------------------------------------------------------------------------

quint32 RoadGraphBuilder::getOrCreateNode(RoadGraph &graph,
                                          const GeoPoint &point, int layer) {
  // Approximate coordinate quantization.
  // Latitude degrees converted to meters.
  constexpr double EarthRadius = 6371000.0;

  double latMeters = qDegreesToRadians(point.lat) * EarthRadius;

  double lonMeters = qDegreesToRadians(point.lon) * EarthRadius *
                     qCos(qDegreesToRadians(point.lat));

  double tolerance = qMax(settings.coordinateToleranceMeters, 0.01);

  qint64 x = qRound64(lonMeters / tolerance);
  qint64 y = qRound64(latMeters / tolerance);

  // Search nearby cells to avoid missing points
  // located on opposite sides of a quantization boundary.

  for (int dx = -1; dx <= 1; ++dx) {
    for (int dy = -1; dy <= 1; ++dy) {

      QString key = QString("%1_%2_%3").arg(x + dx).arg(y + dy).arg(layer);

      auto it = nodeLookup.constFind(key);

      if (it == nodeLookup.constEnd())
        continue;

      quint32 id = it.value();

      if (id >= static_cast<quint32>(graph.nodes.size()))
        continue;

      const RoadNode &node = graph.nodes[id];

      GeoPoint existing;
      existing.lat = node.lat;
      existing.lon = node.lon;

      if (calculateDistance(point, existing) <= tolerance)
        return id;
    }
  }

  quint32 id = graph.addNode(point.lat, point.lon);

  QString key = QString("%1_%2_%3").arg(x).arg(y).arg(layer);

  nodeLookup.insert(key, id);

  return id;
}

// -----------------------------------------------------------------------------
// Process feature
// -----------------------------------------------------------------------------

void RoadGraphBuilder::processFeature(
    RoadGraph &graph, const RoadClassifier::ClassifiedRoad &road) {
  const OSMFeature &feature = road.feature;

  const QVector<GeoPoint> &geometry = feature.getGeometry();

  if (geometry.size() < 2)
    return;

  const QJsonObject &tags = feature.getTags();

  QString highway = tags.value("highway").toString();

  QString name = tags.value("name").toString();

  int layer = getLayer(tags);

  QString oneway = tags.value("oneway").toString();

  bool oneWay =
      oneway == "yes" || oneway == "1" || oneway == "true" || oneway == "-1";

  bool reverse = oneway == "-1";

  QString bridgeTag = tags.value("bridge").toString();
  QString tunnelTag = tags.value("tunnel").toString();

  bool bridge = bridgeTag == "yes" || bridgeTag == "true";

  bool tunnel = tunnelTag == "yes" || tunnelTag == "true";

  for (int i = 0; i < geometry.size() - 1; ++i) {
    const GeoPoint &a = geometry[i];
    const GeoPoint &b = geometry[i + 1];

    double length = calculateDistance(a, b);

    if (length < 0.01)
      continue;

    quint32 from = getOrCreateNode(graph, a, layer);
    quint32 to = getOrCreateNode(graph, b, layer);

    if (from == to)
      continue;

    auto addRoadEdge = [&](quint32 edgeFrom, quint32 edgeTo, bool isReverse) {
      quint32 edgeId = graph.addEdge(edgeFrom, edgeTo, length);

      if (edgeId == UINT32_MAX)
        return;

      RoadEdge &edge = graph.edges[edgeId];

      edge.featureId = feature.getId();
      edge.highway = highway;
      edge.name = name;

      edge.layer = layer;

      edge.oneWay = oneWay;
      edge.isBridge = bridge;
      edge.isTunnel = tunnel;

      edge.category = road.category;
      edge.lanes = road.lanes;

      edge.isPedestrian = road.category == RoadCategory::Pedestrian;

      edge.isReverse = isReverse;
    };

    if (reverse) {
      addRoadEdge(to, from, true);
    } else {
      addRoadEdge(from, to, false);
    }

    if (!oneWay) {
      addRoadEdge(to, from, true);
    }
  }
}

// -----------------------------------------------------------------------------
// Build graph
// -----------------------------------------------------------------------------

RoadGraph
RoadGraphBuilder::build(const QVector<RoadClassifier::ClassifiedRoad> &roads) {
  RoadGraph graph;

  nodeLookup.clear();

  for (const RoadClassifier::ClassifiedRoad &road : roads) {
    processFeature(graph, road);
  }

  return graph;
}

#include "roadclassifier.hpp"
#include "roadgraph.hpp"

#include <QHash>
#include <QString>

class RoadGraphBuilder {
public:
  struct Settings {
    double coordinateToleranceMeters = 1.0;
  };

  RoadGraphBuilder();
  explicit RoadGraphBuilder(const Settings &settings);

  RoadGraph build(const QVector<RoadClassifier::ClassifiedRoad> &roads);

private:
  Settings settings;

  // Ключ координаты и слоя для поиска существующего узла.
  QHash<QString, quint32> nodeLookup;

  quint32 getOrCreateNode(RoadGraph &graph, const GeoPoint &point, int layer);

  double calculateDistance(const GeoPoint &a, const GeoPoint &b) const;

  int getLayer(const QJsonObject &tags) const;

  void processFeature(RoadGraph &graph,
                      const RoadClassifier::ClassifiedRoad &road);
};

#include "roadgraph.hpp"

void RoadGraph::clear() {
  nodes.clear();
  edges.clear();
}

quint32 RoadGraph::addNode(double lat, double lon) {
  quint32 id = static_cast<quint32>(nodes.size());

  RoadNode node;
  node.id = id;
  node.lat = lat;
  node.lon = lon;

  nodes.append(node);

  return id;
}

quint32 RoadGraph::addEdge(quint32 from, quint32 to, double length) {
  if (from >= static_cast<quint32>(nodes.size()) ||
      to >= static_cast<quint32>(nodes.size()) || from == to) {
    return UINT32_MAX;
  }

  quint32 id = static_cast<quint32>(edges.size());

  RoadEdge edge;
  edge.id = id;
  edge.from = from;
  edge.to = to;
  edge.length = length;

  edges.append(edge);

  nodes[from].edges.append(id);

  return id;
}

int RoadGraph::nodeCount() const { return nodes.size(); }

int RoadGraph::edgeCount() const { return edges.size(); }

const QVector<quint32> &RoadGraph::getEdges(quint32 nodeId) const {
  static const QVector<quint32> empty;

  if (nodeId >= static_cast<quint32>(nodes.size()))
    return empty;

  return nodes[nodeId].edges;
}

#pragma once

#include <QString>
#include <QVector>
#include <QtGlobal>

#include "roadcategory.hpp"

struct RoadNode {
  quint32 id = 0;

  double lat = 0.0;
  double lon = 0.0;

  QVector<quint32> edges;
};

struct RoadEdge {
  quint32 id = 0;

  quint32 from = 0;
  quint32 to = 0;

  qint64 featureId = 0;

  double length = 0.0;

  QString highway;
  QString name;

  int layer = 0;

  bool oneWay = false;
  bool isBridge = false;
  bool isTunnel = false;
  bool isPedestrian = false;
  bool isReverse = false;

  RoadCategory category = RoadCategory::Unknown;
  int lanes = 0;
};

class RoadGraph {
public:
  QVector<RoadNode> nodes;
  QVector<RoadEdge> edges;

  void clear();

  quint32 addNode(double lat, double lon);

  quint32 addEdge(quint32 from, quint32 to, double length);

  int nodeCount() const;
  int edgeCount() const;

  const QVector<quint32> &getEdges(quint32 nodeId) const;
};
