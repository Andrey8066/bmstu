bool DEMReader::loadDem(const QString &fileName, DEMData &out) {
  GDALAllRegister();

  GDALDataset *dataset = static_cast<GDALDataset *>(
      GDALOpen(fileName.toLocal8Bit().constData(), GA_ReadOnly));

  if (!dataset)
    return false;

  const int width = dataset->GetRasterXSize();
  const int height = dataset->GetRasterYSize();

  if (width < 2 || height < 2 || dataset->GetRasterCount() < 1) {
    GDALClose(dataset);
    return false;
  }

  GDALRasterBand *band = dataset->GetRasterBand(1);

  out.width = width;
  out.height = height;

  qDebug() << "Размер DEM:" << width << "x" << height;

  qDebug() << "Projection:" << dataset->GetProjectionRef();

  qDebug() << "Raster type:" << GDALGetDataTypeName(band->GetRasterDataType());

  int hasNoData = 0;
  double noData = band->GetNoDataValue(&hasNoData);

  qDebug() << "NoData exists:" << hasNoData << "value:" << noData;

  qDebug() << "Scale:" << band->GetScale() << "Offset:" << band->GetOffset();

  if (dataset->GetGeoTransform(out.geoTransform) != CE_None) {
    GDALClose(dataset);

    const char *projection = dataset->GetProjectionRef();

    out.projectionWkt = projection ? QString::fromUtf8(projection) : QString();

    return false;
  }

  out.elevation.resize(static_cast<std::size_t>(width) * height);

  const CPLErr result =
      band->RasterIO(GF_Read, 0, 0, width, height, out.elevation.data(), width,
                     height, GDT_Float32, 0, 0);

  if (result != CE_None) {
    GDALClose(dataset);
    out = DEMData{};
    return false;
  }

  float minZ = std::numeric_limits<float>::max();
  float maxZ = std::numeric_limits<float>::lowest();

  for (float &z : out.elevation) {
    if (!std::isfinite(z) || (hasNoData && z == static_cast<float>(noData))) {
      // Для первого прототипа заполняем пропуски нулём.
      // Для точной модели NoData лучше интерполировать.
      z = 0.0f;
    }

    minZ = std::min(minZ, z);
    maxZ = std::max(maxZ, z);
  }

  out.minElevation = minZ;
  out.maxElevation = maxZ;

  GDALClose(dataset);
  return true;
}

FeatureDispatcher::FeatureDispatcher(GLWidget *widget) : m_widget(widget) {}

void FeatureDispatcher::dispatch(const OSMFeature &feature) {
  if (!m_widget) {
    qWarning() << "FeatureDispatcher: GLWidget is null";
    return;
  }

  if (feature.geometry.isEmpty())
    return;

  switch (feature.type) {
  case OSMFeatureType::Building:
    dispatchBuilding(feature);
    break;

  case OSMFeatureType::Road:
    dispatchRoad(feature);
    break;

  case OSMFeatureType::Water:
    dispatchWater(feature);
    break;

  case OSMFeatureType::Vegetation:
    dispatchVegetation(feature);
    break;

  case OSMFeatureType::Landuse:
    dispatchLanduse(feature);
    break;

  case OSMFeatureType::Railway:
    dispatchRailway(feature);
    break;

  case OSMFeatureType::Amenity:
    dispatchAmenity(feature);
    break;

  case OSMFeatureType::Unknown:
  default:
    break;
  }
}

void FeatureDispatcher::dispatchAll(const QVector<OSMFeature> &features) {
  for (const OSMFeature &feature : features)
    dispatch(feature);
}

void FeatureDispatcher::dispatchBuilding(const OSMFeature &feature) {
  if (feature.geometryType == OSMGeometryType::Polygon)
    m_widget->addBuildingFeature(feature);
}

void FeatureDispatcher::dispatchRoad(const OSMFeature &feature) {
  if (feature.geometryType == OSMGeometryType::LineString)
    m_widget->addRoadFeature(feature);
  else if (feature.geometryType == OSMGeometryType::Polygon)
    m_widget->addRoadAreaFeature(feature);
}

void FeatureDispatcher::dispatchWater(const OSMFeature &feature) {
  if (feature.geometryType == OSMGeometryType::Polygon)
    m_widget->addWaterAreaFeature(feature);
  else if (feature.geometryType == OSMGeometryType::LineString)
    m_widget->addWaterLineFeature(feature);
}

void FeatureDispatcher::dispatchVegetation(const OSMFeature &feature) {
  if (feature.geometryType == OSMGeometryType::Polygon)
    m_widget->addVegetationAreaFeature(feature);
  else if (feature.geometryType == OSMGeometryType::Point)
    m_widget->addVegetationPointFeature(feature);
}

void FeatureDispatcher::dispatchLanduse(const OSMFeature &feature) {
  if (feature.geometryType == OSMGeometryType::Polygon)
    m_widget->addLanduseAreaFeature(feature);
}

void FeatureDispatcher::dispatchRailway(const OSMFeature &feature) {
  if (feature.geometryType == OSMGeometryType::LineString)
    m_widget->addRailwayFeature(feature);
}

void FeatureDispatcher::dispatchAmenity(const OSMFeature &feature) {
  if (feature.geometryType == OSMGeometryType::Point)
    m_widget->addAmenityPointFeature(feature);
  else if (feature.geometryType == OSMGeometryType::Polygon)
    m_widget->addAmenityAreaFeature(feature);
}

float cross2D(const QVector2D &a, const QVector2D &b, const QVector2D &c) {
  return (b.x() - a.x()) * (c.y() - a.y()) - (b.y() - a.y()) * (c.x() - a.x());
}

bool pointInTriangle(const QVector2D &p, const QVector2D &a, const QVector2D &b,
                     const QVector2D &c) {
  constexpr float eps = 1e-7f;

  return cross2D(a, b, p) >= -eps && cross2D(b, c, p) >= -eps &&
         cross2D(c, a, p) >= -eps;
}

std::vector<std::array<int, 3>>
triangulatePolygon(const std::vector<QVector2D> &points) {
  std::vector<std::array<int, 3>> triangles;

  if (points.size() < 3)
    return triangles;

  const int n = static_cast<int>(points.size());

  double area = 0.0;

  for (int i = 0; i < n; ++i) {
    const auto &a = points[i];
    const auto &b = points[(i + 1) % n];

    area += a.x() * b.y() - b.x() * a.y();
  }

  if (std::abs(area) < 1e-10)
    return triangles;

  std::vector<int> indices(n);
  std::iota(indices.begin(), indices.end(), 0);

  if (area < 0.0)
    std::reverse(indices.begin(), indices.end());

  constexpr float eps = 1e-7f;

  while (indices.size() > 3) {
    bool clipped = false;

    for (std::size_t i = 0; i < indices.size(); ++i) {
      const int prev = indices[(i + indices.size() - 1) % indices.size()];

      const int curr = indices[i];
      const int next = indices[(i + 1) % indices.size()];

      const auto &a = points[prev];
      const auto &b = points[curr];
      const auto &c = points[next];

      if (cross2D(a, b, c) <= eps)
        continue;

      bool containsPoint = false;

      for (int index : indices) {
        if (index == prev || index == curr || index == next)
          continue;

        if (pointInTriangle(points[index], a, b, c)) {
          containsPoint = true;
          break;
        }
      }

      if (containsPoint)
        continue;

      triangles.push_back({prev, curr, next});
      indices.erase(indices.begin() + static_cast<long>(i));

      clipped = true;
      break;
    }

    if (!clipped)
      return {};
  }

  if (indices.size() == 3) {
    triangles.push_back({indices[0], indices[1], indices[2]});
  }

  return triangles;
}

bool sameGeoPoint(const GeoPoint &a, const GeoPoint &b) {
  return std::abs(a.lat - b.lat) < 1e-10 && std::abs(a.lon - b.lon) < 1e-10;
}

GLWidget::GLWidget(QWidget *parent) : QOpenGLWidget(parent) {
  setFocusPolicy(Qt::StrongFocus);
}

GLWidget::~GLWidget() {
  if (context() && context()->isValid()) {
    makeCurrent();

    vertexBuffer.destroy();
    featureBuffer.destroy();

    doneCurrent();
  }
}

// -----------------------------------------------------------------------------
// OpenGL initialization
// -----------------------------------------------------------------------------

void GLWidget::initializeGL() {
  initializeOpenGLFunctions();

  glClearColor(0.08f, 0.10f, 0.13f, 1.0f);

  glEnable(GL_DEPTH_TEST);
  glDisable(GL_CULL_FACE);

  const char *vertexShader = R"(
        attribute vec3 position;
        attribute vec3 color;

        uniform mat4 mvp;

        varying vec3 vertexColor;

        void main()
        {
            gl_Position = mvp * vec4(position, 1.0);
            vertexColor = color;
        }
    )";

  const char *fragmentShader = R"(
        varying vec3 vertexColor;

        void main()
        {
            gl_FragColor = vec4(vertexColor, 1.0);
        }
    )";

  if (!shaderProgram.addShaderFromSourceCode(QOpenGLShader::Vertex,
                                             vertexShader)) {
    qWarning() << "Vertex shader error:" << shaderProgram.log();
    return;
  }

  if (!shaderProgram.addShaderFromSourceCode(QOpenGLShader::Fragment,
                                             fragmentShader)) {
    qWarning() << "Fragment shader error:" << shaderProgram.log();
    return;
  }

  if (!shaderProgram.link()) {
    qWarning() << "Shader link error:" << shaderProgram.log();
    return;
  }

  shaderReady = true;

  vertexBuffer.create();
  vertexBuffer.setUsagePattern(QOpenGLBuffer::StaticDraw);

  featureBuffer.create();
  featureBuffer.setUsagePattern(QOpenGLBuffer::StaticDraw);

  rebuildMesh();
  rebuildFeatureMesh();
}

// -----------------------------------------------------------------------------
// Terrain mesh
// -----------------------------------------------------------------------------

void GLWidget::rebuildMesh() {
  vertices.clear();
  meshReady = false;

  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2) {
    uploadTerrainBuffer();
    update();
    return;
  }

  const std::size_t required = static_cast<std::size_t>(width) * height;

  if (dem.elevation.size() < required) {
    qWarning() << "Invalid DEM elevation array";
    uploadTerrainBuffer();
    update();
    return;
  }

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const TerrainSize terrainSize = calculateTerrainSize();

  if (terrainSize.width <= 0.0f || terrainSize.height <= 0.0f) {
    uploadTerrainBuffer();
    update();
    return;
  }

  constexpr int maxDimension = 400;

  const int stepX =
      std::max(1, (width - 1 + maxDimension - 2) / (maxDimension - 1));

  const int stepY =
      std::max(1, (height - 1 + maxDimension - 2) / (maxDimension - 1));

  auto vertexAt = [&](int x, int y, Vertex &result) -> bool {
    const float elevation =
        dem.elevation[static_cast<std::size_t>(y) * width + x];

    if (!std::isfinite(elevation))
      return false;

    const float nx =
        (static_cast<float>(x) / (width - 1) - 0.5f) * terrainSize.width;

    const double *gt = dem.geoTransform;

    const float rowDirection = gt[5] < 0.0 ? -1.0f : 1.0f;

    const float nz = -rowDirection *
                     (static_cast<float>(y) / (height - 1) - 0.5f) *
                     terrainSize.height;

    const float normalized =
        elevationRange > 0.0f
            ? std::clamp((elevation - dem.minElevation) / elevationRange, 0.0f,
                         1.0f)
            : 0.5f;

    const float vertical = elevationRange > 0.0f ? normalized * 0.12f : 0.0f;

    float r, g, b;

    if (normalized < 0.5f) {
      const float t = normalized * 2.0f;

      r = 0.12f + 0.30f * t;
      g = 0.35f + 0.30f * t;
      b = 0.12f + 0.05f * t;
    } else {
      const float t = (normalized - 0.5f) * 2.0f;

      r = 0.42f + 0.50f * t;
      g = 0.65f + 0.30f * t;
      b = 0.17f + 0.75f * t;
    }

    result = {nx, vertical, nz, r, g, b};

    return true;
  };

  for (int y = 0; y < height - 1; y += stepY) {
    const int y2 = std::min(y + stepY, height - 1);

    for (int x = 0; x < width - 1; x += stepX) {
      const int x2 = std::min(x + stepX, width - 1);

      Vertex a, b, c, d;

      if (!vertexAt(x, y, a) || !vertexAt(x2, y, b) || !vertexAt(x, y2, c) ||
          !vertexAt(x2, y2, d)) {
        continue;
      }

      vertices.insert(vertices.end(), {a, c, b, b, c, d});
    }
  }

  meshReady = !vertices.empty();

  uploadTerrainBuffer();

  qDebug() << "Terrain vertices:" << vertices.size();
}

void GLWidget::uploadTerrainBuffer() {
  if (!context() || !context()->isValid())
    return;

  makeCurrent();

  if (!vertexBuffer.isCreated())
    vertexBuffer.create();

  if (vertexBuffer.isCreated()) {
    vertexBuffer.bind();

    if (vertices.empty())
      vertexBuffer.allocate(0);
    else
      vertexBuffer.allocate(vertices.data(),
                            static_cast<int>(vertices.size() * sizeof(Vertex)));

    vertexBuffer.release();
  }

  doneCurrent();
}

// -----------------------------------------------------------------------------
// Geographic coordinates -> DEM pixels -> scene coordinates
// -----------------------------------------------------------------------------

bool GLWidget::sampleElevation(double pixelX, double pixelY,
                               float &elevation) const {
  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2)
    return false;

  if (pixelX < 0.0 || pixelY < 0.0 || pixelX > width - 1 ||
      pixelY > height - 1) {
    return false;
  }

  const int x0 = static_cast<int>(std::floor(pixelX));
  const int y0 = static_cast<int>(std::floor(pixelY));

  const int x1 = std::min(x0 + 1, width - 1);
  const int y1 = std::min(y0 + 1, height - 1);

  const float fx = static_cast<float>(pixelX - x0);
  const float fy = static_cast<float>(pixelY - y0);

  const auto at = [&](int x, int y) -> float {
    return dem.elevation[static_cast<std::size_t>(y) * width + x];
  };

  const float z00 = at(x0, y0);
  const float z10 = at(x1, y0);
  const float z01 = at(x0, y1);
  const float z11 = at(x1, y1);

  if (!std::isfinite(z00) || !std::isfinite(z10) || !std::isfinite(z01) ||
      !std::isfinite(z11)) {
    return false;
  }

  const float z0 = z00 * (1.0f - fx) + z10 * fx;
  const float z1 = z01 * (1.0f - fx) + z11 * fx;

  elevation = z0 * (1.0f - fy) + z1 * fy;

  return true;
}

bool GLWidget::geoToScene(const GeoPoint &point, ScenePoint &result) const {
  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2)
    return false;

  const double *gt = dem.geoTransform;

  const double determinant = gt[1] * gt[5] - gt[2] * gt[4];

  if (std::abs(determinant) < 1e-15)
    return false;

  const double dx = point.lon - gt[0];
  const double dy = point.lat - gt[3];

  const double pixelX = (gt[5] * dx - gt[2] * dy) / determinant - 0.5;

  const double pixelY = (-gt[4] * dx + gt[1] * dy) / determinant - 0.5;

  float elevation = 0.0f;

  if (!sampleElevation(pixelX, pixelY, elevation))
    return false;

  const TerrainSize terrainSize = calculateTerrainSize();

  const float sceneX =
      (static_cast<float>(pixelX) / (width - 1) - 0.5f) * terrainSize.width;

  const float rowDirection = gt[5] < 0.0 ? -1.0f : 1.0f;

  const float sceneZ = -rowDirection *
                       (static_cast<float>(pixelY) / (height - 1) - 0.5f) *
                       terrainSize.height;

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const float verticalScale =
      elevationRange > 0.0f ? 0.12f / elevationRange : 0.001f;

  result.position = QVector2D(sceneX, sceneZ);

  result.groundY = (elevation - dem.minElevation) * verticalScale;

  return true;
}

void GLWidget::addVertex(std::vector<Vertex> &target, const QVector2D &position,
                         float y, float r, float g, float b) {

// -----------------------------------------------------------------------------
// Building geometry: walls + roof
// -----------------------------------------------------------------------------

void GLWidget::appendBuilding(const OSMFeature &feature) {
  if (feature.geometry.size() < 3)
    return;

  std::vector<ScenePoint> scenePoints;
  scenePoints.reserve(feature.geometry.size());

  for (const GeoPoint &geo : feature.geometry) {
    ScenePoint point;

    if (!geoToScene(geo, point))
      return;

    scenePoints.push_back(point);
  }

  if (scenePoints.size() >= 2 &&
      sameGeoPoint(feature.geometry.first(), feature.geometry.last())) {
    scenePoints.pop_back();
  }

  if (scenePoints.size() < 3)
    return;

  std::vector<QVector2D> polygon;
  polygon.reserve(scenePoints.size());

  float groundSum = 0.0f;

  for (const ScenePoint &point : scenePoints) {
    polygon.push_back(point.position);
    groundSum += point.groundY;
  }

  const auto triangles = triangulatePolygon(polygon);

  if (triangles.empty())
    return;

  const float baseY = groundSum / static_cast<float>(scenePoints.size());

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const float verticalScale =
      elevationRange > 0.0f ? 0.12f / elevationRange : 0.001f;

  const float buildingHeight = feature.heightMeters > 0.0
                                   ? static_cast<float>(feature.heightMeters)
                                   : 6.0f;

  const float roofY = baseY + buildingHeight * verticalScale;

  const Colors &color = feature.colors;

  const int count = static_cast<int>(polygon.size());

  // Стены.
  for (int i = 0; i < count; ++i) {
    const int next = (i + 1) % count;

    const QVector2D &a = polygon[i];
    const QVector2D &b = polygon[next];

    addVertex(featureVertices, a, baseY, color.wallR, color.wallG, color.wallB);

    addVertex(featureVertices, b, baseY, color.wallR, color.wallG, color.wallB);

    addVertex(featureVertices, b, roofY, color.wallR, color.wallG, color.wallB);

    addVertex(featureVertices, a, baseY, color.wallR, color.wallG, color.wallB);

    addVertex(featureVertices, b, roofY, color.wallR, color.wallG, color.wallB);

    addVertex(featureVertices, a, roofY, color.wallR, color.wallG, color.wallB);
  }

  // Крыша.
  for (const auto &triangle : triangles) {
    for (int index : triangle) {
      addVertex(featureVertices, polygon[index], roofY, color.roofR,
                color.roofG, color.roofB);
    }
  }
}

// -----------------------------------------------------------------------------
// Flat polygon geometry: water, vegetation, landuse, amenity, road areas
// -----------------------------------------------------------------------------

void GLWidget::appendPolygon(const OSMFeature &feature, float heightOffset) {
  if (feature.geometry.size() < 3)
    return;

  std::vector<QVector2D> polygon;
  polygon.reserve(feature.geometry.size());

  float groundSum = 0.0f;
  int validCount = 0;

  for (const GeoPoint &geo : feature.geometry) {
    ScenePoint point;

    if (!geoToScene(geo, point))
      return;

    polygon.push_back(point.position);
    groundSum += point.groundY;
    ++validCount;
  }

  if (polygon.size() >= 2 &&
      sameGeoPoint(feature.geometry.first(), feature.geometry.last())) {
    polygon.pop_back();
  }

  if (polygon.size() < 3 || validCount == 0)
    return;

  const auto triangles = triangulatePolygon(polygon);

  if (triangles.empty())
    return;

  const float y = groundSum / static_cast<float>(validCount) + heightOffset;

  const Colors &color = feature.colors;

  for (const auto &triangle : triangles) {
    for (int index : triangle) {
      addVertex(featureVertices, polygon[index], y, color.roofR, color.roofG,
                color.roofB);
    }
  }
}

// -----------------------------------------------------------------------------
// Line geometry: roads, rivers, railway
// Each line segment becomes a flat-width quad.
// -----------------------------------------------------------------------------

void GLWidget::appendLine(const OSMFeature &feature, float widthMeters,
                          float heightOffset) {
  if (feature.geometry.size() < 2)
    return;

  std::vector<ScenePoint> points;
  points.reserve(feature.geometry.size());

  for (const GeoPoint &geo : feature.geometry) {
    ScenePoint point;

    if (!geoToScene(geo, point)) {
      // Не соединяем участки через точки вне DEM.
      points.clear();
      return;
    }

    points.push_back(point);
  }

  const TerrainSize terrainSize = calculateTerrainSize();

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double physicalWidth =
      std::hypot((dem.width - 1) * gt[1] * metersPerDegreeLon,
                 (dem.width - 1) * gt[4] * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot((dem.height - 1) * gt[2] * metersPerDegreeLon,
                 (dem.height - 1) * gt[5] * metersPerDegreeLat);

  const double maxPhysicalDimension = std::max(physicalWidth, physicalHeight);

  if (maxPhysicalDimension <= 0.0)
    return;

  const float metersToScene = static_cast<float>(2.0 / maxPhysicalDimension);

  const float halfWidth = std::max(0.0001f, widthMeters * metersToScene * 0.5f);

  const Colors &color = feature.colors;

  for (std::size_t i = 0; i + 1 < points.size(); ++i) {
    const ScenePoint &a = points[i];
    const ScenePoint &b = points[i + 1];

    const QVector2D direction = b.position - a.position;

    const float length = direction.length();

    if (length < 1e-7f)
      continue;

    const QVector2D normal(-direction.y() / length, direction.x() / length);

    const QVector2D offset = normal * halfWidth;

    const QVector2D aLeft = a.position + offset;
    const QVector2D aRight = a.position - offset;

    const QVector2D bLeft = b.position + offset;
    const QVector2D bRight = b.position - offset;

    const float yA = a.groundY + heightOffset;
    const float yB = b.groundY + heightOffset;

    // Первый треугольник.
    addVertex(featureVertices, aLeft, yA, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, aRight, yA, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, bLeft, yB, color.roofR, color.roofG,
              color.roofB);

    // Второй треугольник.
    addVertex(featureVertices, aRight, yA, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, bRight, yB, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, bLeft, yB, color.roofR, color.roofG,
              color.roofB);
  }
}

// -----------------------------------------------------------------------------
// Point geometry: amenity and other OSM nodes
// -----------------------------------------------------------------------------

void GLWidget::appendPoint(const OSMFeature &feature, float sizeMeters,
                           float heightOffset) {
  if (feature.geometry.isEmpty())
    return;

  ScenePoint point;

  if (!geoToScene(feature.geometry.first(), point))
    return;

  const TerrainSize terrainSize = calculateTerrainSize();

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double physicalWidth =
      std::hypot((dem.width - 1) * gt[1] * metersPerDegreeLon,
                 (dem.width - 1) * gt[4] * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot((dem.height - 1) * gt[2] * metersPerDegreeLon,
                 (dem.height - 1) * gt[5] * metersPerDegreeLat);

  const double maxDimension = std::max(physicalWidth, physicalHeight);

  if (maxDimension <= 0.0)
    return;

  const float halfSize =
      std::max(0.0002f, static_cast<float>(sizeMeters / maxDimension));

  const QVector2D p = point.position;

  const QVector2D a(p.x() - halfSize, p.y() - halfSize);
  const QVector2D b(p.x() + halfSize, p.y() - halfSize);
  const QVector2D c(p.x() + halfSize, p.y() + halfSize);
  const QVector2D d(p.x() - halfSize, p.y() + halfSize);

  const float y = point.groundY + heightOffset;

  const Colors &color = feature.colors;

  addVertex(featureVertices, a, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, b, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, c, y, color.roofR, color.roofG, color.roofB);

  addVertex(featureVertices, a, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, c, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, d, y, color.roofR, color.roofG, color.roofB);
}

void GLWidget::rebuildFeatureMesh() {
  featureVertices.clear();

  if (dem.width < 2 || dem.height < 2) {
    uploadFeatureBuffer();
    update();
    return;
  }

  int buildings = 0;
  int roads = 0;
  int water = 0;
  int vegetation = 0;
  int railways = 0;
  int other = 0;

  for (const OSMFeature &feature : features) {
    if (feature.geometry.isEmpty())
      continue;

    switch (feature.type) {
    case OSMFeatureType::Building:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendBuilding(feature);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 5.0f);

      ++buildings;
      break;

    case OSMFeatureType::Road:
      if (feature.geometryType == OSMGeometryType::Polygon) {
        appendPolygon(feature, 0.0015f);
      } else if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 4.0f;

        appendLine(feature, width, 0.0015f);
      }

      ++roads;
      break;

    case OSMFeatureType::Water:
      if (feature.geometryType == OSMGeometryType::Polygon) {
        appendPolygon(feature, 0.0025f);
      } else if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 5.0f;

        appendLine(feature, width, 0.0025f);
      }

      ++water;
      break;

    case OSMFeatureType::Vegetation:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.001f);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 4.0f, 0.001f);

      ++vegetation;
      break;

    case OSMFeatureType::Landuse:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.0008f);

      ++other;
      break;

    case OSMFeatureType::Railway:
      if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 4.5f;

        appendLine(feature, width, 0.002f);
      }

      ++railways;
      break;

    case OSMFeatureType::Amenity:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.0015f);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 5.0f, 0.002f);

      ++other;
      break;

    case OSMFeatureType::Unknown:
    default:
      break;
    }
  }

  uploadFeatureBuffer();

  qDebug() << "OSM mesh rebuilt:"
           << "buildings =" << buildings << "roads =" << roads
           << "water =" << water << "vegetation =" << vegetation
           << "railways =" << railways << "other =" << other
           << "vertices =" << featureVertices.size();
}

void GLWidget::uploadFeatureBuffer() {
  if (!context() || !context()->isValid())
    return;

  makeCurrent();

  if (!featureBuffer.isCreated())
    featureBuffer.create();

  if (featureBuffer.isCreated()) {
    featureBuffer.bind();

    if (featureVertices.empty())
      featureBuffer.allocate(0);
    else
      featureBuffer.allocate(
          featureVertices.data(),
          static_cast<int>(featureVertices.size() * sizeof(Vertex)));

    featureBuffer.release();
  }

  doneCurrent();
}

void GLWidget::resizeGL(int width, int height) {
  glViewport(0, 0, width, height);
}

void GLWidget::paintGL() {
  glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

  if (!shaderReady)
    return;

  QMatrix4x4 projection;

  projection.perspective(45.0f,
                         static_cast<float>(std::max(1, width())) /
                             static_cast<float>(std::max(1, height())),
                         0.01f, 100.0f);

  QMatrix4x4 view;

  view.translate(panX, panY, -zoom);

  view.rotate(rotationX, 1.0f, 0.0f, 0.0f);
  view.rotate(rotationZ, 0.0f, 1.0f, 0.0f);

  const QMatrix4x4 mvp = projection * view;

  auto drawBuffer = [&](QOpenGLBuffer &buffer, std::size_t vertexCount) {
    if (vertexCount == 0 || !buffer.isCreated())
      return;

    shaderProgram.bind();
    shaderProgram.setUniformValue("mvp", mvp);

    buffer.bind();

    const int positionLocation = shaderProgram.attributeLocation("position");

    const int colorLocation = shaderProgram.attributeLocation("color");

    shaderProgram.enableAttributeArray(positionLocation);
    shaderProgram.enableAttributeArray(colorLocation);

    shaderProgram.setAttributeBuffer(positionLocation, GL_FLOAT,
                                     offsetof(Vertex, x), 3, sizeof(Vertex));

    shaderProgram.setAttributeBuffer(colorLocation, GL_FLOAT,
                                     offsetof(Vertex, r), 3, sizeof(Vertex));

    glDrawArrays(GL_TRIANGLES, 0, static_cast<GLsizei>(vertexCount));

    shaderProgram.disableAttributeArray(positionLocation);
    shaderProgram.disableAttributeArray(colorLocation);

    buffer.release();
    shaderProgram.release();
  };

  // Сначала рельеф.
  if (meshReady)
    drawBuffer(vertexBuffer, vertices.size());

  // Затем здания и остальные OSM-объекты.
  drawBuffer(featureBuffer, featureVertices.size());
}

// -----------------------------------------------------------------------------
// Camera controls
// -----------------------------------------------------------------------------

void GLWidget::mousePressEvent(QMouseEvent *event) {
  lastMousePosition = event->position().toPoint();
}

void GLWidget::mouseMoveEvent(QMouseEvent *event) {
  const QPoint current = event->position().toPoint();

  const QPoint delta = current - lastMousePosition;

  if (event->buttons() & Qt::LeftButton) {
    rotationZ += delta.x() * 0.5f;
    rotationX += delta.y() * 0.5f;

    update();
  }

  if (event->buttons() & Qt::RightButton) {
    const float worldPerPixel =
        (2.0f * zoom * 0.41421356f) / static_cast<float>(std::max(1, height()));

    panX += delta.x() * worldPerPixel;
    panY -= delta.y() * worldPerPixel;

    update();
  }

  lastMousePosition = current;
}

void GLWidget::wheelEvent(QWheelEvent *event) {
  zoom -= event->angleDelta().y() / 120.0f * 0.25f;

  zoom = std::clamp(zoom, 0.01f, 15.0f);

  update();
}

// -----------------------------------------------------------------------------
// DEM physical dimensions
// -----------------------------------------------------------------------------

GLWidget::TerrainSize GLWidget::calculateTerrainSize() const {
  TerrainSize result;

  if (dem.width < 2 || dem.height < 2)
    return result;

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double dxLon = (dem.width - 1) * gt[1];
  const double dxLat = (dem.width - 1) * gt[4];

  const double dyLon = (dem.height - 1) * gt[2];
  const double dyLat = (dem.height - 1) * gt[5];

  const double physicalWidth =
      std::hypot(dxLon * metersPerDegreeLon, dxLat * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot(dyLon * metersPerDegreeLon, dyLat * metersPerDegreeLat);

  const double maxDimension = std::max(physicalWidth, physicalHeight);

  if (maxDimension <= 0.0)
    return result;

  result.width = static_cast<float>(2.0 * physicalWidth / maxDimension);

  result.height = static_cast<float>(2.0 * physicalHeight / maxDimension);

  return result;
}

// Загрузка локального JSON
void MainWindow::onLoadOsmJsonButtonClicked() {
  const QString fileName =
      QFileDialog::getOpenFileName(this, "Выбрать файл OSM JSON", QString(),
                                   "JSON files (*.json);;Все файлы (*)");

  if (fileName.isEmpty())
    return;

  QFile file(fileName);

  if (!file.open(QIODevice::ReadOnly)) {
    QMessageBox::warning(this, "Ошибка", "Не удалось открыть файл OSM JSON.");
    return;
  }

  const QByteArray jsonData = file.readAll();
  file.close();

  QJsonParseError parseError;

  const QJsonDocument document = QJsonDocument::fromJson(jsonData, &parseError);

  if (parseError.error != QJsonParseError::NoError) {

    QMessageBox::warning(this, "Ошибка JSON", parseError.errorString());

    return;
  }

  QJsonArray elements;

  // Формат с корневым массивом
  if (document.isArray()) {

    elements = document.array();
  }
  // Стандартный формат Overpass
  else if (document.isObject()) {

    const QJsonObject root = document.object();

    if (!root.value("elements").isArray()) {

      QMessageBox::warning(this, "Ошибка",
                           "В JSON отсутствует массив elements.");

      return;
    }

    elements = root.value("elements").toArray();
  } else {

    QMessageBox::warning(this, "Ошибка", "Некорректная структура OSM JSON.");

    return;
  }

  if (elements.isEmpty()) {

    QMessageBox::information(this, "Информация",
                             "В файле отсутствуют OSM-объекты.");

    return;
  }

  loadOSMFeatures(elements);
}

// Общая загрузка объектов OSM
void MainWindow::loadOSMFeatures(const QJsonArray &elements) {
  const QVector<OSMFeature> features = OSMDataParser::parse(elements);

  if (features.isEmpty()) {

    QMessageBox::information(this, "Информация",
                             "Не удалось найти поддерживаемые OSM-объекты.");

    return;
  }

  // Статистика
  int buildings = 0;
  int roads = 0;
  int water = 0;
  int vegetation = 0;
  int other = 0;

  for (const OSMFeature &feature : features) {

    switch (feature.type) {

    case OSMFeatureType::Building:
      ++buildings;
      break;

    case OSMFeatureType::Road:
      ++roads;
      break;

    case OSMFeatureType::Water:
      ++water;
      break;

    case OSMFeatureType::Vegetation:
      ++vegetation;
      break;

    default:
      ++other;
      break;
    }
  }

  qDebug() << "Загружено OSM объектов:" << features.size();

  qDebug() << "Здания:" << buildings;
  qDebug() << "Дороги:" << roads;
  qDebug() << "Вода:" << water;
  qDebug() << "Растительность:" << vegetation;
  qDebug() << "Другие:" << other;

  // Передаём все объекты в OpenGL
  glWidget->setFeatures(features);
}
#include "osmclassifier.hpp"

OSMFeatureType OSMClassifier::classify(const QJsonObject &tags) {
  // Здания
  if (tags.contains("building"))
    return OSMFeatureType::Building;

  // Дороги
  if (tags.contains("highway"))
    return OSMFeatureType::Road;

  // Железные дороги
  if (tags.contains("railway"))
    return OSMFeatureType::Railway;

  // Вода
  if (tags.contains("waterway") ||
      tags.value("natural").toString() == "water" || tags.contains("water")) {
    return OSMFeatureType::Water;
  }

  // Растительность
  const QString natural = tags.value("natural").toString();
  const QString landuse = tags.value("landuse").toString();
  const QString leisure = tags.value("leisure").toString();

  if (natural == "wood" || natural == "tree" || natural == "scrub" ||
      landuse == "forest" || leisure == "park") {
    return OSMFeatureType::Vegetation;
  }

  // Остальные территории
  if (tags.contains("landuse"))
    return OSMFeatureType::Landuse;

  // Общественные объекты
  if (tags.contains("amenity") || tags.contains("shop") ||
      tags.contains("office")) {
    return OSMFeatureType::Amenity;
  }

  return OSMFeatureType::Unknown;
}

QByteArray OSMDataLoader::fetchOSMData(const QString &queryType,
                                       const int &timeout, const QRect &area) {
  Q_UNUSED(area);

  // Формируем Overpass QL
  QString finalQuery = buildOverpassQuery(timeout, queryType);

  if (finalQuery.isEmpty()) {
    qDebug() << "Неизвестный тип запроса:" << queryType;
    return "";
  }

  // Границы области:
  // south, west, north, east
  const double minLat = defaultArea.lat - defaultArea.width;
  const double maxLat = defaultArea.lat + defaultArea.width;
  const double minLon = defaultArea.lon - defaultArea.height;
  const double maxLon = defaultArea.lon + defaultArea.height;

  const QString bbox = QString("%1,%2,%3,%4")
                           .arg(minLat, 0, 'f', 6)
                           .arg(minLon, 0, 'f', 6)
                           .arg(maxLat, 0, 'f', 6)
                           .arg(maxLon, 0, 'f', 6);

  // Подставляем реальные координаты вместо {{bbox}}
  finalQuery.replace("{{bbox}}", bbox);

  qDebug() << "Тип запроса:" << queryType;
  qDebug() << "BBox:" << bbox;
  qDebug() << "Весь запрос:" << finalQuery;

  QNetworkRequest request(url);

  request.setHeader(QNetworkRequest::ContentTypeHeader,
                    "application/x-www-form-urlencoded");
  request.setHeader(
      QNetworkRequest::UserAgentHeader,
      "BMSTU-NIR-OSM-Loader/1.0 (+https://github.com/Andrey8066)");
  request.setRawHeader("Accept", "application/json");

  QByteArray postData = "data=";
  postData += QUrl::toPercentEncoding(finalQuery);

  reply = manager->post(request, postData);

  connect(reply, &QNetworkReply::finished, this, [this, area, queryType]() {
    const QVariant status =
        reply->attribute(QNetworkRequest::HttpStatusCodeAttribute);

    qDebug() << "HTTP status:" << status;

    if (reply->error() != QNetworkReply::NoError) {

      qDebug() << "Ошибка загрузки OSM данных:" << reply->errorString();

      const QByteArray serverResponse = reply->readAll();

      qDebug() << "Ответ сервера:" << serverResponse;

      emit errorOccurred("Не удалось загрузить данные из OSM");

      reply->deleteLater();
      return;
    }

    requestResult = reply->readAll();

    QJsonParseError parseError;

    const QJsonDocument doc =
        QJsonDocument::fromJson(requestResult, &parseError);

    if (parseError.error != QJsonParseError::NoError) {

      qDebug() << "Ошибка парсинга JSON:" << parseError.errorString();

      qDebug() << "Ответ сервера:" << requestResult;

      reply->deleteLater();
      return;
    }

    qDebug() << "OSM JSON успешно получен.";

    qDebug() << "Получено байт:" << requestResult.size();

    const QJsonObject root = doc.object();
    const QJsonArray elements = root["elements"].toArray();

    emit dataLoaded(elements, queryType);
    int ways = 0;
    int relations = 0;
    int nodes = 0;

    int buildings = 0;
    int buildingParts = 0;

    for (const QJsonValue &value : elements) {
      QJsonObject object = value.toObject();
      QJsonObject tags = object["tags"].toObject();

      QString type = object["type"].toString();

      if (type == "way")
        ways++;

      if (type == "relation")
        relations++;

      if (type == "node")
        nodes++;

      if (tags.contains("building"))
        buildings++;

      if (tags.contains("building:part"))
        buildingParts++;
    }

    qDebug() << "========== OSM STATISTICS ==========";
    qDebug() << "Total elements:" << elements.size();
    qDebug() << "Ways:" << ways;
    qDebug() << "Relations:" << relations;
    qDebug() << "Nodes:" << nodes;
    qDebug() << "Building tags:" << buildings;
    qDebug() << "Building parts:" << buildingParts;

    int wayBuildings = 0;
    int relationBuildings = 0;

    int wayParts = 0;
    int relationParts = 0;

    for (const QJsonValue &value : elements) {
      QJsonObject object = value.toObject();
      QJsonObject tags = object["tags"].toObject();

      QString type = object["type"].toString();

      bool hasBuilding = tags.contains("building");
      bool hasPart = tags.contains("building:part");

      if (type == "way" && hasBuilding)
        wayBuildings++;

      if (type == "relation" && hasBuilding)
        relationBuildings++;

      if (type == "way" && hasPart)
        wayParts++;

      if (type == "relation" && hasPart)
        relationParts++;
    }

    qDebug() << "Way buildings:" << wayBuildings;
    qDebug() << "Relation buildings:" << relationBuildings;

    qDebug() << "Way building parts:" << wayParts;
    qDebug() << "Relation building parts:" << relationParts;
  });

  return requestResult;
}

  defaultArea.lon = lon;

// Формирование Overpass QL запроса - правильная syntax

QString OSMDataLoader::buildOverpassQuery(const int &timeout,
                                          const QString &queryType) {
  if (queryType == "buildings") {
    return QString(R"(
            [out:json][timeout:%1];

            (
                nwr["building"]({{bbox}});
                nwr["building:part"]({{bbox}});
                relation["type"="building"]({{bbox}});
            );

            out geom;
        )")
        .arg(timeout);
  }

  if (queryType == "all") {
    return QString(R"([out:json][timeout:%1];
                   (
                   nwr["building"]({{bbox}});
                   nwr["highway"]({{bbox}});
                   );
                   out geom;
                   )")
        .arg(timeout);
  }

  qWarning() << "Неизвестный тип запроса:" << queryType;
  return QString();
}

// Сохранение JSON в файл
QString OSMDataLoader::saveToJSONFile(const QJsonArray &data,
                                      const QString &filename) {
  QFile file(filename);

  if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
    qDebug() << "Ошибка открытия файла для записи:" << filename;
    return "";
  }

  file.write(QJsonDocument(data).toJson());
  file.close();

  return filename;
}

  if (!reply) {
    qDebug() << "Нет активного запроса";

  }

  if (!reply->isFinished()) {
    QEventLoop loop;

    connect(reply, &QNetworkReply::finished, &loop, &QEventLoop::quit);

    loop.exec();
  }

  qDebug() << "waitData(): запрос завершён";
}

double OSMDataLoader::radToLat(double radius) {
  return radius / (earthRadius * M_PI) * 180;
}
double OSMDataLoader::radToLon(double radius) {
  const double latRadians = defaultArea.lat * M_PI / 180.0;

  return radius / (earthRadius * std::cos(latRadians)) * 180.0 / M_PI;
}

double parseLength(const QJsonValue &value, bool *ok) {
  *ok = false;

  double result = 0.0;

  if (value.isDouble()) {
    result = value.toDouble();
    *ok = true;
    return result;
  }

  QString text = value.toString().trimmed().toLower();

  bool feet = text.endsWith("ft");

  if (feet)
    text.chop(2);
  else if (text.endsWith("m"))
    text.chop(1);

  result = text.trimmed().toDouble(ok);

  if (feet && *ok)
    result *= 0.3048;

  return result;
}

float parseBuildingHeight(const QJsonObject &tags) {
  bool ok = false;

  double height = parseLength(tags.value("height"), &ok);

  if (ok && height > 0.0)
    return static_cast<float>(std::clamp(height, 2.0, 300.0));

  const double levels = tags.value("building:levels").toString().toDouble(&ok);

  if (ok && levels > 0.0) {
    return static_cast<float>(std::clamp(levels * 3.0, 2.0, 300.0));
  }

  return 9.0f;
}

float parseRoadWidth(const QJsonObject &tags) {
  bool ok = false;

  double width = parseLength(tags.value("width"), &ok);

  if (ok && width > 0.0)
    return static_cast<float>(width);

  const double lanes = tags.value("lanes").toString().toDouble(&ok);

  if (ok && lanes > 0.0)
    return static_cast<float>(lanes * 3.5);

  const QString highway = tags.value("highway").toString();

  if (highway == "motorway" || highway == "trunk")
    return 12.0f;

  if (highway == "primary" || highway == "secondary")
    return 9.0f;

  if (highway == "residential")
    return 6.0f;

  if (highway == "footway" || highway == "path" || highway == "cycleway")
    return 2.0f;

  return 5.0f;
}

Colors getBuildingColors(const QJsonObject &tags) {
  const QString type = tags.value("building").toString();

  if (type == "industrial") {
    return Colors(0.55f, 0.55f, 0.55f, 0.35f, 0.35f, 0.35f);
  }

  if (type == "commercial" || type == "retail") {
    return Colors(0.72f, 0.62f, 0.48f, 0.35f, 0.32f, 0.28f);
  }

  if (type == "house") {
    return Colors(0.75f, 0.65f, 0.52f, 0.45f, 0.25f, 0.20f);
  }

  return Colors(0.58f, 0.63f, 0.70f, 0.28f, 0.36f, 0.28f);
}

bool samePoint(const GeoPoint &a, const GeoPoint &b) {
  constexpr double eps = 1e-9;

  return std::abs(a.lat - b.lat) < eps && std::abs(a.lon - b.lon) < eps;
}

// Читает координаты одного way из участника relation.
QVector<GeoPoint> readMemberGeometry(const QJsonObject &member) {
  QVector<GeoPoint> points;

  const QJsonArray geometry = member.value("geometry").toArray();

  for (const QJsonValue &value : geometry) {
    const QJsonObject point = value.toObject();

    if (!point.contains("lat") || !point.contains("lon"))
      continue;

    points.push_back(
        {point.value("lat").toDouble(), point.value("lon").toDouble()});
  }

  return points;
}

// Собирает внешние кольца relation из нескольких way.
QVector<QVector<GeoPoint>> readOuterRings(const QJsonObject &object) {
  QVector<QVector<GeoPoint>> segments;

  const QJsonArray members = object.value("members").toArray();

  for (const QJsonValue &value : members) {
    const QJsonObject member = value.toObject();

    if (member.value("type").toString() != "way")
      continue;

    if (member.value("role").toString() != "outer")
      continue;

    QVector<GeoPoint> points = readMemberGeometry(member);

    if (points.size() >= 2)
      segments.push_back(std::move(points));
  }

  QVector<QVector<GeoPoint>> rings;

  while (!segments.isEmpty()) {
    QVector<GeoPoint> ring = segments.takeFirst();

    bool changed = true;

    while (changed && !segments.isEmpty()) {
      changed = false;

      for (int i = 0; i < segments.size(); ++i) {
        QVector<GeoPoint> segment = segments[i];

        if (samePoint(ring.back(), segment.front())) {
          for (int j = 1; j < segment.size(); ++j)
            ring.push_back(segment[j]);

        } else if (samePoint(ring.back(), segment.back())) {
          std::reverse(segment.begin(), segment.end());

          for (int j = 1; j < segment.size(); ++j)
            ring.push_back(segment[j]);

        } else if (samePoint(ring.front(), segment.back())) {
          for (int j = 0; j < segment.size() - 1; ++j)
            ring.prepend(segment[j]);

        } else if (samePoint(ring.front(), segment.front())) {
          std::reverse(segment.begin(), segment.end());

          for (int j = 0; j < segment.size() - 1; ++j)
            ring.prepend(segment[j]);

        } else {
          continue;
        }

        segments.removeAt(i);
        changed = true;
        break;
      }
    }

    // Незамкнутые контуры не передаём рендереру.
    if (ring.size() < 4 || !samePoint(ring.front(), ring.back()))
      continue;

    // Убираем повтор последней точки.
    ring.removeLast();

    if (ring.size() >= 3)
      rings.push_back(std::move(ring));
  }

  return rings;
}

bool readGeometry(const QJsonObject &object, OSMFeature &feature) {
  const QString osmType = object.value("type").toString();

  if (osmType == "node") {

    if (!object.contains("lat") || !object.contains("lon")) {
      return false;
    }

    feature.geometryType = OSMGeometryType::Point;

    feature.geometry.push_back(
        {object.value("lat").toDouble(), object.value("lon").toDouble()});

    return true;
  }

  if (osmType == "way") {

    const QJsonArray geometry = object.value("geometry").toArray();

    for (const QJsonValue &value : geometry) {

      const QJsonObject point = value.toObject();

      if (!point.contains("lat") || !point.contains("lon")) {
        continue;
      }

      feature.geometry.push_back(
          {point.value("lat").toDouble(), point.value("lon").toDouble()});
    }

    if (feature.geometry.isEmpty())
      return false;

    const bool closed =
        feature.geometry.size() >= 4 &&
        std::abs(feature.geometry.front().lat - feature.geometry.back().lat) <
            1e-10 &&
        std::abs(feature.geometry.front().lon - feature.geometry.back().lon) <
            1e-10;

    const bool area = feature.tags.value("area").toString() == "yes" ||
                      (feature.tags.contains("building") && closed) ||
                      (feature.tags.contains("landuse") && closed) ||
                      (feature.tags.contains("leisure") && closed) ||
                      (feature.tags.contains("natural") && closed) ||
                      (feature.tags.contains("water") && closed);

    if (area) {
      feature.geometryType = OSMGeometryType::Polygon;

      if (closed)
        feature.geometry.pop_back();
    } else {
      feature.geometryType = OSMGeometryType::LineString;
    }

    return true;
  }

  // Relation пока не обрабатываем
  return false;
}

QVector<OSMFeature> OSMDataParser::parse(const QJsonArray &elements) {
  QVector<OSMFeature> result;

  result.reserve(elements.size());

  // ID way, которые уже войдут в геометрию relation.
  QSet<qint64> relationWayIds;

  // Первый проход: собираем ID внешних way поддерживаемых relation.
  for (const QJsonValue &value : elements) {
    if (!value.isObject())
      continue;

    const QJsonObject object = value.toObject();

    if (object.value("type").toString() != "relation")
      continue;

    const QJsonObject tags = object.value("tags").toObject();
    const QString relationType = tags.value("type").toString();

    if (relationType != "multipolygon" && relationType != "boundary" &&
        relationType != "building") {
      continue;
    }

    const QJsonArray members = object.value("members").toArray();

    for (const QJsonValue &memberValue : members) {
      const QJsonObject member = memberValue.toObject();

      if (member.value("type").toString() != "way")
        continue;

      if (member.value("role").toString() != "outer")
        continue;

      relationWayIds.insert(
          static_cast<qint64>(member.value("ref").toDouble()));
    }
  }

  // Второй проход: обрабатываем объекты.
  for (const QJsonValue &value : elements) {
    if (!value.isObject())
      continue;

    const QJsonObject object = value.toObject();

    const QString osmType = object.value("type").toString();

    // Не обрабатываем way повторно, если он уже вошёл в relation.
    if (osmType == "way") {
      const qint64 wayId = static_cast<qint64>(object.value("id").toDouble());

      if (relationWayIds.contains(wayId))
        continue;
    }

    OSMFeature feature;

    feature.id = static_cast<qint64>(object.value("id").toDouble());

    feature.tags = object.value("tags").toObject();

    feature.type = OSMClassifier::classify(feature.tags);

    if (feature.type == OSMFeatureType::Unknown)
      continue;

    // Relation обрабатываем отдельно.
    if (osmType == "relation") {
      const QString relationType = feature.tags.value("type").toString();

      if (relationType != "multipolygon" && relationType != "boundary" &&
          relationType != "building") {
        continue;
      }

      const auto rings = readOuterRings(object);

      for (const auto &ring : rings) {
        OSMFeature relationFeature = feature;

        relationFeature.geometryType = OSMGeometryType::Polygon;

        relationFeature.geometry = ring;

        if (relationFeature.type == OSMFeatureType::Building) {
          relationFeature.heightMeters =
              parseBuildingHeight(relationFeature.tags);

          relationFeature.colors = getBuildingColors(relationFeature.tags);
        }

        result.push_back(std::move(relationFeature));
      }

      continue;
    }

    // Обычные node и way.
    if (!readGeometry(object, feature))
      continue;

    if (feature.type == OSMFeatureType::Building) {
      feature.heightMeters = parseBuildingHeight(feature.tags);
      feature.colors = getBuildingColors(feature.tags);
    }

    if (feature.type == OSMFeatureType::Road) {
      feature.widthMeters = parseRoadWidth(feature.tags);
    }

    result.push_back(std::move(feature));
  }

  return result;
}

bool DEMReader::loadDem(const QString &fileName, DEMData &out) {
  GDALAllRegister();

  GDALDataset *dataset = static_cast<GDALDataset *>(
      GDALOpen(fileName.toLocal8Bit().constData(), GA_ReadOnly));

  if (!dataset)
    return false;

  const int width = dataset->GetRasterXSize();
  const int height = dataset->GetRasterYSize();

  if (width < 2 || height < 2 || dataset->GetRasterCount() < 1) {
    GDALClose(dataset);
    return false;
  }

  GDALRasterBand *band = dataset->GetRasterBand(1);

  out.width = width;
  out.height = height;

  qDebug() << "Размер DEM:" << width << "x" << height;

  qDebug() << "Projection:" << dataset->GetProjectionRef();

  qDebug() << "Raster type:" << GDALGetDataTypeName(band->GetRasterDataType());

  int hasNoData = 0;
  double noData = band->GetNoDataValue(&hasNoData);

  qDebug() << "NoData exists:" << hasNoData << "value:" << noData;

  qDebug() << "Scale:" << band->GetScale() << "Offset:" << band->GetOffset();

  if (dataset->GetGeoTransform(out.geoTransform) != CE_None) {
    GDALClose(dataset);

    const char *projection = dataset->GetProjectionRef();

    out.projectionWkt = projection ? QString::fromUtf8(projection) : QString();

    return false;
  }

  out.elevation.resize(static_cast<std::size_t>(width) * height);

  const CPLErr result =
      band->RasterIO(GF_Read, 0, 0, width, height, out.elevation.data(), width,
                     height, GDT_Float32, 0, 0);

  if (result != CE_None) {
    GDALClose(dataset);
    out = DEMData{};
    return false;
  }

  float minZ = std::numeric_limits<float>::max();
  float maxZ = std::numeric_limits<float>::lowest();

  for (float &z : out.elevation) {
    if (!std::isfinite(z) || (hasNoData && z == static_cast<float>(noData))) {
      // Для первого прототипа заполняем пропуски нулём.
      // Для точной модели NoData лучше интерполировать.
      z = 0.0f;
    }

    minZ = std::min(minZ, z);
    maxZ = std::max(maxZ, z);
  }

  out.minElevation = minZ;
  out.maxElevation = maxZ;

  GDALClose(dataset);
  return true;
}

FeatureDispatcher::FeatureDispatcher(GLWidget *widget) : m_widget(widget) {}

void FeatureDispatcher::dispatch(const OSMFeature &feature) {
  if (!m_widget) {
    qWarning() << "FeatureDispatcher: GLWidget is null";
    return;
  }

  if (feature.geometry.isEmpty())
    return;

  switch (feature.type) {
  case OSMFeatureType::Building:
    dispatchBuilding(feature);
    break;

  case OSMFeatureType::Road:
    dispatchRoad(feature);
    break;

  case OSMFeatureType::Water:
    dispatchWater(feature);
    break;

  case OSMFeatureType::Vegetation:
    dispatchVegetation(feature);
    break;

  case OSMFeatureType::Landuse:
    dispatchLanduse(feature);
    break;

  case OSMFeatureType::Railway:
    dispatchRailway(feature);
    break;

  case OSMFeatureType::Amenity:
    dispatchAmenity(feature);
    break;

  case OSMFeatureType::Unknown:
  default:
    break;
  }
}

void FeatureDispatcher::dispatchAll(const QVector<OSMFeature> &features) {
  for (const OSMFeature &feature : features)
    dispatch(feature);
}

float cross2D(const QVector2D &a, const QVector2D &b, const QVector2D &c) {
  return (b.x() - a.x()) * (c.y() - a.y()) - (b.y() - a.y()) * (c.x() - a.x());
}

bool pointInTriangle(const QVector2D &p, const QVector2D &a, const QVector2D &b,
                     const QVector2D &c) {
  constexpr float eps = 1e-7f;

  return cross2D(a, b, p) >= -eps && cross2D(b, c, p) >= -eps &&
         cross2D(c, a, p) >= -eps;
}

std::vector<std::array<int, 3>>
triangulatePolygon(const std::vector<QVector2D> &points) {
  std::vector<std::array<int, 3>> triangles;

  if (points.size() < 3)
    return triangles;

  const int n = static_cast<int>(points.size());

  double area = 0.0;

  for (int i = 0; i < n; ++i) {
    const auto &a = points[i];
    const auto &b = points[(i + 1) % n];

    area += a.x() * b.y() - b.x() * a.y();
  }

  if (std::abs(area) < 1e-10)
    return triangles;

  std::vector<int> indices(n);
  std::iota(indices.begin(), indices.end(), 0);

  if (area < 0.0)
    std::reverse(indices.begin(), indices.end());

  constexpr float eps = 1e-7f;

  while (indices.size() > 3) {
    bool clipped = false;

    for (std::size_t i = 0; i < indices.size(); ++i) {
      const int prev = indices[(i + indices.size() - 1) % indices.size()];

      const int curr = indices[i];
      const int next = indices[(i + 1) % indices.size()];

      const auto &a = points[prev];
      const auto &b = points[curr];
      const auto &c = points[next];

      if (cross2D(a, b, c) <= eps)
        continue;

      bool containsPoint = false;

      for (int index : indices) {
        if (index == prev || index == curr || index == next)
          continue;

        if (pointInTriangle(points[index], a, b, c)) {
          containsPoint = true;
          break;
        }
      }

      if (containsPoint)
        continue;

      triangles.push_back({prev, curr, next});
      indices.erase(indices.begin() + static_cast<long>(i));

      clipped = true;
      break;
    }

    if (!clipped)
      return {};
  }

  if (indices.size() == 3) {
    triangles.push_back({indices[0], indices[1], indices[2]});
  }

  return triangles;
}

void GLWidget::initializeGL() {
  initializeOpenGLFunctions();

  glClearColor(0.08f, 0.10f, 0.13f, 1.0f);

  glEnable(GL_DEPTH_TEST);
  glDisable(GL_CULL_FACE);

  const char *vertexShader = R"(
        attribute vec3 position;
        attribute vec3 color;

        uniform mat4 mvp;

        varying vec3 vertexColor;

        void main()
        {
            gl_Position = mvp * vec4(position, 1.0);
            vertexColor = color;
        }
    )";

  const char *fragmentShader = R"(
        varying vec3 vertexColor;

        void main()
        {
            gl_FragColor = vec4(vertexColor, 1.0);
        }
    )";

  if (!shaderProgram.addShaderFromSourceCode(QOpenGLShader::Vertex,
                                             vertexShader)) {
    qWarning() << "Vertex shader error:" << shaderProgram.log();
    return;
  }

  if (!shaderProgram.addShaderFromSourceCode(QOpenGLShader::Fragment,
                                             fragmentShader)) {
    qWarning() << "Fragment shader error:" << shaderProgram.log();
    return;
  }

  if (!shaderProgram.link()) {
    qWarning() << "Shader link error:" << shaderProgram.log();
    return;
  }

  shaderReady = true;

  vertexBuffer.create();
  vertexBuffer.setUsagePattern(QOpenGLBuffer::StaticDraw);

  featureBuffer.create();
  featureBuffer.setUsagePattern(QOpenGLBuffer::StaticDraw);

  rebuildMesh();
  rebuildFeatureMesh();
}

// -----------------------------------------------------------------------------
// Terrain mesh
// -----------------------------------------------------------------------------

void GLWidget::rebuildMesh() {
  vertices.clear();
  meshReady = false;

  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2) {
    uploadTerrainBuffer();
    update();
    return;
  }

  const std::size_t required = static_cast<std::size_t>(width) * height;

  if (dem.elevation.size() < required) {
    qWarning() << "Invalid DEM elevation array";
    uploadTerrainBuffer();
    update();
    return;
  }

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const TerrainSize terrainSize = calculateTerrainSize();

  if (terrainSize.width <= 0.0f || terrainSize.height <= 0.0f) {
    uploadTerrainBuffer();
    update();
    return;
  }

  constexpr int maxDimension = 400;

  const int stepX =
      std::max(1, (width - 1 + maxDimension - 2) / (maxDimension - 1));

  const int stepY =
      std::max(1, (height - 1 + maxDimension - 2) / (maxDimension - 1));

  auto vertexAt = [&](int x, int y, Vertex &result) -> bool {
    const float elevation =
        dem.elevation[static_cast<std::size_t>(y) * width + x];

    if (!std::isfinite(elevation))
      return false;

    const float nx =
        (static_cast<float>(x) / (width - 1) - 0.5f) * terrainSize.width;

    const double *gt = dem.geoTransform;

    const float rowDirection = gt[5] < 0.0 ? -1.0f : 1.0f;

    const float nz = -rowDirection *
                     (static_cast<float>(y) / (height - 1) - 0.5f) *
                     terrainSize.height;

    const float normalized =
        elevationRange > 0.0f
            ? std::clamp((elevation - dem.minElevation) / elevationRange, 0.0f,
                         1.0f)
            : 0.5f;

    const float vertical = elevationRange > 0.0f ? normalized * 0.12f : 0.0f;

    float r, g, b;

    if (normalized < 0.5f) {
      const float t = normalized * 2.0f;

      r = 0.12f + 0.30f * t;
      g = 0.35f + 0.30f * t;
      b = 0.12f + 0.05f * t;
    } else {
      const float t = (normalized - 0.5f) * 2.0f;

      r = 0.42f + 0.50f * t;
      g = 0.65f + 0.30f * t;
      b = 0.17f + 0.75f * t;
    }

    result = {nx, vertical, nz, r, g, b};

    return true;
  };

  for (int y = 0; y < height - 1; y += stepY) {
    const int y2 = std::min(y + stepY, height - 1);

    for (int x = 0; x < width - 1; x += stepX) {
      const int x2 = std::min(x + stepX, width - 1);

      Vertex a, b, c, d;

      if (!vertexAt(x, y, a) || !vertexAt(x2, y, b) || !vertexAt(x, y2, c) ||
          !vertexAt(x2, y2, d)) {
        continue;
      }

      vertices.insert(vertices.end(), {a, c, b, b, c, d});
    }
  }

  meshReady = !vertices.empty();

  uploadTerrainBuffer();

  qDebug() << "Terrain vertices:" << vertices.size();
}

void GLWidget::uploadTerrainBuffer() {
  if (!context() || !context()->isValid())
    return;

  makeCurrent();

  if (!vertexBuffer.isCreated())
    vertexBuffer.create();

  if (vertexBuffer.isCreated()) {
    vertexBuffer.bind();

    if (vertices.empty())
      vertexBuffer.allocate(0);
    else
      vertexBuffer.allocate(vertices.data(),
                            static_cast<int>(vertices.size() * sizeof(Vertex)));

    vertexBuffer.release();
  }

  doneCurrent();
}

// -----------------------------------------------------------------------------
// Geographic coordinates -> DEM pixels -> scene coordinates
// -----------------------------------------------------------------------------

bool GLWidget::sampleElevation(double pixelX, double pixelY,
                               float &elevation) const {
  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2)
    return false;

  if (pixelX < 0.0 || pixelY < 0.0 || pixelX > width - 1 ||
      pixelY > height - 1) {
    return false;
  }

  const int x0 = static_cast<int>(std::floor(pixelX));
  const int y0 = static_cast<int>(std::floor(pixelY));

  const int x1 = std::min(x0 + 1, width - 1);
  const int y1 = std::min(y0 + 1, height - 1);

  const float fx = static_cast<float>(pixelX - x0);
  const float fy = static_cast<float>(pixelY - y0);

  const auto at = [&](int x, int y) -> float {
    return dem.elevation[static_cast<std::size_t>(y) * width + x];
  };

  const float z00 = at(x0, y0);
  const float z10 = at(x1, y0);
  const float z01 = at(x0, y1);
  const float z11 = at(x1, y1);

  if (!std::isfinite(z00) || !std::isfinite(z10) || !std::isfinite(z01) ||
      !std::isfinite(z11)) {
    return false;
  }

  const float z0 = z00 * (1.0f - fx) + z10 * fx;
  const float z1 = z01 * (1.0f - fx) + z11 * fx;

  elevation = z0 * (1.0f - fy) + z1 * fy;

  return true;
}

bool GLWidget::geoToScene(const GeoPoint &point, ScenePoint &result) const {
  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2)
    return false;

  const double *gt = dem.geoTransform;

  const double determinant = gt[1] * gt[5] - gt[2] * gt[4];

  if (std::abs(determinant) < 1e-15)
    return false;

  const double dx = point.lon - gt[0];
  const double dy = point.lat - gt[3];

  const double pixelX = (gt[5] * dx - gt[2] * dy) / determinant - 0.5;

  const double pixelY = (-gt[4] * dx + gt[1] * dy) / determinant - 0.5;

  float elevation = 0.0f;

  if (!sampleElevation(pixelX, pixelY, elevation))
    return false;

  const TerrainSize terrainSize = calculateTerrainSize();

  const float sceneX =
      (static_cast<float>(pixelX) / (width - 1) - 0.5f) * terrainSize.width;

  const float rowDirection = gt[5] < 0.0 ? -1.0f : 1.0f;

  const float sceneZ = -rowDirection *
                       (static_cast<float>(pixelY) / (height - 1) - 0.5f) *
                       terrainSize.height;

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const float verticalScale =
      elevationRange > 0.0f ? 0.12f / elevationRange : 0.001f;

  result.position = QVector2D(sceneX, sceneZ);

  result.groundY = (elevation - dem.minElevation) * verticalScale;

  return true;
}

// -----------------------------------------------------------------------------
// Common vertex helper
// -----------------------------------------------------------------------------

void GLWidget::addVertex(std::vector<Vertex> &target, const QVector2D &position,
                         float y, float r, float g, float b) {
  target.push_back({position.x(), y, position.y(), r, g, b});
}

// -----------------------------------------------------------------------------
// Building geometry: walls + roof
// -----------------------------------------------------------------------------

void GLWidget::appendBuilding(const OSMFeature &feature) {
  if (feature.geometry.size() < 3)
    return;

  // Преобразует географическое кольцо в координаты сцены.
  auto convertRing = [&](const QVector<GeoPoint> &geoRing,
                         std::vector<ScenePoint> &sceneRing) -> bool {
    sceneRing.clear();

    for (const GeoPoint &geo : geoRing) {
      ScenePoint point;

      if (!geoToScene(geo, point))
        return false;

      sceneRing.push_back(point);
    }

    // Убираем повтор последней точки замкнутого кольца.
    if (sceneRing.size() >= 2 &&
        sameGeoPoint(geoRing.first(), geoRing.last())) {
      sceneRing.pop_back();
    }

    return sceneRing.size() >= 3;
  };

  std::vector<ScenePoint> outerScene;

  if (!convertRing(feature.geometry, outerScene))
    return;

  // Все кольца: сначала outer, затем inner.
  std::vector<std::vector<ScenePoint>> sceneRings;
  sceneRings.push_back(outerScene);

  for (const auto &innerGeo : feature.innerRings) {
    std::vector<ScenePoint> innerScene;

    if (!convertRing(innerGeo, innerScene))
      continue;

    sceneRings.push_back(std::move(innerScene));
  }

  // Формат для Mapbox Earcut.
  std::vector<std::vector<std::array<double, 2>>> polygon;

  // Плоский массив координат для последующего получения вершин крыши.
  std::vector<QVector2D> roofPoints;

  for (const auto &ring : sceneRings) {
    std::vector<std::array<double, 2>> earcutRing;

    for (const ScenePoint &point : ring) {
      earcutRing.push_back({static_cast<double>(point.position.x()),
                            static_cast<double>(point.position.y())});

      roofPoints.push_back(point.position);
    }

    polygon.push_back(std::move(earcutRing));
  }

  if (roofPoints.size() < 3)
    return;

  const std::vector<unsigned int> indices =
      mapbox::earcut<unsigned int>(polygon);

  if (indices.empty())
    return;

  // Высота основания по внешнему контуру.
  float groundSum = 0.0f;

  for (const ScenePoint &point : outerScene)
    groundSum += point.groundY;

  const float baseY = groundSum / static_cast<float>(outerScene.size());

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const float verticalScale =
      elevationRange > 0.0f ? 0.12f / elevationRange : 0.001f;

  const float buildingHeight =
      feature.heightMeters > 0.0f ? feature.heightMeters : 6.0f;

  const float baseBuildingY = baseY + feature.minHeightMeters * verticalScale;

  const float roofY = baseBuildingY + buildingHeight * verticalScale;

  const Colors &color = feature.colors;

  // Стены внешнего и внутренних контуров.
  for (const auto &ring : sceneRings) {
    const int count = static_cast<int>(ring.size());

    for (int i = 0; i < count; ++i) {
      const QVector2D &a = ring[i].position;
      const QVector2D &b = ring[(i + 1) % count].position;

      addVertex(featureVertices, a, baseBuildingY, color.wallR, color.wallG,
                color.wallB);

      addVertex(featureVertices, b, baseBuildingY, color.wallR, color.wallG,
                color.wallB);

      addVertex(featureVertices, b, roofY, color.wallR, color.wallG,
                color.wallB);

      addVertex(featureVertices, a, baseBuildingY, color.wallR, color.wallG,
                color.wallB);

      addVertex(featureVertices, b, roofY, color.wallR, color.wallG,
                color.wallB);

      addVertex(featureVertices, a, roofY, color.wallR, color.wallG,
                color.wallB);
    }
  }

  // Крыша с отверстиями.
  for (unsigned int index : indices) {
    if (index >= roofPoints.size())
      continue;

    addVertex(featureVertices, roofPoints[index], roofY, color.roofR,
              color.roofG, color.roofB);
  }
}

// -----------------------------------------------------------------------------
// Flat polygon geometry: water, vegetation, landuse, amenity, road areas
// -----------------------------------------------------------------------------

void GLWidget::appendPolygon(const OSMFeature &feature, float heightOffset) {
  if (feature.geometry.size() < 3)
    return;

  std::vector<QVector2D> polygon;
  polygon.reserve(feature.geometry.size());

  float groundSum = 0.0f;
  int validCount = 0;

  for (const GeoPoint &geo : feature.geometry) {
    ScenePoint point;

    if (!geoToScene(geo, point))
      return;

    polygon.push_back(point.position);
    groundSum += point.groundY;
    ++validCount;
  }

  if (polygon.size() >= 2 &&
      sameGeoPoint(feature.geometry.first(), feature.geometry.last())) {
    polygon.pop_back();
  }

  if (polygon.size() < 3 || validCount == 0)
    return;

  const auto triangles = triangulatePolygon(polygon);

  if (triangles.empty())
    return;

  const float y = groundSum / static_cast<float>(validCount) + heightOffset;

  const Colors &color = feature.colors;

  for (const auto &triangle : triangles) {
    for (int index : triangle) {
      addVertex(featureVertices, polygon[index], y, color.roofR, color.roofG,
                color.roofB);
    }
  }
}

// -----------------------------------------------------------------------------
// Line geometry: roads, rivers, railway
// Each line segment becomes a flat-width quad.
// -----------------------------------------------------------------------------

void GLWidget::appendLine(const OSMFeature &feature, float widthMeters,
                          float heightOffset) {
  if (feature.geometry.size() < 2)
    return;

  std::vector<ScenePoint> points;
  points.reserve(feature.geometry.size());

  for (const GeoPoint &geo : feature.geometry) {
    ScenePoint point;

    if (!geoToScene(geo, point)) {
      // Не соединяем участки через точки вне DEM.
      points.clear();
      return;
    }

    points.push_back(point);
  }

  const TerrainSize terrainSize = calculateTerrainSize();

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double physicalWidth =
      std::hypot((dem.width - 1) * gt[1] * metersPerDegreeLon,
                 (dem.width - 1) * gt[4] * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot((dem.height - 1) * gt[2] * metersPerDegreeLon,
                 (dem.height - 1) * gt[5] * metersPerDegreeLat);

  const double maxPhysicalDimension = std::max(physicalWidth, physicalHeight);

  if (maxPhysicalDimension <= 0.0)
    return;

  const float metersToScene = static_cast<float>(2.0 / maxPhysicalDimension);

  const float halfWidth = std::max(0.0001f, widthMeters * metersToScene * 0.5f);

  const Colors &color = feature.colors;

  for (std::size_t i = 0; i + 1 < points.size(); ++i) {
    const ScenePoint &a = points[i];
    const ScenePoint &b = points[i + 1];

    const QVector2D direction = b.position - a.position;

    const float length = direction.length();

    if (length < 1e-7f)
      continue;

    const QVector2D normal(-direction.y() / length, direction.x() / length);

    const QVector2D offset = normal * halfWidth;

    const QVector2D aLeft = a.position + offset;
    const QVector2D aRight = a.position - offset;

    const QVector2D bLeft = b.position + offset;
    const QVector2D bRight = b.position - offset;

    const float yA = a.groundY + heightOffset;
    const float yB = b.groundY + heightOffset;

    // Первый треугольник.
    addVertex(featureVertices, aLeft, yA, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, aRight, yA, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, bLeft, yB, color.roofR, color.roofG,
              color.roofB);

    // Второй треугольник.
    addVertex(featureVertices, aRight, yA, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, bRight, yB, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, bLeft, yB, color.roofR, color.roofG,
              color.roofB);
  }
}

// -----------------------------------------------------------------------------
// Point geometry: amenity and other OSM nodes
// -----------------------------------------------------------------------------

void GLWidget::appendPoint(const OSMFeature &feature, float sizeMeters,
                           float heightOffset) {
  if (feature.geometry.isEmpty())
    return;

  ScenePoint point;

  if (!geoToScene(feature.geometry.first(), point))
    return;

  const TerrainSize terrainSize = calculateTerrainSize();

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double physicalWidth =
      std::hypot((dem.width - 1) * gt[1] * metersPerDegreeLon,
                 (dem.width - 1) * gt[4] * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot((dem.height - 1) * gt[2] * metersPerDegreeLon,
                 (dem.height - 1) * gt[5] * metersPerDegreeLat);

  const double maxDimension = std::max(physicalWidth, physicalHeight);

  if (maxDimension <= 0.0)
    return;

  const float halfSize =
      std::max(0.0002f, static_cast<float>(sizeMeters / maxDimension));

  const QVector2D p = point.position;

  const QVector2D a(p.x() - halfSize, p.y() - halfSize);
  const QVector2D b(p.x() + halfSize, p.y() - halfSize);
  const QVector2D c(p.x() + halfSize, p.y() + halfSize);
  const QVector2D d(p.x() - halfSize, p.y() + halfSize);

  const float y = point.groundY + heightOffset;

  const Colors &color = feature.colors;

  addVertex(featureVertices, a, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, b, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, c, y, color.roofR, color.roofG, color.roofB);

  addVertex(featureVertices, a, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, c, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, d, y, color.roofR, color.roofG, color.roofB);
}

// -----------------------------------------------------------------------------
// OSM feature dispatcher
// -----------------------------------------------------------------------------

void GLWidget::rebuildFeatureMesh() {
  featureVertices.clear();

  if (dem.width < 2 || dem.height < 2) {
    uploadFeatureBuffer();
    update();
    return;
  }

  int buildings = 0;
  int roads = 0;
  int water = 0;
  int vegetation = 0;
  int railways = 0;
  int other = 0;

  for (const OSMFeature &feature : features) {
    if (feature.geometry.isEmpty())
      continue;

    switch (feature.type) {
    case OSMFeatureType::Building:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendBuilding(feature);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 5.0f);

      ++buildings;
      break;

    case OSMFeatureType::Road:
      if (feature.geometryType == OSMGeometryType::Polygon) {
        appendPolygon(feature, 0.0015f);
      } else if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 4.0f;

        appendLine(feature, width, 0.0015f);
      }

      ++roads;
      break;

    case OSMFeatureType::Water:
      if (feature.geometryType == OSMGeometryType::Polygon) {
        appendPolygon(feature, 0.0025f);
      } else if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 5.0f;

        appendLine(feature, width, 0.0025f);
      }

      ++water;
      break;

    case OSMFeatureType::Vegetation:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.001f);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 4.0f, 0.001f);

      ++vegetation;
      break;

    case OSMFeatureType::Landuse:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.0008f);

      ++other;
      break;

    case OSMFeatureType::Railway:
      if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 4.5f;

        appendLine(feature, width, 0.002f);
      }

      ++railways;
      break;

    case OSMFeatureType::Amenity:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.0015f);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 5.0f, 0.002f);

      ++other;
      break;

    case OSMFeatureType::Unknown:
    default:
      break;
    }
  }

  uploadFeatureBuffer();

  qDebug() << "OSM mesh rebuilt:"
           << "buildings =" << buildings << "roads =" << roads
           << "water =" << water << "vegetation =" << vegetation
           << "railways =" << railways << "other =" << other
           << "vertices =" << featureVertices.size();
}

void GLWidget::uploadFeatureBuffer() {
  if (!context() || !context()->isValid())
    return;

  makeCurrent();

  if (!featureBuffer.isCreated())
    featureBuffer.create();

  if (featureBuffer.isCreated()) {
    featureBuffer.bind();

    if (featureVertices.empty())
      featureBuffer.allocate(0);
    else
      featureBuffer.allocate(
          featureVertices.data(),
          static_cast<int>(featureVertices.size() * sizeof(Vertex)));

    featureBuffer.release();
  }

  doneCurrent();
}

// -----------------------------------------------------------------------------
// Rendering
// -----------------------------------------------------------------------------

void GLWidget::resizeGL(int width, int height) {
  glViewport(0, 0, width, height);
}

void GLWidget::paintGL() {
  glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

  if (!shaderReady)
    return;

  QMatrix4x4 projection;

  projection.perspective(45.0f,
                         static_cast<float>(std::max(1, width())) /
                             static_cast<float>(std::max(1, height())),
                         0.01f, 100.0f);

  QMatrix4x4 view;

  view.translate(panX, panY, -zoom);

  view.rotate(rotationX, 1.0f, 0.0f, 0.0f);
  view.rotate(rotationZ, 0.0f, 1.0f, 0.0f);

  const QMatrix4x4 mvp = projection * view;

  auto drawBuffer = [&](QOpenGLBuffer &buffer, std::size_t vertexCount) {
    if (vertexCount == 0 || !buffer.isCreated())
      return;

    shaderProgram.bind();
    shaderProgram.setUniformValue("mvp", mvp);

    buffer.bind();

    const int positionLocation = shaderProgram.attributeLocation("position");

    const int colorLocation = shaderProgram.attributeLocation("color");

    shaderProgram.enableAttributeArray(positionLocation);
    shaderProgram.enableAttributeArray(colorLocation);

    shaderProgram.setAttributeBuffer(positionLocation, GL_FLOAT,
                                     offsetof(Vertex, x), 3, sizeof(Vertex));

    shaderProgram.setAttributeBuffer(colorLocation, GL_FLOAT,
                                     offsetof(Vertex, r), 3, sizeof(Vertex));

    glDrawArrays(GL_TRIANGLES, 0, static_cast<GLsizei>(vertexCount));

    shaderProgram.disableAttributeArray(positionLocation);
    shaderProgram.disableAttributeArray(colorLocation);

    buffer.release();
    shaderProgram.release();
  };

  // Сначала рельеф.
  if (meshReady)
    drawBuffer(vertexBuffer, vertices.size());

  // Затем здания и остальные OSM-объекты.
  drawBuffer(featureBuffer, featureVertices.size());
}

// -----------------------------------------------------------------------------
// Camera controls
// -----------------------------------------------------------------------------

void GLWidget::mousePressEvent(QMouseEvent *event) {
  lastMousePosition = event->position().toPoint();
}

void GLWidget::mouseMoveEvent(QMouseEvent *event) {
  const QPoint current = event->position().toPoint();

  const QPoint delta = current - lastMousePosition;

  if (event->buttons() & Qt::LeftButton) {
    rotationZ += delta.x() * 0.5f;
    rotationX += delta.y() * 0.5f;

    update();
  }

  if (event->buttons() & Qt::RightButton) {
    const float worldPerPixel =
        (2.0f * zoom * 0.41421356f) / static_cast<float>(std::max(1, height()));

    panX += delta.x() * worldPerPixel;
    panY -= delta.y() * worldPerPixel;

    update();
  }

  lastMousePosition = current;
}

void GLWidget::wheelEvent(QWheelEvent *event) {
  zoom -= event->angleDelta().y() / 120.0f * 0.25f;

  zoom = std::clamp(zoom, 0.01f, 15.0f);

  update();
}

GLWidget::TerrainSize GLWidget::calculateTerrainSize() const {
  TerrainSize result;

  if (dem.width < 2 || dem.height < 2)
    return result;

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double dxLon = (dem.width - 1) * gt[1];
  const double dxLat = (dem.width - 1) * gt[4];

  const double dyLon = (dem.height - 1) * gt[2];
  const double dyLat = (dem.height - 1) * gt[5];

  const double physicalWidth =
      std::hypot(dxLon * metersPerDegreeLon, dxLat * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot(dyLon * metersPerDegreeLon, dyLat * metersPerDegreeLat);

  const double maxDimension = std::max(physicalWidth, physicalHeight);

  if (maxDimension <= 0.0)
    return result;

  result.width = static_cast<float>(2.0 * physicalWidth / maxDimension);

  result.height = static_cast<float>(2.0 * physicalHeight / maxDimension);

  return result;
}

// Загрузка локального JSON
void MainWindow::onLoadOsmJsonButtonClicked() {
  const QString fileName =
      QFileDialog::getOpenFileName(this, "Выбрать файл OSM JSON", QString(),
                                   "JSON files (*.json);;Все файлы (*)");

  if (fileName.isEmpty())
    return;

  QFile file(fileName);

  if (!file.open(QIODevice::ReadOnly)) {
    QMessageBox::warning(this, "Ошибка", "Не удалось открыть файл OSM JSON.");
    return;
  }

  const QByteArray jsonData = file.readAll();
  file.close();

  QJsonParseError parseError;

  const QJsonDocument document = QJsonDocument::fromJson(jsonData, &parseError);

  if (parseError.error != QJsonParseError::NoError) {

    QMessageBox::warning(this, "Ошибка JSON", parseError.errorString());

    return;
  }

  QJsonArray elements;

  // Формат с корневым массивом
  if (document.isArray()) {

    elements = document.array();
  }
  // Стандартный формат Overpass
  else if (document.isObject()) {

    const QJsonObject root = document.object();

    if (!root.value("elements").isArray()) {

      QMessageBox::warning(this, "Ошибка",
                           "В JSON отсутствует массив elements.");

      return;
    }

    elements = root.value("elements").toArray();
  } else {

    QMessageBox::warning(this, "Ошибка", "Некорректная структура OSM JSON.");

    return;
  }

  if (elements.isEmpty()) {

    QMessageBox::information(this, "Информация",
                             "В файле отсутствуют OSM-объекты.");

    return;
  }

  loadOSMFeatures(elements);
}

// Общая загрузка объектов OSM
void MainWindow::loadOSMFeatures(const QJsonArray &elements) {
  const QVector<OSMFeature> features = OSMDataParser::parse(elements);

  if (features.isEmpty()) {

    QMessageBox::information(this, "Информация",
                             "Не удалось найти поддерживаемые OSM-объекты.");

    return;
  }

  // Статистика
  int buildings = 0;
  int roads = 0;
  int water = 0;
  int vegetation = 0;
  int other = 0;

  for (const OSMFeature &feature : features) {

    switch (feature.type) {

    case OSMFeatureType::Building:
      ++buildings;
      break;

    case OSMFeatureType::Road:
      ++roads;
      break;

    case OSMFeatureType::Water:
      ++water;
      break;

    case OSMFeatureType::Vegetation:
      ++vegetation;
      break;

    default:
      ++other;
      break;
    }
  }

  qDebug() << "Загружено OSM объектов:" << features.size();

  qDebug() << "Здания:" << buildings;
  qDebug() << "Дороги:" << roads;
  qDebug() << "Вода:" << water;
  qDebug() << "Растительность:" << vegetation;
  qDebug() << "Другие:" << other;

  // Передаём все объекты в OpenGL
  glWidget->setFeatures(features);
}
#include "osmclassifier.hpp"

OSMFeatureType OSMClassifier::classify(const QJsonObject &tags) {
  // Здания
  if (tags.contains("building"))
    return OSMFeatureType::Building;

  // Дороги
  if (tags.contains("highway"))
    return OSMFeatureType::Road;

  // Железные дороги
  if (tags.contains("railway"))
    return OSMFeatureType::Railway;

  // Вода
  if (tags.contains("waterway") ||
      tags.value("natural").toString() == "water" || tags.contains("water")) {
    return OSMFeatureType::Water;
  }

  // Растительность
  const QString natural = tags.value("natural").toString();
  const QString landuse = tags.value("landuse").toString();
  const QString leisure = tags.value("leisure").toString();

  if (natural == "wood" || natural == "tree" || natural == "scrub" ||
      landuse == "forest" || leisure == "park") {
    return OSMFeatureType::Vegetation;
  }

  // Остальные территории
  if (tags.contains("landuse"))
    return OSMFeatureType::Landuse;

  // Общественные объекты
  if (tags.contains("amenity") || tags.contains("shop") ||
      tags.contains("office")) {
    return OSMFeatureType::Amenity;
  }

  return OSMFeatureType::Unknown;
}

QByteArray OSMDataLoader::fetchOSMData(const QString &queryType,
                                       const int &timeout, const QRect &area) {
  Q_UNUSED(area);

  // Формируем Overpass QL
  QString finalQuery = buildOverpassQuery(timeout, queryType);

  if (finalQuery.isEmpty()) {
    qDebug() << "Неизвестный тип запроса:" << queryType;
    return "";
  }

  // Границы области:
  // south, west, north, east
  const double minLat = defaultArea.lat - defaultArea.width;
  const double maxLat = defaultArea.lat + defaultArea.width;
  const double minLon = defaultArea.lon - defaultArea.height;
  const double maxLon = defaultArea.lon + defaultArea.height;

  const QString bbox = QString("%1,%2,%3,%4")
                           .arg(minLat, 0, 'f', 6)
                           .arg(minLon, 0, 'f', 6)
                           .arg(maxLat, 0, 'f', 6)
                           .arg(maxLon, 0, 'f', 6);

  // Подставляем реальные координаты вместо {{bbox}}
  finalQuery.replace("{{bbox}}", bbox);

  qDebug() << "Тип запроса:" << queryType;
  qDebug() << "BBox:" << bbox;
  qDebug() << "Весь запрос:" << finalQuery;

  QNetworkRequest request(url);

  request.setHeader(QNetworkRequest::ContentTypeHeader,
                    "application/x-www-form-urlencoded");
  request.setHeader(
      QNetworkRequest::UserAgentHeader,
      "BMSTU-NIR-OSM-Loader/1.0 (+https://github.com/Andrey8066)");
  request.setRawHeader("Accept", "application/json");

  QByteArray postData = "data=";
  postData += QUrl::toPercentEncoding(finalQuery);

  reply = manager->post(request, postData);

  connect(reply, &QNetworkReply::finished, this, [this, area, queryType]() {
    const QVariant status =
        reply->attribute(QNetworkRequest::HttpStatusCodeAttribute);

    qDebug() << "HTTP status:" << status;

    if (reply->error() != QNetworkReply::NoError) {

      qDebug() << "Ошибка загрузки OSM данных:" << reply->errorString();

      const QByteArray serverResponse = reply->readAll();

      qDebug() << "Ответ сервера:" << serverResponse;

      emit errorOccurred("Не удалось загрузить данные из OSM");

      reply->deleteLater();
      return;
    }

    requestResult = reply->readAll();

    QJsonParseError parseError;

    const QJsonDocument doc =
        QJsonDocument::fromJson(requestResult, &parseError);

    if (parseError.error != QJsonParseError::NoError) {

      qDebug() << "Ошибка парсинга JSON:" << parseError.errorString();

      qDebug() << "Ответ сервера:" << requestResult;

      reply->deleteLater();
      return;
    }

    qDebug() << "OSM JSON успешно получен.";

    qDebug() << "Получено байт:" << requestResult.size();

    const QJsonObject root = doc.object();
    const QJsonArray elements = root["elements"].toArray();

    emit dataLoaded(elements, queryType);
    int ways = 0;
    int relations = 0;
    int nodes = 0;

    int buildings = 0;
    int buildingParts = 0;

    for (const QJsonValue &value : elements) {
      QJsonObject object = value.toObject();
      QJsonObject tags = object["tags"].toObject();

      QString type = object["type"].toString();

      if (type == "way")
        ways++;

      if (type == "relation")
        relations++;

      if (type == "node")
        nodes++;

      if (tags.contains("building"))
        buildings++;

      if (tags.contains("building:part"))
        buildingParts++;
    }

    qDebug() << "========== OSM STATISTICS ==========";
    qDebug() << "Total elements:" << elements.size();
    qDebug() << "Ways:" << ways;
    qDebug() << "Relations:" << relations;
    qDebug() << "Nodes:" << nodes;
    qDebug() << "Building tags:" << buildings;
    qDebug() << "Building parts:" << buildingParts;

    int wayBuildings = 0;
    int relationBuildings = 0;

    int wayParts = 0;
    int relationParts = 0;

    for (const QJsonValue &value : elements) {
      QJsonObject object = value.toObject();
      QJsonObject tags = object["tags"].toObject();

      QString type = object["type"].toString();

      bool hasBuilding = tags.contains("building");
      bool hasPart = tags.contains("building:part");

      if (type == "way" && hasBuilding)
        wayBuildings++;

      if (type == "relation" && hasBuilding)
        relationBuildings++;

      if (type == "way" && hasPart)
        wayParts++;

      if (type == "relation" && hasPart)
        relationParts++;
    }

    qDebug() << "Way buildings:" << wayBuildings;
    qDebug() << "Relation buildings:" << relationBuildings;

    qDebug() << "Way building parts:" << wayParts;
    qDebug() << "Relation building parts:" << relationParts;
  });

  return requestResult;
}

// Формирование Overpass QL запроса - правильная syntax

QString OSMDataLoader::buildOverpassQuery(const int &timeout,
                                          const QString &queryType) {
  if (queryType == "buildings") {
    return QString(R"(
            [out:json][timeout:%1];

            (
                nwr["building"]({{bbox}});
                nwr["building:part"]({{bbox}});
                relation["type"="building"]({{bbox}});
            );

            out geom;
        )")
        .arg(timeout);
  }

  if (queryType == "all") {
    return QString(R"([out:json][timeout:%1];
                   (
                   nwr["building"]({{bbox}});
                   nwr["highway"]({{bbox}});
                   );
                   out geom;
                   )")
        .arg(timeout);
  }

  qWarning() << "Неизвестный тип запроса:" << queryType;
  return QString();
}

// Сохранение JSON в файл
QString OSMDataLoader::saveToJSONFile(const QJsonArray &data,
                                      const QString &filename) {
  QFile file(filename);

  if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
    qDebug() << "Ошибка открытия файла для записи:" << filename;

  }

  file.close();

  return filename;
}

void OSMDataLoader::waitData() {
  if (!reply) {
    qDebug() << "Нет активного запроса";
    return;
  }

  if (!reply->isFinished()) {
    QEventLoop loop;

    connect(reply, &QNetworkReply::finished, &loop, &QEventLoop::quit);

    loop.exec();
  }

  qDebug() << "waitData(): запрос завершён";
}

double OSMDataLoader::radToLat(double radius) {
  return radius / (earthRadius * M_PI) * 180;
}
double OSMDataLoader::radToLon(double radius) {
  const double latRadians = defaultArea.lat * M_PI / 180.0;

  return radius / (earthRadius * std::cos(latRadians)) * 180.0 / M_PI;
}

double parseLength(const QJsonValue &value, bool *ok) {
  *ok = false;

  double result = 0.0;

  if (value.isDouble()) {
    result = value.toDouble();
    *ok = true;
    return result;
  }

  QString text = value.toString().trimmed().toLower();

  bool feet = text.endsWith("ft");

  if (feet)
    text.chop(2);
  else if (text.endsWith("m"))
    text.chop(1);

  result = text.trimmed().toDouble(ok);

  if (feet && *ok)
    result *= 0.3048;

  return result;
}

float parseBuildingHeight(const QJsonObject &tags) {
  bool ok = false;

  double height = parseLength(tags.value("height"), &ok);

  if (ok && height > 0.0)
    return static_cast<float>(std::clamp(height, 2.0, 300.0));

  const double levels = tags.value("building:levels").toString().toDouble(&ok);

  if (ok && levels > 0.0) {
    return static_cast<float>(std::clamp(levels * 3.0, 2.0, 300.0));
  }

  return 9.0f;
}

float parseRoadWidth(const QJsonObject &tags) {
  bool ok = false;

  double width = parseLength(tags.value("width"), &ok);

  if (ok && width > 0.0)
    return static_cast<float>(width);

  const double lanes = tags.value("lanes").toString().toDouble(&ok);

  if (ok && lanes > 0.0)
    return static_cast<float>(lanes * 3.5);

  const QString highway = tags.value("highway").toString();

  if (highway == "motorway" || highway == "trunk")
    return 12.0f;

  if (highway == "primary" || highway == "secondary")
    return 9.0f;

  if (highway == "residential")
    return 6.0f;

  if (highway == "footway" || highway == "path" || highway == "cycleway")
    return 2.0f;

  return 5.0f;
}

bool samePoint(const GeoPoint &a, const GeoPoint &b) {
  constexpr double eps = 1e-9;

  return std::abs(a.lat - b.lat) < eps && std::abs(a.lon - b.lon) < eps;
}

bool pointInRing(const GeoPoint &point, const QVector<GeoPoint> &ring) {
  bool inside = false;

  if (ring.size() < 3)
    return false;

  for (int i = 0, j = ring.size() - 1; i < ring.size(); j = i++) {

    const GeoPoint &a = ring[i];
    const GeoPoint &b = ring[j];

    const bool intersects =
        ((a.lat > point.lat) != (b.lat > point.lat)) &&
        (point.lon <
         (b.lon - a.lon) * (point.lat - a.lat) / (b.lat - a.lat) + a.lon);

    if (intersects)
      inside = !inside;
  }

  return inside;
}

// Читает координаты одного way из участника relation.
QVector<GeoPoint> readMemberGeometry(const QJsonObject &member) {
  QVector<GeoPoint> points;

  const QJsonArray geometry = member.value("geometry").toArray();

  for (const QJsonValue &value : geometry) {
    const QJsonObject point = value.toObject();

    if (!point.contains("lat") || !point.contains("lon"))
      continue;

    points.push_back(
        {point.value("lat").toDouble(), point.value("lon").toDouble()});
  }

  return points;
}

QVector<QVector<GeoPoint>> readRings(const QJsonObject &object,
                                     const QString &role) {
  QVector<QVector<GeoPoint>> segments;

  const QJsonArray members = object.value("members").toArray();

  for (const QJsonValue &value : members) {
    const QJsonObject member = value.toObject();

    if (member.value("type").toString() != "way")
      continue;

    if (member.value("role").toString() != role)
      continue;

    QVector<GeoPoint> points = readMemberGeometry(member);

    if (points.size() >= 2)
      segments.push_back(std::move(points));
  }

  QVector<QVector<GeoPoint>> rings;

  while (!segments.isEmpty()) {
    QVector<GeoPoint> ring = segments.takeFirst();

    bool changed = true;

    while (changed && !segments.isEmpty()) {
      changed = false;

      for (int i = 0; i < segments.size(); ++i) {
        QVector<GeoPoint> segment = segments[i];

        if (samePoint(ring.back(), segment.front())) {
          for (int j = 1; j < segment.size(); ++j)
            ring.push_back(segment[j]);

        } else if (samePoint(ring.back(), segment.back())) {
          std::reverse(segment.begin(), segment.end());

          for (int j = 1; j < segment.size(); ++j)
            ring.push_back(segment[j]);

        } else if (samePoint(ring.front(), segment.back())) {
          // Вставляем в начало в обратном порядке.
          for (int j = segment.size() - 2; j >= 0; --j)
            ring.prepend(segment[j]);

        } else if (samePoint(ring.front(), segment.front())) {
          std::reverse(segment.begin(), segment.end());

          for (int j = segment.size() - 2; j >= 0; --j)
            ring.prepend(segment[j]);

        } else {
          continue;
        }

        segments.removeAt(i);
        changed = true;
        break;
      }
    }

    if (ring.size() < 4 || !samePoint(ring.front(), ring.back()))
      continue;

    ring.removeLast();

    if (ring.size() >= 3)
      rings.push_back(std::move(ring));
  }

  return rings;
}

bool readGeometry(const QJsonObject &object, OSMFeature &feature) {
  const QString osmType = object.value("type").toString();

  if (osmType == "node") {

    if (!object.contains("lat") || !object.contains("lon")) {
      return false;
    }

    feature.geometryType = OSMGeometryType::Point;

    feature.geometry.push_back(
        {object.value("lat").toDouble(), object.value("lon").toDouble()});

    return true;
  }

  if (osmType == "way") {

    const QJsonArray geometry = object.value("geometry").toArray();

    for (const QJsonValue &value : geometry) {

      const QJsonObject point = value.toObject();

      if (!point.contains("lat") || !point.contains("lon")) {
        continue;
      }

      feature.geometry.push_back(
          {point.value("lat").toDouble(), point.value("lon").toDouble()});
    }

    if (feature.geometry.isEmpty())
      return false;

    const bool closed =
        feature.geometry.size() >= 4 &&
        std::abs(feature.geometry.front().lat - feature.geometry.back().lat) <
            1e-10 &&
        std::abs(feature.geometry.front().lon - feature.geometry.back().lon) <
            1e-10;

    const bool area = feature.tags.value("area").toString() == "yes" ||
                      (feature.tags.contains("building") && closed) ||
                      (feature.tags.contains("landuse") && closed) ||
                      (feature.tags.contains("leisure") && closed) ||
                      (feature.tags.contains("natural") && closed) ||
                      (feature.tags.contains("water") && closed);

    if (area) {
      feature.geometryType = OSMGeometryType::Polygon;

      if (closed)
        feature.geometry.pop_back();
    } else {
      feature.geometryType = OSMGeometryType::LineString;
    }

    return true;
  }

  // Relation пока не обрабатываем
  return false;
}

QVector<OSMFeature> OSMDataParser::parse(const QJsonArray &elements) {
  QVector<OSMFeature> result;

  result.reserve(elements.size());

  // ID way, которые уже войдут в геометрию relation.
  QSet<qint64> relationWayIds;

  // Первый проход: собираем ID внешних way поддерживаемых relation.
  for (const QJsonValue &value : elements) {
    if (!value.isObject())
      continue;

    const QJsonObject object = value.toObject();

    if (object.value("type").toString() != "relation")
      continue;

    const QJsonObject tags = object.value("tags").toObject();
    const QString relationType = tags.value("type").toString();

    if (relationType != "multipolygon" && relationType != "boundary" &&
        relationType != "building") {
      continue;
    }

    const QJsonArray members = object.value("members").toArray();

    for (const QJsonValue &memberValue : members) {
      const QJsonObject member = memberValue.toObject();

      if (member.value("type").toString() != "way")
        continue;

      const QString role = member.value("role").toString();

      const bool isOuter = role == "outer";
      const bool isBuildingInner =
          OSMClassifier::classify(tags) == OSMFeatureType::Building &&
          role == "inner";

      if (!isOuter && !isBuildingInner)
        continue;
      relationWayIds.insert(
          static_cast<qint64>(member.value("ref").toDouble()));
    }
  }

  // Второй проход: обрабатываем объекты.
  for (const QJsonValue &value : elements) {
    if (!value.isObject())
      continue;

    const QJsonObject object = value.toObject();

    const QString osmType = object.value("type").toString();

    // Не обрабатываем way повторно, если он уже вошёл в relation.
    if (osmType == "way") {
      const qint64 wayId = static_cast<qint64>(object.value("id").toDouble());

      if (relationWayIds.contains(wayId))
        continue;
    }

    OSMFeature feature;

    feature.id = static_cast<qint64>(object.value("id").toDouble());

    feature.tags = object.value("tags").toObject();

    feature.type = OSMClassifier::classify(feature.tags);

    if (feature.type == OSMFeatureType::Unknown)
      continue;

    // Relation обрабатываем отдельно.
    if (osmType == "relation") {
      const QString relationType = feature.tags.value("type").toString();

      if (relationType != "multipolygon" && relationType != "boundary" &&
          relationType != "building") {
        continue;
      }

      // 1. Собираем внешние кольца relation.
      const auto outerRings = readRings(object, "outer");

      // 2. Внутренние кольца нужны только зданиям.
      QVector<QVector<GeoPoint>> innerRings;

      if (feature.type == OSMFeatureType::Building) {
        innerRings = readRings(object, "inner");
      }

      // 3. Создаём отдельный OSMFeature для каждого внешнего кольца.
      for (const auto &outer : outerRings) {
        OSMFeature relationFeature = feature;

        relationFeature.geometryType = OSMGeometryType::Polygon;
        relationFeature.geometry = outer;

        if (relationFeature.type == OSMFeatureType::Building) {
          relationFeature.heightMeters =
              parseBuildingHeight(relationFeature.tags);
          feature.minHeightMeters = 0.0f;

          const QString minLevel =
              feature.tags.value("building:min_level").toString();

          if (!minLevel.isEmpty()) {
            bool ok = false;
            const float levels = minLevel.toFloat(&ok);

            if (ok)
              feature.minHeightMeters = levels * 3.0f;
          }

          relationFeature.colors = getBuildingColors(relationFeature.tags);

          // 4. Прикрепляем к зданию только те inner,
          // которые находятся внутри текущего outer.
          for (const auto &inner : innerRings) {
            if (inner.isEmpty())
              continue;

            if (pointInRing(inner.first(), outer)) {
              relationFeature.innerRings.push_back(inner);
            }
          }
        }

        result.push_back(std::move(relationFeature));
      }

      continue;
    }

    // Обычные node и way.
    if (!readGeometry(object, feature))
      continue;

    if (feature.type == OSMFeatureType::Building) {
      feature.heightMeters = parseBuildingHeight(feature.tags);
      feature.minHeightMeters = 0.0f;

      const QString minLevel =
          feature.tags.value("building:min_level").toString();

      if (!minLevel.isEmpty()) {
        bool ok = false;
        const float levels = minLevel.toFloat(&ok);

        if (ok)
          feature.minHeightMeters = levels * 3.0f;
      }
      feature.colors = getBuildingColors(feature.tags);
    }

    if (feature.type == OSMFeatureType::Road) {
      feature.widthMeters = parseRoadWidth(feature.tags);
    }

    result.push_back(std::move(feature));
  }

  return result;
}

bool DEMReader::loadDem(const QString &fileName, DEMData &out) {
  GDALAllRegister();

  GDALDataset *dataset = static_cast<GDALDataset *>(
      GDALOpen(fileName.toLocal8Bit().constData(), GA_ReadOnly));

  if (!dataset)
    return false;

  const int width = dataset->GetRasterXSize();
  const int height = dataset->GetRasterYSize();

  if (width < 2 || height < 2 || dataset->GetRasterCount() < 1) {
    GDALClose(dataset);
    return false;
  }

  GDALRasterBand *band = dataset->GetRasterBand(1);

  out.width = width;
  out.height = height;

  qDebug() << "Размер DEM:" << width << "x" << height;

  qDebug() << "Projection:" << dataset->GetProjectionRef();

  qDebug() << "Raster type:" << GDALGetDataTypeName(band->GetRasterDataType());

  int hasNoData = 0;
  double noData = band->GetNoDataValue(&hasNoData);

  qDebug() << "NoData exists:" << hasNoData << "value:" << noData;

  qDebug() << "Scale:" << band->GetScale() << "Offset:" << band->GetOffset();

  if (dataset->GetGeoTransform(out.geoTransform) != CE_None) {
    GDALClose(dataset);

    const char *projection = dataset->GetProjectionRef();

    out.projectionWkt = projection ? QString::fromUtf8(projection) : QString();

    return false;
  }

  out.elevation.resize(static_cast<std::size_t>(width) * height);

  const CPLErr result =
      band->RasterIO(GF_Read, 0, 0, width, height, out.elevation.data(), width,
                     height, GDT_Float32, 0, 0);

  if (result != CE_None) {
    GDALClose(dataset);
    out = DEMData{};
    return false;
  }

  float minZ = std::numeric_limits<float>::max();
  float maxZ = std::numeric_limits<float>::lowest();

  for (float &z : out.elevation) {
    if (!std::isfinite(z) || (hasNoData && z == static_cast<float>(noData))) {
      // Для первого прототипа заполняем пропуски нулём.
      // Для точной модели NoData лучше интерполировать.
      z = 0.0f;
    }

    minZ = std::min(minZ, z);
    maxZ = std::max(maxZ, z);
  }

  out.minElevation = minZ;
  out.maxElevation = maxZ;

  GDALClose(dataset);
  return true;
}

FeatureDispatcher::FeatureDispatcher(GLWidget *widget) : m_widget(widget) {}

void FeatureDispatcher::dispatch(const OSMFeature &feature) {
  if (!m_widget) {
    qWarning() << "FeatureDispatcher: GLWidget is null";
    return;
  }

  if (feature.geometry.isEmpty())
    return;

  switch (feature.type) {
  case OSMFeatureType::Building:
    dispatchBuilding(feature);
    break;

  case OSMFeatureType::Road:
    dispatchRoad(feature);
    break;

  case OSMFeatureType::Water:
    dispatchWater(feature);
    break;

  case OSMFeatureType::Vegetation:
    dispatchVegetation(feature);
    break;

  case OSMFeatureType::Landuse:
    dispatchLanduse(feature);
    break;

  case OSMFeatureType::Railway:
    dispatchRailway(feature);
    break;

  case OSMFeatureType::Amenity:
    dispatchAmenity(feature);
    break;

  case OSMFeatureType::Unknown:
  default:
    break;
  }
}

void FeatureDispatcher::dispatchAll(const QVector<OSMFeature> &features) {
  for (const OSMFeature &feature : features)
    dispatch(feature);
}

float cross2D(const QVector2D &a, const QVector2D &b, const QVector2D &c) {
  return (b.x() - a.x()) * (c.y() - a.y()) - (b.y() - a.y()) * (c.x() - a.x());
}

bool pointInTriangle(const QVector2D &p, const QVector2D &a, const QVector2D &b,
                     const QVector2D &c) {
  constexpr float eps = 1e-7f;

  return cross2D(a, b, p) >= -eps && cross2D(b, c, p) >= -eps &&
         cross2D(c, a, p) >= -eps;
}

std::vector<std::array<int, 3>>
triangulatePolygon(const std::vector<QVector2D> &points) {
  std::vector<std::array<int, 3>> triangles;

  if (points.size() < 3)
    return triangles;

  const int n = static_cast<int>(points.size());

  double area = 0.0;

  for (int i = 0; i < n; ++i) {
    const auto &a = points[i];
    const auto &b = points[(i + 1) % n];

    area += a.x() * b.y() - b.x() * a.y();
  }

  if (std::abs(area) < 1e-10)
    return triangles;

  std::vector<int> indices(n);
  std::iota(indices.begin(), indices.end(), 0);

  if (area < 0.0)
    std::reverse(indices.begin(), indices.end());

  constexpr float eps = 1e-7f;

  while (indices.size() > 3) {
    bool clipped = false;

    for (std::size_t i = 0; i < indices.size(); ++i) {
      const int prev = indices[(i + indices.size() - 1) % indices.size()];

      const int curr = indices[i];
      const int next = indices[(i + 1) % indices.size()];

      const auto &a = points[prev];
      const auto &b = points[curr];
      const auto &c = points[next];

      if (cross2D(a, b, c) <= eps)
        continue;

      bool containsPoint = false;

      for (int index : indices) {
        if (index == prev || index == curr || index == next)
          continue;

        if (pointInTriangle(points[index], a, b, c)) {
          containsPoint = true;
          break;
        }
      }

      if (containsPoint)
        continue;

      triangles.push_back({prev, curr, next});
      indices.erase(indices.begin() + static_cast<long>(i));

      clipped = true;
      break;
    }

    if (!clipped)
      return {};
  }

  if (indices.size() == 3) {
    triangles.push_back({indices[0], indices[1], indices[2]});
  }

  return triangles;
}

}

void GLWidget::initializeGL() {
  initializeOpenGLFunctions();

  glClearColor(0.08f, 0.10f, 0.13f, 1.0f);

  glEnable(GL_DEPTH_TEST);
  glDisable(GL_CULL_FACE);

  const char *vertexShader = R"(
        attribute vec3 position;
        attribute vec3 color;

        uniform mat4 mvp;

        varying vec3 vertexColor;

        void main()
        {
            gl_Position = mvp * vec4(position, 1.0);
            vertexColor = color;
        }
    )";

  const char *fragmentShader = R"(
        varying vec3 vertexColor;

        void main()
        {
            gl_FragColor = vec4(vertexColor, 1.0);
        }
    )";

  if (!shaderProgram.addShaderFromSourceCode(QOpenGLShader::Vertex,
                                             vertexShader)) {
    qWarning() << "Vertex shader error:" << shaderProgram.log();
    return;
  }

  if (!shaderProgram.addShaderFromSourceCode(QOpenGLShader::Fragment,
                                             fragmentShader)) {
    qWarning() << "Fragment shader error:" << shaderProgram.log();
    return;
  }

  if (!shaderProgram.link()) {
    qWarning() << "Shader link error:" << shaderProgram.log();
    return;
  }

  shaderReady = true;

  vertexBuffer.create();
  vertexBuffer.setUsagePattern(QOpenGLBuffer::StaticDraw);

  featureBuffer.create();
  featureBuffer.setUsagePattern(QOpenGLBuffer::StaticDraw);

  rebuildMesh();
  rebuildFeatureMesh();
}

void GLWidget::rebuildMesh() {
  vertices.clear();
  meshReady = false;

  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2) {
    uploadTerrainBuffer();
    update();
    return;
  }

  const std::size_t required = static_cast<std::size_t>(width) * height;

  if (dem.elevation.size() < required) {
    qWarning() << "Invalid DEM elevation array";
    uploadTerrainBuffer();
    update();
    return;
  }

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const TerrainSize terrainSize = calculateTerrainSize();

  if (terrainSize.width <= 0.0f || terrainSize.height <= 0.0f) {
    uploadTerrainBuffer();
    update();
    return;
  }

  constexpr int maxDimension = 10000;

  const int stepX =
      std::max(1, (width - 1 + maxDimension - 2) / (maxDimension - 1));

  const int stepY =
      std::max(1, (height - 1 + maxDimension - 2) / (maxDimension - 1));

  auto vertexAt = [&](int x, int y, Vertex &result) -> bool {
    const float elevation =
        dem.elevation[static_cast<std::size_t>(y) * width + x];

    if (!std::isfinite(elevation))
      return false;

    const float nx =
        (static_cast<float>(x) / (width - 1) - 0.5f) * terrainSize.width;

    const double *gt = dem.geoTransform;

    const float rowDirection = gt[5] < 0.0 ? -1.0f : 1.0f;

    const float nz = -rowDirection *
                     (static_cast<float>(y) / (height - 1) - 0.5f) *
                     terrainSize.height;

    const float normalized =
        elevationRange > 0.0f
            ? std::clamp((elevation - dem.minElevation) / elevationRange, 0.0f,
                         1.0f)
            : 0.5f;

    const float vertical = elevationRange > 0.0f
                               ? (elevation - dem.minElevation) *
                                     (verticalScaleCoefficient / elevationRange)
                               : 0.0f;

    float r, g, b;

    if (normalized < 0.5f) {
      const float t = normalized * 2.0f;

      r = 0.12f + 0.30f * t;
      g = 0.35f + 0.30f * t;
      b = 0.12f + 0.05f * t;
    } else {
      const float t = (normalized - 0.5f) * 2.0f;

      r = 0.42f + 0.50f * t;
      g = 0.65f + 0.30f * t;
      b = 0.17f + 0.75f * t;
    }

    result = {nx, vertical, nz, r, g, b};

    return true;
  };

  for (int y = 0; y < height - 1; y += stepY) {
    const int y2 = std::min(y + stepY, height - 1);

    for (int x = 0; x < width - 1; x += stepX) {
      const int x2 = std::min(x + stepX, width - 1);

      Vertex a, b, c, d;

      if (!vertexAt(x, y, a) || !vertexAt(x2, y, b) || !vertexAt(x, y2, c) ||
          !vertexAt(x2, y2, d)) {
        continue;
      }

      vertices.insert(vertices.end(), {a, c, b, b, c, d});
    }
  }

  meshReady = !vertices.empty();

  uploadTerrainBuffer();

  qDebug() << "Terrain vertices:" << vertices.size();
}

void GLWidget::uploadTerrainBuffer() {
  if (!context() || !context()->isValid())
    return;

  makeCurrent();

  if (!vertexBuffer.isCreated())
    vertexBuffer.create();

  if (vertexBuffer.isCreated()) {
    vertexBuffer.bind();

    if (vertices.empty())
      vertexBuffer.allocate(0);
    else
      vertexBuffer.allocate(vertices.data(),
                            static_cast<int>(vertices.size() * sizeof(Vertex)));

    vertexBuffer.release();
  }

  doneCurrent();
}

// -----------------------------------------------------------------------------
// Geographic coordinates -> DEM pixels -> scene coordinates
// -----------------------------------------------------------------------------

bool GLWidget::sampleElevation(double pixelX, double pixelY,
                               float &elevation) const {
  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2)
    return false;

  if (pixelX < 0.0 || pixelY < 0.0 || pixelX > width - 1 ||
      pixelY > height - 1) {
    return false;
  }

  const int x0 = static_cast<int>(std::floor(pixelX));
  const int y0 = static_cast<int>(std::floor(pixelY));

  const int x1 = std::min(x0 + 1, width - 1);
  const int y1 = std::min(y0 + 1, height - 1);

  const float fx = static_cast<float>(pixelX - x0);
  const float fy = static_cast<float>(pixelY - y0);

  const auto at = [&](int x, int y) -> float {
    return dem.elevation[static_cast<std::size_t>(y) * width + x];
  };

  const float z00 = at(x0, y0);
  const float z10 = at(x1, y0);
  const float z01 = at(x0, y1);
  const float z11 = at(x1, y1);

  if (!std::isfinite(z00) || !std::isfinite(z10) || !std::isfinite(z01) ||
      !std::isfinite(z11)) {
    return false;
  }

  const float z0 = z00 * (1.0f - fx) + z10 * fx;
  const float z1 = z01 * (1.0f - fx) + z11 * fx;

  elevation = z0 * (1.0f - fy) + z1 * fy;

  return true;
}

bool GLWidget::geoToScene(const GeoPoint &point, ScenePoint &result) const {
  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2)
    return false;

  const double *gt = dem.geoTransform;

  const double determinant = gt[1] * gt[5] - gt[2] * gt[4];

  if (std::abs(determinant) < 1e-15)
    return false;

  const double dx = point.lon - gt[0];
  const double dy = point.lat - gt[3];

  const double pixelX = (gt[5] * dx - gt[2] * dy) / determinant - 0.5;

  const double pixelY = (-gt[4] * dx + gt[1] * dy) / determinant - 0.5;

  float elevation = 0.0f;

  if (!sampleElevation(pixelX, pixelY, elevation))
    return false;

  const TerrainSize terrainSize = calculateTerrainSize();

  const float sceneX =
      (static_cast<float>(pixelX) / (width - 1) - 0.5f) * terrainSize.width;

  const float rowDirection = gt[5] < 0.0 ? -1.0f : 1.0f;

  const float sceneZ = -rowDirection *
                       (static_cast<float>(pixelY) / (height - 1) - 0.5f) *
                       terrainSize.height;

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const float verticalScale = elevationRange > 0.0f
                                  ? verticalScaleCoefficient / elevationRange
                                  : 0.001f;

  result.position = QVector2D(sceneX, sceneZ);

  result.groundY = (elevation - dem.minElevation) * verticalScale;

  return true;
}

// -----------------------------------------------------------------------------
// Common vertex helper
// -----------------------------------------------------------------------------

void GLWidget::addVertex(std::vector<Vertex> &target, const QVector2D &position,
                         float y, float r, float g, float b) {
  target.push_back({position.x(), y, position.y(), r, g, b});
}

// -----------------------------------------------------------------------------
// Building geometry: walls + roof
// -----------------------------------------------------------------------------

void GLWidget::appendBuilding(const OSMFeature &feature) {
  if (feature.geometry.size() < 3)
    return;

  // Преобразует географическое кольцо в координаты сцены.
  auto convertRing = [&](const QVector<GeoPoint> &geoRing,
                         std::vector<ScenePoint> &sceneRing) -> bool {
    sceneRing.clear();

    for (const GeoPoint &geo : geoRing) {
      ScenePoint point;

      if (!geoToScene(geo, point))
        return false;

      sceneRing.push_back(point);
    }

    // Убираем повтор последней точки замкнутого кольца.
    if (sceneRing.size() >= 2 &&
        sameGeoPoint(geoRing.first(), geoRing.last())) {
      sceneRing.pop_back();
    }

    return sceneRing.size() >= 3;
  };

  std::vector<ScenePoint> outerScene;

  if (!convertRing(feature.geometry, outerScene))
    return;

  // Все кольца: сначала outer, затем inner.
  std::vector<std::vector<ScenePoint>> sceneRings;
  sceneRings.push_back(outerScene);

  for (const auto &innerGeo : feature.innerRings) {
    std::vector<ScenePoint> innerScene;

    if (!convertRing(innerGeo, innerScene))
      continue;

    sceneRings.push_back(std::move(innerScene));
  }

  // Формат для Mapbox Earcut.
  std::vector<std::vector<std::array<double, 2>>> polygon;

  // Плоский массив координат для последующего получения вершин крыши.
  std::vector<QVector2D> roofPoints;

  for (const auto &ring : sceneRings) {
    std::vector<std::array<double, 2>> earcutRing;

    for (const ScenePoint &point : ring) {
      earcutRing.push_back({static_cast<double>(point.position.x()),
                            static_cast<double>(point.position.y())});

      roofPoints.push_back(point.position);
    }

    polygon.push_back(std::move(earcutRing));
  }

  if (roofPoints.size() < 3)
    return;

  const std::vector<unsigned int> indices =
      mapbox::earcut<unsigned int>(polygon);

  if (indices.empty())
    return;

  // Высота основания по внешнему контуру.
  float groundSum = 0.0f;

  for (const ScenePoint &point : outerScene)
    groundSum += point.groundY;

  const float baseY = groundSum / static_cast<float>(outerScene.size());

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const float verticalScale = elevationRange > 0.0f
                                  ? verticalScaleCoefficient / elevationRange
                                  : 0.001f;

  const bool isRoof = feature.tags.value("building").toString() == "roof";

  const float buildingHeight =
      feature.heightMeters > 0.0f ? feature.heightMeters : 6.0f;

  const float baseBuildingY = baseY + feature.minHeightMeters * verticalScale;

  // Для building=roof верхняя поверхность находится
  // непосредственно на уровне building:min_level.
  const float roofY =
      isRoof ? baseBuildingY : baseBuildingY + buildingHeight * verticalScale;

  const Colors &color = feature.colors;

  // Стены внешнего и внутренних контуров.
  if (!isRoof)
    for (const auto &ring : sceneRings) {
      const int count = static_cast<int>(ring.size());

      for (int i = 0; i < count; ++i) {
        const QVector2D &a = ring[i].position;
        const QVector2D &b = ring[(i + 1) % count].position;

        addVertex(featureVertices, a, baseBuildingY, color.wallR, color.wallG,
                  color.wallB);

        addVertex(featureVertices, b, baseBuildingY, color.wallR, color.wallG,
                  color.wallB);

        addVertex(featureVertices, b, roofY, color.wallR, color.wallG,
                  color.wallB);

        addVertex(featureVertices, a, baseBuildingY, color.wallR, color.wallG,
                  color.wallB);

        addVertex(featureVertices, b, roofY, color.wallR, color.wallG,
                  color.wallB);

        addVertex(featureVertices, a, roofY, color.wallR, color.wallG,
                  color.wallB);
      }
    }

  // Крыша с отверстиями.
  for (unsigned int index : indices) {
    if (index >= roofPoints.size())
      continue;

    addVertex(featureVertices, roofPoints[index], roofY, color.roofR,
              color.roofG, color.roofB);
  }
}

// -----------------------------------------------------------------------------
// Flat polygon geometry: water, vegetation, landuse, amenity, road areas
// -----------------------------------------------------------------------------

void GLWidget::appendPolygon(const OSMFeature &feature, float heightOffset) {
  if (feature.geometry.size() < 3)
    return;

  std::vector<QVector2D> polygon;
  polygon.reserve(feature.geometry.size());

  float groundSum = 0.0f;
  int validCount = 0;

  for (const GeoPoint &geo : feature.geometry) {
    ScenePoint point;

    if (!geoToScene(geo, point))
      return;

    polygon.push_back(point.position);
    groundSum += point.groundY;
    ++validCount;
  }

  if (polygon.size() >= 2 &&
      sameGeoPoint(feature.geometry.first(), feature.geometry.last())) {
    polygon.pop_back();
  }

  if (polygon.size() < 3 || validCount == 0)
    return;

  const auto triangles = triangulatePolygon(polygon);

  if (triangles.empty())
    return;

  const float y = groundSum / static_cast<float>(validCount) + heightOffset;

  const Colors &color = feature.colors;

  for (const auto &triangle : triangles) {
    for (int index : triangle) {
      addVertex(featureVertices, polygon[index], y, color.roofR, color.roofG,
                color.roofB);
    }
  }
}

// -----------------------------------------------------------------------------
// Line geometry: roads, rivers, railway
// Each line segment becomes a flat-width quad.
// -----------------------------------------------------------------------------

void GLWidget::appendLine(const OSMFeature &feature, float widthMeters,
                          float heightOffset) {
  if (feature.geometry.size() < 2)
    return;

  std::vector<ScenePoint> points;
  points.reserve(feature.geometry.size());

  for (const GeoPoint &geo : feature.geometry) {
    ScenePoint point;

    if (!geoToScene(geo, point)) {
      // Не соединяем участки через точки вне DEM.
      points.clear();
      return;
    }

    points.push_back(point);
  }

  const TerrainSize terrainSize = calculateTerrainSize();

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double physicalWidth =
      std::hypot((dem.width - 1) * gt[1] * metersPerDegreeLon,
                 (dem.width - 1) * gt[4] * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot((dem.height - 1) * gt[2] * metersPerDegreeLon,
                 (dem.height - 1) * gt[5] * metersPerDegreeLat);

  const double maxPhysicalDimension = std::max(physicalWidth, physicalHeight);

  if (maxPhysicalDimension <= 0.0)
    return;

  const float metersToScene = static_cast<float>(2.0 / maxPhysicalDimension);

  const float halfWidth = std::max(0.0001f, widthMeters * metersToScene * 0.5f);

  const Colors &color = feature.colors;

  for (std::size_t i = 0; i + 1 < points.size(); ++i) {
    const ScenePoint &a = points[i];
    const ScenePoint &b = points[i + 1];

    const QVector2D direction = b.position - a.position;

    const float length = direction.length();

    if (length < 1e-7f)
      continue;

    const QVector2D normal(-direction.y() / length, direction.x() / length);

    const QVector2D offset = normal * halfWidth;

    const QVector2D aLeft = a.position + offset;
    const QVector2D aRight = a.position - offset;

    const QVector2D bLeft = b.position + offset;
    const QVector2D bRight = b.position - offset;

    const float yA = a.groundY + heightOffset;
    const float yB = b.groundY + heightOffset;

    // Первый треугольник.
    addVertex(featureVertices, aLeft, yA, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, aRight, yA, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, bLeft, yB, color.roofR, color.roofG,
              color.roofB);

    // Второй треугольник.
    addVertex(featureVertices, aRight, yA, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, bRight, yB, color.roofR, color.roofG,
              color.roofB);

    addVertex(featureVertices, bLeft, yB, color.roofR, color.roofG,
              color.roofB);
  }
}

// -----------------------------------------------------------------------------
// Point geometry: amenity and other OSM nodes
// -----------------------------------------------------------------------------

void GLWidget::appendPoint(const OSMFeature &feature, float sizeMeters,
                           float heightOffset) {
  if (feature.geometry.isEmpty())
    return;

  ScenePoint point;

  if (!geoToScene(feature.geometry.first(), point))
    return;

  const TerrainSize terrainSize = calculateTerrainSize();

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double physicalWidth =
      std::hypot((dem.width - 1) * gt[1] * metersPerDegreeLon,
                 (dem.width - 1) * gt[4] * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot((dem.height - 1) * gt[2] * metersPerDegreeLon,
                 (dem.height - 1) * gt[5] * metersPerDegreeLat);

  const double maxDimension = std::max(physicalWidth, physicalHeight);

  if (maxDimension <= 0.0)
    return;

  const float halfSize =
      std::max(0.0002f, static_cast<float>(sizeMeters / maxDimension));

  const QVector2D p = point.position;

  const QVector2D a(p.x() - halfSize, p.y() - halfSize);
  const QVector2D b(p.x() + halfSize, p.y() - halfSize);
  const QVector2D c(p.x() + halfSize, p.y() + halfSize);
  const QVector2D d(p.x() - halfSize, p.y() + halfSize);

  const float y = point.groundY + heightOffset;

  const Colors &color = feature.colors;

  addVertex(featureVertices, a, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, b, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, c, y, color.roofR, color.roofG, color.roofB);

  addVertex(featureVertices, a, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, c, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, d, y, color.roofR, color.roofG, color.roofB);
}

// -----------------------------------------------------------------------------
// OSM feature dispatcher
// -----------------------------------------------------------------------------

void GLWidget::rebuildFeatureMesh() {
  featureVertices.clear();

  if (dem.width < 2 || dem.height < 2) {
    uploadFeatureBuffer();
    update();
    return;
  }

  int buildings = 0;
  int roads = 0;
  int water = 0;
  int vegetation = 0;
  int railways = 0;
  int other = 0;

  for (const OSMFeature &feature : features) {
    if (feature.geometry.isEmpty())
      continue;

    switch (feature.type) {
    case OSMFeatureType::Building:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendBuilding(feature);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 5.0f);

      ++buildings;
      break;

    case OSMFeatureType::Road:
      if (feature.geometryType == OSMGeometryType::Polygon) {
        appendPolygon(feature, 0.0015f);
      } else if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 4.0f;

        appendLine(feature, width, 0.0015f);
      }

      ++roads;
      break;

    case OSMFeatureType::Water:
      if (feature.geometryType == OSMGeometryType::Polygon) {
        appendPolygon(feature, 0.0025f);
      } else if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 5.0f;

        appendLine(feature, width, 0.0025f);
      }

      ++water;
      break;

    case OSMFeatureType::Vegetation:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.001f);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 4.0f, 0.001f);

      ++vegetation;
      break;

    case OSMFeatureType::Landuse:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.0008f);

      ++other;
      break;

    case OSMFeatureType::Railway:
      if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 4.5f;

        appendLine(feature, width, 0.002f);
      }

      ++railways;
      break;

    case OSMFeatureType::Amenity:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.0015f);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 5.0f, 0.002f);

      ++other;
      break;

    case OSMFeatureType::Unknown:
    default:
      break;
    }
  }

  uploadFeatureBuffer();

  qDebug() << "OSM mesh rebuilt:"
           << "buildings =" << buildings << "roads =" << roads
           << "water =" << water << "vegetation =" << vegetation
           << "railways =" << railways << "other =" << other
           << "vertices =" << featureVertices.size();
}

void GLWidget::uploadFeatureBuffer() {
  if (!context() || !context()->isValid())
    return;

  makeCurrent();

  if (!featureBuffer.isCreated())
    featureBuffer.create();

  if (featureBuffer.isCreated()) {
    featureBuffer.bind();

    if (featureVertices.empty())
      featureBuffer.allocate(0);
    else
      featureBuffer.allocate(
          featureVertices.data(),
          static_cast<int>(featureVertices.size() * sizeof(Vertex)));

    featureBuffer.release();
  }

  doneCurrent();
}

// -----------------------------------------------------------------------------
// Rendering
// -----------------------------------------------------------------------------

void GLWidget::resizeGL(int width, int height) {
  glViewport(0, 0, width, height);
}

void GLWidget::paintGL() {
  glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

  if (!shaderReady)
    return;

  QMatrix4x4 projection;

  projection.perspective(45.0f,
                         static_cast<float>(std::max(1, width())) /
                             static_cast<float>(std::max(1, height())),
                         0.01f, 100.0f);

  QMatrix4x4 view;

  view.translate(panX, panY, -zoom);

  view.rotate(rotationX, 1.0f, 0.0f, 0.0f);
  view.rotate(rotationZ, 0.0f, 1.0f, 0.0f);

  const QMatrix4x4 mvp = projection * view;

  auto drawBuffer = [&](QOpenGLBuffer &buffer, std::size_t vertexCount) {
    if (vertexCount == 0 || !buffer.isCreated())
      return;

    shaderProgram.bind();
    shaderProgram.setUniformValue("mvp", mvp);

    buffer.bind();

    const int positionLocation = shaderProgram.attributeLocation("position");

    const int colorLocation = shaderProgram.attributeLocation("color");

    shaderProgram.enableAttributeArray(positionLocation);
    shaderProgram.enableAttributeArray(colorLocation);

    shaderProgram.setAttributeBuffer(positionLocation, GL_FLOAT,
                                     offsetof(Vertex, x), 3, sizeof(Vertex));

    shaderProgram.setAttributeBuffer(colorLocation, GL_FLOAT,
                                     offsetof(Vertex, r), 3, sizeof(Vertex));

    glDrawArrays(GL_TRIANGLES, 0, static_cast<GLsizei>(vertexCount));

    shaderProgram.disableAttributeArray(positionLocation);
    shaderProgram.disableAttributeArray(colorLocation);

    buffer.release();
    shaderProgram.release();
  };

  // Сначала рельеф.
  if (meshReady)
    drawBuffer(vertexBuffer, vertices.size());

  // Затем здания и остальные OSM-объекты.
  drawBuffer(featureBuffer, featureVertices.size());
}

// -----------------------------------------------------------------------------
// Camera controls
// -----------------------------------------------------------------------------

void GLWidget::mousePressEvent(QMouseEvent *event) {
  lastMousePosition = event->position().toPoint();
}

void GLWidget::mouseMoveEvent(QMouseEvent *event) {
  const QPoint current = event->position().toPoint();

  const QPoint delta = current - lastMousePosition;

  if (event->buttons() & Qt::LeftButton) {
    rotationZ += delta.x() * 0.5f;
    rotationX += delta.y() * 0.5f;

    update();
  }

  if (event->buttons() & Qt::RightButton) {
    const float worldPerPixel =
        (2.0f * zoom * 0.41421356f) / static_cast<float>(std::max(1, height()));

    panX += delta.x() * worldPerPixel;
    panY -= delta.y() * worldPerPixel;

    update();
  }

  lastMousePosition = current;
}

void GLWidget::wheelEvent(QWheelEvent *event) {
  zoom -= event->angleDelta().y() / 120.0f * 0.25f;

  zoom = std::clamp(zoom, 0.01f, 15.0f);

  update();
}

// -----------------------------------------------------------------------------
// DEM physical dimensions
// -----------------------------------------------------------------------------

GLWidget::TerrainSize GLWidget::calculateTerrainSize() const {
  TerrainSize result;

  if (dem.width < 2 || dem.height < 2)
    return result;

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double dxLon = (dem.width - 1) * gt[1];
  const double dxLat = (dem.width - 1) * gt[4];

  const double dyLon = (dem.height - 1) * gt[2];
  const double dyLat = (dem.height - 1) * gt[5];

  const double physicalWidth =
      std::hypot(dxLon * metersPerDegreeLon, dxLat * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot(dyLon * metersPerDegreeLon, dyLat * metersPerDegreeLat);

  const double maxDimension = std::max(physicalWidth, physicalHeight);

  if (maxDimension <= 0.0)
    return result;

  result.width = static_cast<float>(2.0 * physicalWidth / maxDimension);

  result.height = static_cast<float>(2.0 * physicalHeight / maxDimension);

  return result;
}

void MainWindow::onLoadOsmButtonClicked() {

  qDebug() << "Longitude:" << longitude;
  qDebug() << "Radius:" << radius;

  loader->setArea(latitude, longitude, radius);

  // Пока загружаем только здания.
  // Запрос можно расширить для других типов объектов.
  loader->fetchOSMData("buildings", 60);
}

// Загрузка локального JSON
void MainWindow::onLoadOsmJsonButtonClicked() {
  const QString fileName =
      QFileDialog::getOpenFileName(this, "Выбрать файл OSM JSON", QString(),
                                   "JSON files (*.json);;Все файлы (*)");

  if (fileName.isEmpty())
    return;

  QFile file(fileName);

  if (!file.open(QIODevice::ReadOnly)) {
    QMessageBox::warning(this, "Ошибка", "Не удалось открыть файл OSM JSON.");
    return;
  }

  const QByteArray jsonData = file.readAll();
  file.close();

  QJsonParseError parseError;

  const QJsonDocument document = QJsonDocument::fromJson(jsonData, &parseError);

  if (parseError.error != QJsonParseError::NoError) {

    QMessageBox::warning(this, "Ошибка JSON", parseError.errorString());

    return;
  }

  QJsonArray elements;

  // Формат с корневым массивом
  if (document.isArray()) {

    elements = document.array();
  }
  // Стандартный формат Overpass
  else if (document.isObject()) {

    const QJsonObject root = document.object();

    if (!root.value("elements").isArray()) {

      QMessageBox::warning(this, "Ошибка",
                           "В JSON отсутствует массив elements.");

      return;
    }

    elements = root.value("elements").toArray();
  } else {

    QMessageBox::warning(this, "Ошибка", "Некорректная структура OSM JSON.");

    return;
  }

  if (elements.isEmpty()) {

    QMessageBox::information(this, "Информация",
                             "В файле отсутствуют OSM-объекты.");

    return;
  }

  loadOSMFeatures(elements);
}

// Общая загрузка объектов OSM
void MainWindow::loadOSMFeatures(const QJsonArray &elements) {
  const QVector<OSMFeature> features = OSMDataParser::parse(elements);

  if (features.isEmpty()) {

    QMessageBox::information(this, "Информация",
                             "Не удалось найти поддерживаемые OSM-объекты.");

    return;
  }

  // Статистика
  int buildings = 0;
  int roads = 0;
  int water = 0;
  int vegetation = 0;
  int other = 0;

  for (const OSMFeature &feature : features) {

    switch (feature.type) {

    case OSMFeatureType::Building:
      ++buildings;
      break;

    case OSMFeatureType::Road:
      ++roads;
      break;

    case OSMFeatureType::Water:
      ++water;
      break;

    case OSMFeatureType::Vegetation:
      ++vegetation;
      break;

    default:
      ++other;
      break;
    }
  }

  qDebug() << "Загружено OSM объектов:" << features.size();

  qDebug() << "Здания:" << buildings;
  qDebug() << "Дороги:" << roads;
  qDebug() << "Вода:" << water;
  qDebug() << "Растительность:" << vegetation;
  qDebug() << "Другие:" << other;

  // Передаём все объекты в OpenGL
  glWidget->setFeatures(features);
}
#include "osmclassifier.hpp"

OSMFeatureType OSMClassifier::classify(const QJsonObject &tags) {
  // Здания
  if (tags.contains("building"))
    return OSMFeatureType::Building;

  // Дороги
  if (tags.contains("highway"))
    return OSMFeatureType::Road;

  // Железные дороги
  if (tags.contains("railway"))
    return OSMFeatureType::Railway;

  // Вода
  if (tags.contains("waterway") ||
      tags.value("natural").toString() == "water" || tags.contains("water")) {
    return OSMFeatureType::Water;
  }

  // Растительность
  const QString natural = tags.value("natural").toString();
  const QString landuse = tags.value("landuse").toString();
  const QString leisure = tags.value("leisure").toString();

  if (natural == "wood" || natural == "tree" || natural == "scrub" ||
      landuse == "forest" || leisure == "park") {
    return OSMFeatureType::Vegetation;
  }

  // Остальные территории
  if (tags.contains("landuse"))
    return OSMFeatureType::Landuse;

  // Общественные объекты
  if (tags.contains("amenity") || tags.contains("shop") ||
      tags.contains("office")) {
    return OSMFeatureType::Amenity;
  }

  return OSMFeatureType::Unknown;
}

QByteArray OSMDataLoader::fetchOSMData(const QString &queryType,
                                       const int &timeout, const QRect &area) {
  Q_UNUSED(area);

  // Формируем Overpass QL
  QString finalQuery = buildOverpassQuery(timeout, queryType);

  if (finalQuery.isEmpty()) {
    qDebug() << "Неизвестный тип запроса:" << queryType;
    return "";
  }

  // Границы области:
  // south, west, north, east
  const double minLat = defaultArea.lat - defaultArea.width;
  const double maxLat = defaultArea.lat + defaultArea.width;
  const double minLon = defaultArea.lon - defaultArea.height;
  const double maxLon = defaultArea.lon + defaultArea.height;

  const QString bbox = QString("%1,%2,%3,%4")
                           .arg(minLat, 0, 'f', 6)
                           .arg(minLon, 0, 'f', 6)
                           .arg(maxLat, 0, 'f', 6)
                           .arg(maxLon, 0, 'f', 6);

  // Подставляем реальные координаты вместо {{bbox}}
  finalQuery.replace("{{bbox}}", bbox);

  qDebug() << "Тип запроса:" << queryType;
  qDebug() << "BBox:" << bbox;
  qDebug() << "Весь запрос:" << finalQuery;

  QNetworkRequest request(url);

  request.setHeader(QNetworkRequest::ContentTypeHeader,
                    "application/x-www-form-urlencoded");
  request.setHeader(
      QNetworkRequest::UserAgentHeader,
      "BMSTU-NIR-OSM-Loader/1.0 (+https://github.com/Andrey8066)");
  request.setRawHeader("Accept", "application/json");

  QByteArray postData = "data=";
  postData += QUrl::toPercentEncoding(finalQuery);

  reply = manager->post(request, postData);

  connect(reply, &QNetworkReply::finished, this, [this, area, queryType]() {
    const QVariant status =
        reply->attribute(QNetworkRequest::HttpStatusCodeAttribute);

    qDebug() << "HTTP status:" << status;

    if (reply->error() != QNetworkReply::NoError) {

      qDebug() << "Ошибка загрузки OSM данных:" << reply->errorString();

      const QByteArray serverResponse = reply->readAll();

      qDebug() << "Ответ сервера:" << serverResponse;

      emit errorOccurred("Не удалось загрузить данные из OSM");

      reply->deleteLater();
      return;
    }

    requestResult = reply->readAll();

    QJsonParseError parseError;

    const QJsonDocument doc =
        QJsonDocument::fromJson(requestResult, &parseError);

    if (parseError.error != QJsonParseError::NoError) {

      qDebug() << "Ошибка парсинга JSON:" << parseError.errorString();

      qDebug() << "Ответ сервера:" << requestResult;

      reply->deleteLater();
      return;
    }

    qDebug() << "OSM JSON успешно получен.";

    qDebug() << "Получено байт:" << requestResult.size();

    const QJsonObject root = doc.object();
    const QJsonArray elements = root["elements"].toArray();

    emit dataLoaded(elements, queryType);
    int ways = 0;
    int relations = 0;
    int nodes = 0;

    int buildings = 0;
    int buildingParts = 0;

    for (const QJsonValue &value : elements) {
      QJsonObject object = value.toObject();
      QJsonObject tags = object["tags"].toObject();

      QString type = object["type"].toString();

      if (type == "way")
        ways++;

      if (type == "relation")
        relations++;

      if (type == "node")
        nodes++;

      if (tags.contains("building"))
        buildings++;

      if (tags.contains("building:part"))
        buildingParts++;
    }

    qDebug() << "========== OSM STATISTICS ==========";
    qDebug() << "Total elements:" << elements.size();
    qDebug() << "Ways:" << ways;
    qDebug() << "Relations:" << relations;
    qDebug() << "Nodes:" << nodes;
    qDebug() << "Building tags:" << buildings;
    qDebug() << "Building parts:" << buildingParts;

    int wayBuildings = 0;
    int relationBuildings = 0;

    int wayParts = 0;
    int relationParts = 0;

    for (const QJsonValue &value : elements) {
      QJsonObject object = value.toObject();
      QJsonObject tags = object["tags"].toObject();

      QString type = object["type"].toString();

      bool hasBuilding = tags.contains("building");
      bool hasPart = tags.contains("building:part");

      if (type == "way" && hasBuilding)
        wayBuildings++;

      if (type == "relation" && hasBuilding)
        relationBuildings++;

      if (type == "way" && hasPart)
        wayParts++;

      if (type == "relation" && hasPart)
        relationParts++;
    }

    qDebug() << "Way buildings:" << wayBuildings;
    qDebug() << "Relation buildings:" << relationBuildings;

    qDebug() << "Way building parts:" << wayParts;
    qDebug() << "Relation building parts:" << relationParts;
  });

  return requestResult;
}

// Формирование Overpass QL запроса - правильная syntax

QString OSMDataLoader::buildOverpassQuery(const int &timeout,
                                          const QString &queryType) {
  if (queryType == "buildings") {
    return QString(R"(
            [out:json][timeout:%1];

            (
                nwr["building"]({{bbox}});
                nwr["building:part"]({{bbox}});
                relation["type"="building"]({{bbox}});
            );

            out geom;
        )")
        .arg(timeout);
  }

  if (queryType == "all") {
    return QString(R"([out:json][timeout:%1];
                   (
                   nwr["building"]({{bbox}});
                   nwr["highway"]({{bbox}});
                   );
                   out geom;
                   )")
        .arg(timeout);
  }

  qWarning() << "Неизвестный тип запроса:" << queryType;
  return QString();
}

// Сохранение JSON в файл
QString OSMDataLoader::saveToJSONFile(const QJsonArray &data,
                                      const QString &filename) {
  QFile file(filename);

  if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
    qDebug() << "Ошибка открытия файла для записи:" << filename;
    return "";
  }

  file.write(QJsonDocument(data).toJson());
  file.close();

  return filename;
}

void OSMDataLoader::waitData() {
  if (!reply) {
    qDebug() << "Нет активного запроса";
    return;
  }

  if (!reply->isFinished()) {
    QEventLoop loop;

    connect(reply, &QNetworkReply::finished, &loop, &QEventLoop::quit);

    loop.exec();
  }

  qDebug() << "waitData(): запрос завершён";
}

double OSMDataLoader::radToLat(double radius) {
  return radius / (earthRadius * M_PI) * 180;
}
double OSMDataLoader::radToLon(double radius) {
  const double latRadians = defaultArea.lat * M_PI / 180.0;

  return radius / (earthRadius * std::cos(latRadians)) * 180.0 / M_PI;
}

double parseLength(const QJsonValue &value, bool *ok) {
  *ok = false;

  double result = 0.0;

  if (value.isDouble()) {
    result = value.toDouble();
    *ok = true;
    return result;
  }

  QString text = value.toString().trimmed().toLower();

  bool feet = text.endsWith("ft");

  if (feet)
    text.chop(2);
  else if (text.endsWith("m"))
    text.chop(1);

  result = text.trimmed().toDouble(ok);

  if (feet && *ok)
    result *= 0.3048;

  return result;
}

float parseBuildingHeight(const QJsonObject &tags) {
  bool ok = false;

  double height = parseLength(tags.value("height"), &ok);

  if (ok && height > 0.0)
    return static_cast<float>(std::clamp(height, 2.0, 300.0));

  const double levels = tags.value("building:levels").toString().toDouble(&ok);

  if (ok && levels > 0.0) {
    return static_cast<float>(std::clamp(levels * 3.0, 2.0, 300.0));
  }

  return 9.0f;
}

float parseRoadWidth(const QJsonObject &tags) {
  bool ok = false;

  double width = parseLength(tags.value("width"), &ok);

  if (ok && width > 0.0)
    return static_cast<float>(width);

  const double lanes = tags.value("lanes").toString().toDouble(&ok);

  if (ok && lanes > 0.0)
    return static_cast<float>(lanes * 3.5);

  const QString highway = tags.value("highway").toString();

  if (highway == "motorway" || highway == "trunk")
    return 12.0f;

  if (highway == "primary" || highway == "secondary")
    return 9.0f;

  if (highway == "residential")
    return 6.0f;

  if (highway == "footway" || highway == "path" || highway == "cycleway")
    return 2.0f;

  return 5.0f;
}

Colors getBuildingColors(const QJsonObject &tags) {
  const QString type = tags.value("building").toString();

  if (type == "industrial") {
    return Colors(0.55f, 0.55f, 0.55f, 0.35f, 0.35f, 0.35f);
  }

  if (type == "commercial" || type == "retail") {
    return Colors(0.72f, 0.62f, 0.48f, 0.35f, 0.32f, 0.28f);
  }

  if (type == "house") {
    return Colors(0.75f, 0.65f, 0.52f, 0.45f, 0.25f, 0.20f);
  }

  return Colors(0.58f, 0.63f, 0.70f, 0.28f, 0.36f, 0.28f);
}

bool samePoint(const GeoPoint &a, const GeoPoint &b) {
  constexpr double eps = 1e-9;

  return std::abs(a.lat - b.lat) < eps && std::abs(a.lon - b.lon) < eps;
}

bool pointInRing(const GeoPoint &point, const QVector<GeoPoint> &ring) {
  bool inside = false;

  if (ring.size() < 3)
    return false;

  for (int i = 0, j = ring.size() - 1; i < ring.size(); j = i++) {

    const GeoPoint &a = ring[i];
    const GeoPoint &b = ring[j];

    const bool intersects =
        ((a.lat > point.lat) != (b.lat > point.lat)) &&
        (point.lon <
         (b.lon - a.lon) * (point.lat - a.lat) / (b.lat - a.lat) + a.lon);

    if (intersects)
      inside = !inside;
  }

  return inside;
}

// Читает координаты одного way из участника relation.
QVector<GeoPoint> readMemberGeometry(const QJsonObject &member) {
  QVector<GeoPoint> points;

  const QJsonArray geometry = member.value("geometry").toArray();

  for (const QJsonValue &value : geometry) {
    const QJsonObject point = value.toObject();

    if (!point.contains("lat") || !point.contains("lon"))
      continue;

    points.push_back(
        {point.value("lat").toDouble(), point.value("lon").toDouble()});
  }

  return points;
}

QVector<QVector<GeoPoint>> readRings(const QJsonObject &object,
                                     const QString &role) {
  QVector<QVector<GeoPoint>> segments;

  const QJsonArray members = object.value("members").toArray();

  for (const QJsonValue &value : members) {
    const QJsonObject member = value.toObject();

    if (member.value("type").toString() != "way")
      continue;

    if (member.value("role").toString() != role)
      continue;

    QVector<GeoPoint> points = readMemberGeometry(member);

    if (points.size() >= 2)
      segments.push_back(std::move(points));
  }

  QVector<QVector<GeoPoint>> rings;

  while (!segments.isEmpty()) {
    QVector<GeoPoint> ring = segments.takeFirst();

    bool changed = true;

    while (changed && !segments.isEmpty()) {
      changed = false;

      for (int i = 0; i < segments.size(); ++i) {
        QVector<GeoPoint> segment = segments[i];

        if (samePoint(ring.back(), segment.front())) {
          for (int j = 1; j < segment.size(); ++j)
            ring.push_back(segment[j]);

        } else if (samePoint(ring.back(), segment.back())) {
          std::reverse(segment.begin(), segment.end());

          for (int j = 1; j < segment.size(); ++j)
            ring.push_back(segment[j]);

        } else if (samePoint(ring.front(), segment.back())) {
          // Вставляем в начало в обратном порядке.
          for (int j = segment.size() - 2; j >= 0; --j)
            ring.prepend(segment[j]);

        } else if (samePoint(ring.front(), segment.front())) {
          std::reverse(segment.begin(), segment.end());

          for (int j = segment.size() - 2; j >= 0; --j)
            ring.prepend(segment[j]);

        } else {
          continue;
        }

        segments.removeAt(i);
        changed = true;
        break;
      }
    }

    if (ring.size() < 4 || !samePoint(ring.front(), ring.back()))
      continue;

    ring.removeLast();

    if (ring.size() >= 3)
      rings.push_back(std::move(ring));
  }

  return rings;
}

bool readGeometry(const QJsonObject &object, OSMFeature &feature) {
  const QString osmType = object.value("type").toString();

  if (osmType == "node") {

    if (!object.contains("lat") || !object.contains("lon")) {
      return false;
    }

    feature.geometryType = OSMGeometryType::Point;

    feature.geometry.push_back(
        {object.value("lat").toDouble(), object.value("lon").toDouble()});

    return true;
  }

  if (osmType == "way") {

    const QJsonArray geometry = object.value("geometry").toArray();

    for (const QJsonValue &value : geometry) {

      const QJsonObject point = value.toObject();

      if (!point.contains("lat") || !point.contains("lon")) {
        continue;
      }

      feature.geometry.push_back(
          {point.value("lat").toDouble(), point.value("lon").toDouble()});
    }

    if (feature.geometry.isEmpty())
      return false;

    const bool closed =
        feature.geometry.size() >= 4 &&
        std::abs(feature.geometry.front().lat - feature.geometry.back().lat) <
            1e-10 &&
        std::abs(feature.geometry.front().lon - feature.geometry.back().lon) <
            1e-10;

    const bool area = feature.tags.value("area").toString() == "yes" ||
                      (feature.tags.contains("building") && closed) ||
                      (feature.tags.contains("landuse") && closed) ||
                      (feature.tags.contains("leisure") && closed) ||
                      (feature.tags.contains("natural") && closed) ||
                      (feature.tags.contains("water") && closed);

    if (area) {
      feature.geometryType = OSMGeometryType::Polygon;

      if (closed)
        feature.geometry.pop_back();
    } else {
      feature.geometryType = OSMGeometryType::LineString;
    }

    return true;
  }

  // Relation пока не обрабатываем
  return false;
}

QVector<OSMFeature> OSMDataParser::parse(const QJsonArray &elements) {
  QVector<OSMFeature> result;

  result.reserve(elements.size());

  // ID way, которые уже войдут в геометрию relation.
  QSet<qint64> relationWayIds;

  // Первый проход: собираем ID внешних way поддерживаемых relation.
  for (const QJsonValue &value : elements) {
    if (!value.isObject())
      continue;

    const QJsonObject object = value.toObject();

    if (object.value("type").toString() != "relation")
      continue;

    const QJsonObject tags = object.value("tags").toObject();
    const QString relationType = tags.value("type").toString();

    if (relationType != "multipolygon" && relationType != "boundary" &&
        relationType != "building") {
      continue;
    }

    const QJsonArray members = object.value("members").toArray();

    for (const QJsonValue &memberValue : members) {
      const QJsonObject member = memberValue.toObject();

      if (member.value("type").toString() != "way")
        continue;

      const QString role = member.value("role").toString();

      const bool isOuter = role == "outer";
      const bool isBuildingInner =
          OSMClassifier::classify(tags) == OSMFeatureType::Building &&
          role == "inner";

      if (!isOuter && !isBuildingInner)
        continue;
      relationWayIds.insert(
          static_cast<qint64>(member.value("ref").toDouble()));
    }
  }

  // Второй проход: обрабатываем объекты.
  for (const QJsonValue &value : elements) {
    if (!value.isObject())
      continue;

    const QJsonObject object = value.toObject();

    const QString osmType = object.value("type").toString();

    // Не обрабатываем way повторно, если он уже вошёл в relation.
    if (osmType == "way") {
      const qint64 wayId = static_cast<qint64>(object.value("id").toDouble());

      if (relationWayIds.contains(wayId))
        continue;
    }

    OSMFeature feature;

    feature.id = static_cast<qint64>(object.value("id").toDouble());

    feature.tags = object.value("tags").toObject();

    feature.type = OSMClassifier::classify(feature.tags);

    if (feature.type == OSMFeatureType::Unknown)
      continue;

    // Relation обрабатываем отдельно.
    if (osmType == "relation") {
      const QString relationType = feature.tags.value("type").toString();

      if (relationType != "multipolygon" && relationType != "boundary" &&
          relationType != "building") {
        continue;
      }

      // 1. Собираем внешние кольца relation.
      const auto outerRings = readRings(object, "outer");

      // 2. Внутренние кольца нужны только зданиям.
      QVector<QVector<GeoPoint>> innerRings;

      if (feature.type == OSMFeatureType::Building) {
        innerRings = readRings(object, "inner");
      }

      // 3. Создаём отдельный OSMFeature для каждого внешнего кольца.
      for (const auto &outer : outerRings) {
        OSMFeature relationFeature = feature;

        relationFeature.geometryType = OSMGeometryType::Polygon;
        relationFeature.geometry = outer;

        if (relationFeature.type == OSMFeatureType::Building) {
          relationFeature.heightMeters =
              parseBuildingHeight(relationFeature.tags);
          relationFeature.minHeightMeters = 0.0f;

          const QString minLevel =
              relationFeature.tags.value("building:min_level").toString();

          if (!minLevel.isEmpty()) {
            bool ok = false;
            const float levels = minLevel.toFloat(&ok);

            if (ok)
              relationFeature.minHeightMeters = levels * 3.0f;
          }

          relationFeature.colors = getBuildingColors(relationFeature.tags);

          // 4. Прикрепляем к зданию только те inner,
          // которые находятся внутри текущего outer.
          for (const auto &inner : innerRings) {
            if (inner.isEmpty())
              continue;

            if (pointInRing(inner.first(), outer)) {
              relationFeature.innerRings.push_back(inner);
            }
          }
        }

        result.push_back(std::move(relationFeature));
      }

      continue;
    }

    // Обычные node и way.
    if (!readGeometry(object, feature))
      continue;

    if (feature.type == OSMFeatureType::Building) {
      feature.heightMeters = parseBuildingHeight(feature.tags);
      feature.minHeightMeters = 0.0f;

      const QString minLevel =
          feature.tags.value("building:min_level").toString();

      if (!minLevel.isEmpty()) {
        bool ok = false;
        const float levels = minLevel.toFloat(&ok);

        if (ok)
          feature.minHeightMeters = levels * 3.0f;
      }
      feature.colors = getBuildingColors(feature.tags);
    }

    if (feature.type == OSMFeatureType::Road) {
      feature.widthMeters = parseRoadWidth(feature.tags);

      const QString highway = feature.tags.value("highway").toString();

      if (highway == "footway" || highway == "path" || highway == "cycleway")
        feature.colors = Colors(0, 0.5f, 0, 0, 0.5f, 0);
    }

    result.push_back(std::move(feature));
  }

  return result;
}

bool DEMReader::loadDem(const QString &fileName, DEMData &out) {
  GDALAllRegister();

  GDALDataset *dataset = static_cast<GDALDataset *>(
      GDALOpen(fileName.toLocal8Bit().constData(), GA_ReadOnly));

  if (!dataset)
    return false;

  const int width = dataset->GetRasterXSize();
  const int height = dataset->GetRasterYSize();

  if (width < 2 || height < 2 || dataset->GetRasterCount() < 1) {
    GDALClose(dataset);
    return false;
  }

  GDALRasterBand *band = dataset->GetRasterBand(1);

  out.width = width;
  out.height = height;

  qDebug() << "Размер DEM:" << width << "x" << height;

  qDebug() << "Projection:" << dataset->GetProjectionRef();

  qDebug() << "Raster type:" << GDALGetDataTypeName(band->GetRasterDataType());

  int hasNoData = 0;
  double noData = band->GetNoDataValue(&hasNoData);

  qDebug() << "NoData exists:" << hasNoData << "value:" << noData;

  qDebug() << "Scale:" << band->GetScale() << "Offset:" << band->GetOffset();

  if (dataset->GetGeoTransform(out.geoTransform) != CE_None) {
    GDALClose(dataset);

    const char *projection = dataset->GetProjectionRef();

    out.projectionWkt = projection ? QString::fromUtf8(projection) : QString();

    return false;
  }

  out.elevation.resize(static_cast<std::size_t>(width) * height);

  const CPLErr result =
      band->RasterIO(GF_Read, 0, 0, width, height, out.elevation.data(), width,
                     height, GDT_Float32, 0, 0);

  if (result != CE_None) {
    GDALClose(dataset);
    out = DEMData{};
    return false;
  }

  float minZ = std::numeric_limits<float>::max();
  float maxZ = std::numeric_limits<float>::lowest();

  for (float &z : out.elevation) {
    if (!std::isfinite(z) || (hasNoData && z == static_cast<float>(noData))) {
      // Для первого прототипа заполняем пропуски нулём.
      // Для точной модели NoData лучше интерполировать.
      z = 0.0f;
    }

    minZ = std::min(minZ, z);
    maxZ = std::max(maxZ, z);
  }

  out.minElevation = minZ;
  out.maxElevation = maxZ;

  GDALClose(dataset);
  return true;
}

void FeatureDispatcher::dispatch(const OSMFeature &feature) {
  if (!m_widget) {
    qWarning() << "FeatureDispatcher: GLWidget is null";
    return;
  }

  if (feature.geometry.isEmpty())
    return;

  switch (feature.type) {
  case OSMFeatureType::Building:
    dispatchBuilding(feature);
    break;

  case OSMFeatureType::Road:
    dispatchRoad(feature);
    break;

  case OSMFeatureType::Water:
    dispatchWater(feature);
    break;

  case OSMFeatureType::Vegetation:
    dispatchVegetation(feature);
    break;

  case OSMFeatureType::Landuse:
    dispatchLanduse(feature);
    break;

  case OSMFeatureType::Railway:
    dispatchRailway(feature);
    break;

  case OSMFeatureType::Amenity:
    dispatchAmenity(feature);
    break;

  case OSMFeatureType::Unknown:
  default:
    break;
  }
}

void FeatureDispatcher::dispatchAll(const QVector<OSMFeature> &features) {
  for (const OSMFeature &feature : features)
    dispatch(feature);
}

float cross2D(const QVector2D &a, const QVector2D &b, const QVector2D &c) {
  return (b.x() - a.x()) * (c.y() - a.y()) - (b.y() - a.y()) * (c.x() - a.x());
}

bool pointInTriangle(const QVector2D &p, const QVector2D &a, const QVector2D &b,
                     const QVector2D &c) {
  constexpr float eps = 1e-7f;

  return cross2D(a, b, p) >= -eps && cross2D(b, c, p) >= -eps &&
         cross2D(c, a, p) >= -eps;
}

std::vector<std::array<int, 3>>
triangulatePolygon(const std::vector<QVector2D> &points) {
  std::vector<std::array<int, 3>> triangles;

  if (points.size() < 3)
    return triangles;

  const int n = static_cast<int>(points.size());

  double area = 0.0;

  for (int i = 0; i < n; ++i) {
    const auto &a = points[i];
    const auto &b = points[(i + 1) % n];

    area += a.x() * b.y() - b.x() * a.y();
  }

  if (std::abs(area) < 1e-10)
    return triangles;

  std::vector<int> indices(n);
  std::iota(indices.begin(), indices.end(), 0);

  if (area < 0.0)
    std::reverse(indices.begin(), indices.end());

  constexpr float eps = 1e-7f;

  while (indices.size() > 3) {
    bool clipped = false;

    for (std::size_t i = 0; i < indices.size(); ++i) {
      const int prev = indices[(i + indices.size() - 1) % indices.size()];

      const int curr = indices[i];
      const int next = indices[(i + 1) % indices.size()];

      const auto &a = points[prev];
      const auto &b = points[curr];
      const auto &c = points[next];

      if (cross2D(a, b, c) <= eps)
        continue;

      bool containsPoint = false;

      for (int index : indices) {
        if (index == prev || index == curr || index == next)
          continue;

        if (pointInTriangle(points[index], a, b, c)) {
          containsPoint = true;
          break;
        }
      }

      if (containsPoint)
        continue;

      triangles.push_back({prev, curr, next});
      indices.erase(indices.begin() + static_cast<long>(i));

      clipped = true;
      break;
    }

    if (!clipped)
      return {};
  }

  if (indices.size() == 3) {
    triangles.push_back({indices[0], indices[1], indices[2]});
  }

  return triangles;
}

bool sameGeoPoint(const GeoPoint &a, const GeoPoint &b) {
  return std::abs(a.lat - b.lat) < 1e-10 && std::abs(a.lon - b.lon) < 1e-10;
}

void GLWidget::initializeGL() {
  initializeOpenGLFunctions();

  glClearColor(0.08f, 0.10f, 0.13f, 1.0f);

  glEnable(GL_DEPTH_TEST);
  glDisable(GL_CULL_FACE);

  const char *vertexShader = R"(
        attribute vec3 position;
        attribute vec3 color;

        uniform mat4 mvp;

        varying vec3 vertexColor;

        void main()
        {
            gl_Position = mvp * vec4(position, 1.0);
            vertexColor = color;
        }
    )";

  const char *fragmentShader = R"(
        varying vec3 vertexColor;

        void main()
        {
            gl_FragColor = vec4(vertexColor, 1.0);
        }
    )";

  if (!shaderProgram.addShaderFromSourceCode(QOpenGLShader::Vertex,
                                             vertexShader)) {
    qWarning() << "Vertex shader error:" << shaderProgram.log();
    return;
  }

  if (!shaderProgram.addShaderFromSourceCode(QOpenGLShader::Fragment,
                                             fragmentShader)) {
    qWarning() << "Fragment shader error:" << shaderProgram.log();
    return;
  }

  if (!shaderProgram.link()) {
    qWarning() << "Shader link error:" << shaderProgram.log();
    return;
  }

  shaderReady = true;

  vertexBuffer.create();
  vertexBuffer.setUsagePattern(QOpenGLBuffer::StaticDraw);

  featureBuffer.create();
  featureBuffer.setUsagePattern(QOpenGLBuffer::StaticDraw);

  rebuildMesh();
  rebuildFeatureMesh();
}

void GLWidget::rebuildMesh() {
  vertices.clear();
  meshReady = false;

  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2) {
    uploadTerrainBuffer();
    update();
    return;
  }

  const std::size_t required = static_cast<std::size_t>(width) * height;

  if (dem.elevation.size() < required) {
    qWarning() << "Invalid DEM elevation array";
    uploadTerrainBuffer();
    update();
    return;
  }

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const TerrainSize terrainSize = calculateTerrainSize();

  if (terrainSize.width <= 0.0f || terrainSize.height <= 0.0f) {
    uploadTerrainBuffer();
    update();
    return;
  }

  constexpr int maxDimension = 10000;

  const int stepX =
      std::max(1, (width - 1 + maxDimension - 2) / (maxDimension - 1));

  const int stepY =
      std::max(1, (height - 1 + maxDimension - 2) / (maxDimension - 1));

  auto vertexAt = [&](int x, int y, Vertex &result) -> bool {
    const float elevation =
        dem.elevation[static_cast<std::size_t>(y) * width + x];

    if (!std::isfinite(elevation))
      return false;

    const float nx =
        (static_cast<float>(x) / (width - 1) - 0.5f) * terrainSize.width;

    const double *gt = dem.geoTransform;

    const float rowDirection = gt[5] < 0.0 ? -1.0f : 1.0f;

    const float nz = -rowDirection *
                     (static_cast<float>(y) / (height - 1) - 0.5f) *
                     terrainSize.height;

    const float normalized =
        elevationRange > 0.0f
            ? std::clamp((elevation - dem.minElevation) / elevationRange, 0.0f,
                         1.0f)
            : 0.5f;

    const float vertical = elevationRange > 0.0f
                               ? (elevation - dem.minElevation) *
                                     (verticalScaleCoefficient / elevationRange)
                               : 0.0f;

    float r, g, b;

    if (normalized < 0.5f) {
      const float t = normalized * 2.0f;

      r = 0.12f + 0.30f * t;
      g = 0.35f + 0.30f * t;
      b = 0.12f + 0.05f * t;
    } else {
      const float t = (normalized - 0.5f) * 2.0f;

      r = 0.42f + 0.50f * t;
      g = 0.65f + 0.30f * t;
      b = 0.17f + 0.75f * t;
    }

    result = {nx, vertical, nz, r, g, b};

    return true;
  };

  for (int y = 0; y < height - 1; y += stepY) {
    const int y2 = std::min(y + stepY, height - 1);

    for (int x = 0; x < width - 1; x += stepX) {
      const int x2 = std::min(x + stepX, width - 1);

      Vertex a, b, c, d;

      if (!vertexAt(x, y, a) || !vertexAt(x2, y, b) || !vertexAt(x, y2, c) ||
          !vertexAt(x2, y2, d)) {
        continue;
      }

      vertices.insert(vertices.end(), {a, c, b, b, c, d});
    }
  }

  meshReady = !vertices.empty();

  uploadTerrainBuffer();

  qDebug() << "Terrain vertices:" << vertices.size();
}

void GLWidget::uploadTerrainBuffer() {
  if (!context() || !context()->isValid())
    return;

  makeCurrent();

  if (!vertexBuffer.isCreated())
    vertexBuffer.create();

  if (vertexBuffer.isCreated()) {
    vertexBuffer.bind();

    if (vertices.empty())
      vertexBuffer.allocate(0);
    else
      vertexBuffer.allocate(vertices.data(),
                            static_cast<int>(vertices.size() * sizeof(Vertex)));

    vertexBuffer.release();
  }

  doneCurrent();
}

// -----------------------------------------------------------------------------
// Geographic coordinates -> DEM pixels -> scene coordinates
// -----------------------------------------------------------------------------

bool GLWidget::sampleElevation(double pixelX, double pixelY,
                               float &elevation) const {
  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2)
    return false;

  if (pixelX < 0.0 || pixelY < 0.0 || pixelX > width - 1 ||
      pixelY > height - 1) {
    return false;
  }

  const int x0 = static_cast<int>(std::floor(pixelX));
  const int y0 = static_cast<int>(std::floor(pixelY));

  const int x1 = std::min(x0 + 1, width - 1);
  const int y1 = std::min(y0 + 1, height - 1);

  const float fx = static_cast<float>(pixelX - x0);
  const float fy = static_cast<float>(pixelY - y0);

  const auto at = [&](int x, int y) -> float {
    return dem.elevation[static_cast<std::size_t>(y) * width + x];
  };

  const float z00 = at(x0, y0); // a
  const float z10 = at(x1, y0); // b
  const float z01 = at(x0, y1); // c
  const float z11 = at(x1, y1); // d

  if (!std::isfinite(z00) || !std::isfinite(z10) || !std::isfinite(z01) ||
      !std::isfinite(z11)) {
    return false;
  }

  // Интерполяция по той же диагонали b-c,
  // по которой строятся треугольники рельефа.
  if (fx + fy <= 1.0f) {
    elevation = z00 * (1.0f - fx - fy) + z10 * fx + z01 * fy;
  } else {
    elevation = z10 * (1.0f - fy) + z01 * (1.0f - fx) + z11 * (fx + fy - 1.0f);
  }

  return true;
}

bool GLWidget::geoToScene(const GeoPoint &point, ScenePoint &result) const {
  const int width = dem.width;
  const int height = dem.height;

  if (width < 2 || height < 2)
    return false;

  const double *gt = dem.geoTransform;

  const double determinant = gt[1] * gt[5] - gt[2] * gt[4];

  if (std::abs(determinant) < 1e-15)
    return false;

  const double dx = point.lon - gt[0];
  const double dy = point.lat - gt[3];

  const double pixelX = (gt[5] * dx - gt[2] * dy) / determinant - 0.5;

  const double pixelY = (-gt[4] * dx + gt[1] * dy) / determinant - 0.5;

  float elevation = 0.0f;

  if (!sampleElevation(pixelX, pixelY, elevation))
    return false;

  const TerrainSize terrainSize = calculateTerrainSize();

  const float sceneX =
      (static_cast<float>(pixelX) / (width - 1) - 0.5f) * terrainSize.width;

  const float rowDirection = gt[5] < 0.0 ? -1.0f : 1.0f;

  const float sceneZ = -rowDirection *
                       (static_cast<float>(pixelY) / (height - 1) - 0.5f) *
                       terrainSize.height;

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const float verticalScale = elevationRange > 0.0f
                                  ? verticalScaleCoefficient / elevationRange
                                  : 0.001f;

  result.position = QVector2D(sceneX, sceneZ);

  result.groundY = (elevation - dem.minElevation) * verticalScale;

  return true;
}

// -----------------------------------------------------------------------------
// Building geometry: walls + roof
// -----------------------------------------------------------------------------

void GLWidget::appendBuilding(const OSMFeature &feature) {
  if (feature.geometry.size() < 3)
    return;

  // Преобразует географическое кольцо в координаты сцены.
  auto convertRing = [&](const QVector<GeoPoint> &geoRing,
                         std::vector<ScenePoint> &sceneRing) -> bool {
    sceneRing.clear();

    for (const GeoPoint &geo : geoRing) {
      ScenePoint point;

      if (!geoToScene(geo, point))
        return false;

      sceneRing.push_back(point);
    }

    // Убираем повтор последней точки замкнутого кольца.
    if (sceneRing.size() >= 2 &&
        sameGeoPoint(geoRing.first(), geoRing.last())) {
      sceneRing.pop_back();
    }

    return sceneRing.size() >= 3;
  };

  std::vector<ScenePoint> outerScene;

  if (!convertRing(feature.geometry, outerScene))
    return;

  // Все кольца: сначала outer, затем inner.
  std::vector<std::vector<ScenePoint>> sceneRings;
  sceneRings.push_back(outerScene);

  for (const auto &innerGeo : feature.innerRings) {
    std::vector<ScenePoint> innerScene;

    if (!convertRing(innerGeo, innerScene))
      continue;

    sceneRings.push_back(std::move(innerScene));
  }

  // Формат для Mapbox Earcut.
  std::vector<std::vector<std::array<double, 2>>> polygon;

  // Плоский массив координат для последующего получения вершин крыши.
  std::vector<QVector2D> roofPoints;

  for (const auto &ring : sceneRings) {
    std::vector<std::array<double, 2>> earcutRing;

    for (const ScenePoint &point : ring) {
      earcutRing.push_back({static_cast<double>(point.position.x()),
                            static_cast<double>(point.position.y())});

      roofPoints.push_back(point.position);
    }

    polygon.push_back(std::move(earcutRing));
  }

  if (roofPoints.size() < 3)
    return;

  const std::vector<unsigned int> indices =
      mapbox::earcut<unsigned int>(polygon);

  if (indices.empty())
    return;

  // Высота основания по внешнему контуру.
  float groundSum = 0.0f;

  for (const ScenePoint &point : outerScene)
    groundSum += point.groundY;

  const float baseY = groundSum / static_cast<float>(outerScene.size());

  const float elevationRange = dem.maxElevation - dem.minElevation;

  const float verticalScale = elevationRange > 0.0f
                                  ? verticalScaleCoefficient / elevationRange
                                  : 0.001f;

  const bool isRoof = feature.tags.value("building").toString() == "roof";

  const float buildingHeight =
      feature.heightMeters > 0.0f ? feature.heightMeters : 6.0f;

  const float baseBuildingY = baseY + feature.minHeightMeters * verticalScale;

  // Для building=roof верхняя поверхность находится
  // непосредственно на уровне building:min_level.
  const float roofY =
      isRoof ? baseBuildingY : baseBuildingY + buildingHeight * verticalScale;

  const Colors &color = feature.colors;

  // Стены внешнего и внутренних контуров.
  if (!isRoof) {
    for (const auto &ring : sceneRings) {
      const int count = static_cast<int>(ring.size());

      for (int i = 0; i < count; ++i) {
        const ScenePoint &a = ring[i];
        const ScenePoint &b = ring[(i + 1) % count];

        // Индивидуальная высота основания каждой вершины.
        const float bottomA =
            a.groundY + feature.minHeightMeters * verticalScale;

        const float bottomB =
            b.groundY + feature.minHeightMeters * verticalScale;

        const QVector2D &pa = a.position;
        const QVector2D &pb = b.position;

        // Первый треугольник.
        addVertex(featureVertices, pa, bottomA, color.wallR, color.wallG,
                  color.wallB);

        addVertex(featureVertices, pb, bottomB, color.wallR, color.wallG,
                  color.wallB);

        addVertex(featureVertices, pb, roofY, color.wallR, color.wallG,
                  color.wallB);

        // Второй треугольник.
        addVertex(featureVertices, pa, bottomA, color.wallR, color.wallG,
                  color.wallB);

        addVertex(featureVertices, pb, roofY, color.wallR, color.wallG,
                  color.wallB);

        addVertex(featureVertices, pa, roofY, color.wallR, color.wallG,
                  color.wallB);
      }
    }
  }

  // Крыша с отверстиями.
  for (unsigned int index : indices) {
    if (index >= roofPoints.size())
      continue;

    addVertex(featureVertices, roofPoints[index], roofY, color.roofR,
              color.roofG, color.roofB);
  }
}

// -----------------------------------------------------------------------------
// Flat polygon geometry: water, vegetation, landuse, amenity, road areas
// -----------------------------------------------------------------------------

void GLWidget::appendPolygon(const OSMFeature &feature, float heightOffset) {
  if (feature.geometry.size() < 3)
    return;

  std::vector<QVector2D> polygon;
  polygon.reserve(feature.geometry.size());

  float groundSum = 0.0f;
  int validCount = 0;

  for (const GeoPoint &geo : feature.geometry) {
    ScenePoint point;

    if (!geoToScene(geo, point))
      return;

    polygon.push_back(point.position);
    groundSum += point.groundY;
    ++validCount;
  }

  if (polygon.size() >= 2 &&
      sameGeoPoint(feature.geometry.first(), feature.geometry.last())) {
    polygon.pop_back();
  }

  if (polygon.size() < 3 || validCount == 0)
    return;

  const auto triangles = triangulatePolygon(polygon);

  if (triangles.empty())
    return;

  const float y = groundSum / static_cast<float>(validCount) + heightOffset;

  const Colors &color = feature.colors;

  for (const auto &triangle : triangles) {
    for (int index : triangle) {
      addVertex(featureVertices, polygon[index], y, color.roofR, color.roofG,
                color.roofB);
    }
  }
}

// -----------------------------------------------------------------------------
// Line geometry: roads, rivers, railway
// Each line segment becomes a flat-width quad.
// -----------------------------------------------------------------------------

void GLWidget::appendLine(const OSMFeature &feature, float widthMeters,
                          float heightOffset) {
  if (feature.geometry.size() < 2)
    return;

  std::vector<ScenePoint> points;
  points.reserve(feature.geometry.size());

  for (const GeoPoint &geo : feature.geometry) {
    ScenePoint point;

    if (!geoToScene(geo, point)) {
      // Не соединяем участки через точки вне DEM.
      points.clear();
      return;
    }

    points.push_back(point);
  }

  const TerrainSize terrainSize = calculateTerrainSize();

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double physicalWidth =
      std::hypot((dem.width - 1) * gt[1] * metersPerDegreeLon,
                 (dem.width - 1) * gt[4] * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot((dem.height - 1) * gt[2] * metersPerDegreeLon,
                 (dem.height - 1) * gt[5] * metersPerDegreeLat);

  const double maxPhysicalDimension = std::max(physicalWidth, physicalHeight);

  if (maxPhysicalDimension <= 0.0)
    return;

  const float metersToScene = static_cast<float>(2.0 / maxPhysicalDimension);

  const float halfWidth = std::max(0.0001f, widthMeters * metersToScene * 0.5f);

  const float rowDirection = gt[5] < 0.0 ? -1.0f : 1.0f;

  auto getGroundY = [&](const QVector2D &pos, float &y) -> bool {
    const double pixelX = (pos.x() / terrainSize.width + 0.5) * (dem.width - 1);

    const double pixelY =
        (0.5 - pos.y() / (rowDirection * terrainSize.height)) *
        (dem.height - 1);

    float elevation = 0.0f;

    if (!sampleElevation(pixelX, pixelY, elevation))
      return false;

    const float elevationRange = dem.maxElevation - dem.minElevation;

    const float verticalScale = elevationRange > 0.0f
                                    ? verticalScaleCoefficient / elevationRange
                                    : 0.001f;

    y = (elevation - dem.minElevation) * verticalScale;

    return true;
  };

  const Colors &color = feature.colors;

  // Размер одного пикселя DEM в координатах сцены.
  const float pixelSizeX =
      terrainSize.width / static_cast<float>(dem.width - 1);

  const float pixelSizeY =
      terrainSize.height / static_cast<float>(dem.height - 1);

  // Максимальная длина одного участка дороги.
  // Чем резче рельеф, тем меньше должен быть этот шаг.
  const float maxSegmentLength =
      std::max(0.0001f, std::min(pixelSizeX, pixelSizeY) * 1.5f);

  for (std::size_t i = 0; i + 1 < points.size(); ++i) {
    const ScenePoint &a = points[i];
    const ScenePoint &b = points[i + 1];

    const QVector2D direction = b.position - a.position;
    const float length = direction.length();

    if (length < 1e-7f)
      continue;

    const QVector2D normal(-direction.y() / length, direction.x() / length);

    const QVector2D offset = normal * halfWidth;

    // Разбиваем длинный сегмент на короткие участки.
    const int steps =
        std::max(1, static_cast<int>(std::ceil(length / maxSegmentLength)));

    for (int j = 0; j < steps; ++j) {
      const float t0 = static_cast<float>(j) / steps;
      const float t1 = static_cast<float>(j + 1) / steps;

      const QVector2D centerA = a.position * (1.0f - t0) + b.position * t0;

      const QVector2D centerB = a.position * (1.0f - t1) + b.position * t1;

      const QVector2D aLeft = centerA + offset;
      const QVector2D aRight = centerA - offset;

      const QVector2D bLeft = centerB + offset;
      const QVector2D bRight = centerB - offset;

      float yALeft, yARight;
      float yBLeft, yBRight;

      if (!getGroundY(aLeft, yALeft) || !getGroundY(aRight, yARight) ||
          !getGroundY(bLeft, yBLeft) || !getGroundY(bRight, yBRight)) {
        continue;
      }

      yALeft += heightOffset;
      yARight += heightOffset;
      yBLeft += heightOffset;
      yBRight += heightOffset;

      // Первый треугольник.
      addVertex(featureVertices, aLeft, yALeft, color.roofR, color.roofG,
                color.roofB);

      addVertex(featureVertices, aRight, yARight, color.roofR, color.roofG,
                color.roofB);

      addVertex(featureVertices, bLeft, yBLeft, color.roofR, color.roofG,
                color.roofB);

      // Второй треугольник.
      addVertex(featureVertices, aRight, yARight, color.roofR, color.roofG,
                color.roofB);

      addVertex(featureVertices, bRight, yBRight, color.roofR, color.roofG,
                color.roofB);

      addVertex(featureVertices, bLeft, yBLeft, color.roofR, color.roofG,
                color.roofB);
    }
  }
}

// -----------------------------------------------------------------------------
// Point geometry: amenity and other OSM nodes
// -----------------------------------------------------------------------------

void GLWidget::appendPoint(const OSMFeature &feature, float sizeMeters,
                           float heightOffset) {
  if (feature.geometry.isEmpty())
    return;

  ScenePoint point;

  if (!geoToScene(feature.geometry.first(), point))
    return;

  const TerrainSize terrainSize = calculateTerrainSize();

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double physicalWidth =
      std::hypot((dem.width - 1) * gt[1] * metersPerDegreeLon,
                 (dem.width - 1) * gt[4] * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot((dem.height - 1) * gt[2] * metersPerDegreeLon,
                 (dem.height - 1) * gt[5] * metersPerDegreeLat);

  const double maxDimension = std::max(physicalWidth, physicalHeight);

  if (maxDimension <= 0.0)
    return;

  const float halfSize =
      std::max(0.0002f, static_cast<float>(sizeMeters / maxDimension));

  const QVector2D p = point.position;

  const QVector2D a(p.x() - halfSize, p.y() - halfSize);
  const QVector2D b(p.x() + halfSize, p.y() - halfSize);
  const QVector2D c(p.x() + halfSize, p.y() + halfSize);
  const QVector2D d(p.x() - halfSize, p.y() + halfSize);

  const float y = point.groundY + heightOffset;

  const Colors &color = feature.colors;

  addVertex(featureVertices, a, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, b, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, c, y, color.roofR, color.roofG, color.roofB);

  addVertex(featureVertices, a, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, c, y, color.roofR, color.roofG, color.roofB);
  addVertex(featureVertices, d, y, color.roofR, color.roofG, color.roofB);
}

// -----------------------------------------------------------------------------
// OSM feature dispatcher
// -----------------------------------------------------------------------------

void GLWidget::rebuildFeatureMesh() {
  featureVertices.clear();

  if (dem.width < 2 || dem.height < 2) {
    uploadFeatureBuffer();
    update();
    return;
  }

  int buildings = 0;
  int roads = 0;
  int water = 0;
  int vegetation = 0;
  int railways = 0;
  int other = 0;

  for (const OSMFeature &feature : features) {
    if (feature.geometry.isEmpty())
      continue;

    switch (feature.type) {
    case OSMFeatureType::Building:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendBuilding(feature);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 5.0f);

      ++buildings;
      break;

    case OSMFeatureType::Road:
      if (feature.geometryType == OSMGeometryType::Polygon) {
        appendPolygon(feature, 0.0015f);
      } else if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 4.0f;

        appendLine(feature, width, 0.0015f);
      }

      ++roads;
      break;

    case OSMFeatureType::Water:
      if (feature.geometryType == OSMGeometryType::Polygon) {
        appendPolygon(feature, 0.0025f);
      } else if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 5.0f;

        appendLine(feature, width, 0.0025f);
      }

      ++water;
      break;

    case OSMFeatureType::Vegetation:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.001f);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 4.0f, 0.001f);

      ++vegetation;
      break;

    case OSMFeatureType::Landuse:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.0008f);

      ++other;
      break;

    case OSMFeatureType::Railway:
      if (feature.geometryType == OSMGeometryType::LineString) {
        const float width = feature.widthMeters > 0.0
                                ? static_cast<float>(feature.widthMeters)
                                : 4.5f;

        appendLine(feature, width, 0.002f);
      }

      ++railways;
      break;

    case OSMFeatureType::Amenity:
      if (feature.geometryType == OSMGeometryType::Polygon)
        appendPolygon(feature, 0.0015f);
      else if (feature.geometryType == OSMGeometryType::Point)
        appendPoint(feature, 5.0f, 0.002f);

      ++other;
      break;

    case OSMFeatureType::Unknown:
    default:
      break;
    }
  }

  uploadFeatureBuffer();

  qDebug() << "OSM mesh rebuilt:"
           << "buildings =" << buildings << "roads =" << roads
           << "water =" << water << "vegetation =" << vegetation
           << "railways =" << railways << "other =" << other
           << "vertices =" << featureVertices.size();
}

void GLWidget::uploadFeatureBuffer() {
  if (!context() || !context()->isValid())
    return;

  makeCurrent();

  if (!featureBuffer.isCreated())
    featureBuffer.create();

  if (featureBuffer.isCreated()) {
    featureBuffer.bind();

    if (featureVertices.empty())
      featureBuffer.allocate(0);
    else
      featureBuffer.allocate(
          featureVertices.data(),
          static_cast<int>(featureVertices.size() * sizeof(Vertex)));

    featureBuffer.release();
  }

  doneCurrent();
}

// -----------------------------------------------------------------------------
// Rendering
// -----------------------------------------------------------------------------

void GLWidget::resizeGL(int width, int height) {
  glViewport(0, 0, width, height);
}

void GLWidget::paintGL() {
  glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

  if (!shaderReady)
    return;

  QMatrix4x4 projection;

  projection.perspective(45.0f,
                         static_cast<float>(std::max(1, width())) /
                             static_cast<float>(std::max(1, height())),
                         0.01f, 100.0f);

  QMatrix4x4 view;

  view.translate(panX, panY, -zoom);

  view.rotate(rotationX, 1.0f, 0.0f, 0.0f);
  view.rotate(rotationZ, 0.0f, 1.0f, 0.0f);

  const QMatrix4x4 mvp = projection * view;

  auto drawBuffer = [&](QOpenGLBuffer &buffer, std::size_t vertexCount) {
    if (vertexCount == 0 || !buffer.isCreated())
      return;

    shaderProgram.bind();
    shaderProgram.setUniformValue("mvp", mvp);

    buffer.bind();

    const int positionLocation = shaderProgram.attributeLocation("position");

    const int colorLocation = shaderProgram.attributeLocation("color");

    shaderProgram.enableAttributeArray(positionLocation);
    shaderProgram.enableAttributeArray(colorLocation);

    shaderProgram.setAttributeBuffer(positionLocation, GL_FLOAT,
                                     offsetof(Vertex, x), 3, sizeof(Vertex));

    shaderProgram.setAttributeBuffer(colorLocation, GL_FLOAT,
                                     offsetof(Vertex, r), 3, sizeof(Vertex));

    glDrawArrays(GL_TRIANGLES, 0, static_cast<GLsizei>(vertexCount));

    shaderProgram.disableAttributeArray(positionLocation);
    shaderProgram.disableAttributeArray(colorLocation);

    buffer.release();
    shaderProgram.release();
  };

  // Сначала рельеф.
  if (meshReady)
    drawBuffer(vertexBuffer, vertices.size());

  // Затем здания и остальные OSM-объекты.
  drawBuffer(featureBuffer, featureVertices.size());
}

// -----------------------------------------------------------------------------
// Camera controls
// -----------------------------------------------------------------------------

void GLWidget::mousePressEvent(QMouseEvent *event) {
  lastMousePosition = event->position().toPoint();
}

void GLWidget::mouseMoveEvent(QMouseEvent *event) {
  const QPoint current = event->position().toPoint();

  const QPoint delta = current - lastMousePosition;

  if (event->buttons() & Qt::LeftButton) {
    rotationZ += delta.x() * 0.5f;
    rotationX += delta.y() * 0.5f;

    update();
  }

  if (event->buttons() & Qt::RightButton) {
    const float worldPerPixel =
        (2.0f * zoom * 0.41421356f) / static_cast<float>(std::max(1, height()));

    panX += delta.x() * worldPerPixel;
    panY -= delta.y() * worldPerPixel;

    update();
  }

  lastMousePosition = current;
}

void GLWidget::wheelEvent(QWheelEvent *event) {
  zoom -= event->angleDelta().y() / 120.0f * 0.25f;

  zoom = std::clamp(zoom, 0.01f, 15.0f);

  update();
}

// -----------------------------------------------------------------------------
// DEM physical dimensions
// -----------------------------------------------------------------------------

GLWidget::TerrainSize GLWidget::calculateTerrainSize() const {
  TerrainSize result;

  if (dem.width < 2 || dem.height < 2)
    return result;

  const double *gt = dem.geoTransform;

  const double centerLat =
      gt[3] + 0.5 * (dem.width - 1) * gt[4] + 0.5 * (dem.height - 1) * gt[5];

  const double metersPerDegreeLat = 111132.0;

  const double metersPerDegreeLon =
      111320.0 * std::cos(qDegreesToRadians(centerLat));

  const double dxLon = (dem.width - 1) * gt[1];
  const double dxLat = (dem.width - 1) * gt[4];

  const double dyLon = (dem.height - 1) * gt[2];
  const double dyLat = (dem.height - 1) * gt[5];

  const double physicalWidth =
      std::hypot(dxLon * metersPerDegreeLon, dxLat * metersPerDegreeLat);

  const double physicalHeight =
      std::hypot(dyLon * metersPerDegreeLon, dyLat * metersPerDegreeLat);

  const double maxDimension = std::max(physicalWidth, physicalHeight);

  if (maxDimension <= 0.0)
    return result;

  result.width = static_cast<float>(2.0 * physicalWidth / maxDimension);

  result.height = static_cast<float>(2.0 * physicalHeight / maxDimension);

  return result;
}

// Загрузка локального JSON
void MainWindow::onLoadOsmJsonButtonClicked() {
  const QString fileName =
      QFileDialog::getOpenFileName(this, "Выбрать файл OSM JSON", QString(),
                                   "JSON files (*.json);;Все файлы (*)");

  if (fileName.isEmpty())
    return;

  QFile file(fileName);

  if (!file.open(QIODevice::ReadOnly)) {
    QMessageBox::warning(this, "Ошибка", "Не удалось открыть файл OSM JSON.");
    return;
  }

  const QByteArray jsonData = file.readAll();
  file.close();

  QJsonParseError parseError;

  const QJsonDocument document = QJsonDocument::fromJson(jsonData, &parseError);

  if (parseError.error != QJsonParseError::NoError) {

    QMessageBox::warning(this, "Ошибка JSON", parseError.errorString());

    return;
  }

  QJsonArray elements;

  // Формат с корневым массивом
  if (document.isArray()) {

    elements = document.array();
  }
  // Стандартный формат Overpass
  else if (document.isObject()) {

    const QJsonObject root = document.object();

    if (!root.value("elements").isArray()) {

      QMessageBox::warning(this, "Ошибка",
                           "В JSON отсутствует массив elements.");

      return;
    }

    elements = root.value("elements").toArray();
  } else {

    QMessageBox::warning(this, "Ошибка", "Некорректная структура OSM JSON.");

    return;
  }

  if (elements.isEmpty()) {

    QMessageBox::information(this, "Информация",
                             "В файле отсутствуют OSM-объекты.");

    return;
  }

  loadOSMFeatures(elements);
}

// Общая загрузка объектов OSM
void MainWindow::loadOSMFeatures(const QJsonArray &elements) {
  const QVector<OSMFeature> features = OSMDataParser::parse(elements);

  if (features.isEmpty()) {

    QMessageBox::information(this, "Информация",
                             "Не удалось найти поддерживаемые OSM-объекты.");

    return;
  }

  // Статистика
  int buildings = 0;
  int roads = 0;
  int water = 0;
  int vegetation = 0;
  int other = 0;

  for (const OSMFeature &feature : features) {

    switch (feature.type) {

    case OSMFeatureType::Building:
      ++buildings;
      break;

    case OSMFeatureType::Road:
      ++roads;
      break;

    case OSMFeatureType::Water:
      ++water;
      break;

    case OSMFeatureType::Vegetation:
      ++vegetation;
      break;

    default:
      ++other;
      break;
    }
  }

  qDebug() << "Загружено OSM объектов:" << features.size();

  qDebug() << "Здания:" << buildings;
  qDebug() << "Дороги:" << roads;
  qDebug() << "Вода:" << water;
  qDebug() << "Растительность:" << vegetation;
  qDebug() << "Другие:" << other;

  // Передаём все объекты в OpenGL
  glWidget->setFeatures(features);
}

OSMFeatureType OSMClassifier::classify(const QJsonObject &tags) {
  // Здания
  if (tags.contains("building"))
    return OSMFeatureType::Building;

  // Дороги
  if (tags.contains("highway"))
    return OSMFeatureType::Road;

  // Железные дороги
  if (tags.contains("railway"))
    return OSMFeatureType::Railway;

  // Вода
  if (tags.contains("waterway") ||
      tags.value("natural").toString() == "water" || tags.contains("water")) {
    return OSMFeatureType::Water;
  }

  // Растительность
  const QString natural = tags.value("natural").toString();
  const QString landuse = tags.value("landuse").toString();
  const QString leisure = tags.value("leisure").toString();

  if (natural == "wood" || natural == "tree" || natural == "scrub" ||
      landuse == "forest" || leisure == "park") {
    return OSMFeatureType::Vegetation;
  }

  // Остальные территории
  if (tags.contains("landuse"))
    return OSMFeatureType::Landuse;

  // Общественные объекты
  if (tags.contains("amenity") || tags.contains("shop") ||
      tags.contains("office")) {
    return OSMFeatureType::Amenity;
  }

  return OSMFeatureType::Unknown;
}

QByteArray OSMDataLoader::fetchOSMData(const QString &queryType,
                                       const int &timeout, const QRect &area) {
  Q_UNUSED(area);

  // Формируем Overpass QL
  QString finalQuery = buildOverpassQuery(timeout, queryType);

  if (finalQuery.isEmpty()) {
    qDebug() << "Неизвестный тип запроса:" << queryType;
    return "";
  }

  // Границы области:
  // south, west, north, east
  const double minLat = defaultArea.lat - defaultArea.width;
  const double maxLat = defaultArea.lat + defaultArea.width;
  const double minLon = defaultArea.lon - defaultArea.height;
  const double maxLon = defaultArea.lon + defaultArea.height;

  const QString bbox = QString("%1,%2,%3,%4")
                           .arg(minLat, 0, 'f', 6)
                           .arg(minLon, 0, 'f', 6)
                           .arg(maxLat, 0, 'f', 6)
                           .arg(maxLon, 0, 'f', 6);

  // Подставляем реальные координаты вместо {{bbox}}
  finalQuery.replace("{{bbox}}", bbox);

  qDebug() << "Тип запроса:" << queryType;
  qDebug() << "BBox:" << bbox;
  qDebug() << "Весь запрос:" << finalQuery;

  QNetworkRequest request(url);

  request.setHeader(QNetworkRequest::ContentTypeHeader,
                    "application/x-www-form-urlencoded");
  request.setHeader(
      QNetworkRequest::UserAgentHeader,
      "BMSTU-NIR-OSM-Loader/1.0 (+https://github.com/Andrey8066)");
  request.setRawHeader("Accept", "application/json");

  QByteArray postData = "data=";
  postData += QUrl::toPercentEncoding(finalQuery);

  reply = manager->post(request, postData);

  connect(reply, &QNetworkReply::finished, this, [this, area, queryType]() {
    const QVariant status =
        reply->attribute(QNetworkRequest::HttpStatusCodeAttribute);

    qDebug() << "HTTP status:" << status;

    if (reply->error() != QNetworkReply::NoError) {

      qDebug() << "Ошибка загрузки OSM данных:" << reply->errorString();

      const QByteArray serverResponse = reply->readAll();

      qDebug() << "Ответ сервера:" << serverResponse;

      emit errorOccurred("Не удалось загрузить данные из OSM");

      reply->deleteLater();
      return;
    }

    requestResult = reply->readAll();

    QJsonParseError parseError;

    const QJsonDocument doc =
        QJsonDocument::fromJson(requestResult, &parseError);

    if (parseError.error != QJsonParseError::NoError) {

      qDebug() << "Ошибка парсинга JSON:" << parseError.errorString();

      qDebug() << "Ответ сервера:" << requestResult;

      reply->deleteLater();
      return;
    }

    qDebug() << "OSM JSON успешно получен.";

    qDebug() << "Получено байт:" << requestResult.size();

    const QJsonObject root = doc.object();
    const QJsonArray elements = root["elements"].toArray();

    emit dataLoaded(elements, queryType);
    int ways = 0;
    int relations = 0;
    int nodes = 0;

    int buildings = 0;
    int buildingParts = 0;

    for (const QJsonValue &value : elements) {
      QJsonObject object = value.toObject();
      QJsonObject tags = object["tags"].toObject();

      QString type = object["type"].toString();

      if (type == "way")
        ways++;

      if (type == "relation")
        relations++;

      if (type == "node")
        nodes++;

      if (tags.contains("building"))
        buildings++;

      if (tags.contains("building:part"))
        buildingParts++;
    }

    qDebug() << "========== OSM STATISTICS ==========";
    qDebug() << "Total elements:" << elements.size();
    qDebug() << "Ways:" << ways;
    qDebug() << "Relations:" << relations;
    qDebug() << "Nodes:" << nodes;
    qDebug() << "Building tags:" << buildings;
    qDebug() << "Building parts:" << buildingParts;

    int wayBuildings = 0;
    int relationBuildings = 0;

    int wayParts = 0;
    int relationParts = 0;

    for (const QJsonValue &value : elements) {
      QJsonObject object = value.toObject();
      QJsonObject tags = object["tags"].toObject();

      QString type = object["type"].toString();

      bool hasBuilding = tags.contains("building");
      bool hasPart = tags.contains("building:part");

      if (type == "way" && hasBuilding)
        wayBuildings++;

      if (type == "relation" && hasBuilding)
        relationBuildings++;

      if (type == "way" && hasPart)
        wayParts++;

      if (type == "relation" && hasPart)
        relationParts++;
    }

    qDebug() << "Way buildings:" << wayBuildings;
    qDebug() << "Relation buildings:" << relationBuildings;

    qDebug() << "Way building parts:" << wayParts;
    qDebug() << "Relation building parts:" << relationParts;
  });

  return requestResult;
}

// Формирование Overpass QL запроса - правильная syntax

QString OSMDataLoader::buildOverpassQuery(const int &timeout,
                                          const QString &queryType) {
  if (queryType == "buildings") {
    return QString(R"(
            [out:json][timeout:%1];

            (
                nwr["building"]({{bbox}});
                nwr["building:part"]({{bbox}});
                relation["type"="building"]({{bbox}});
            );

            out geom;
        )")
        .arg(timeout);
  }

  if (queryType == "all") {
    return QString(R"([out:json][timeout:%1];
                   (
                   nwr["building"]({{bbox}});
                   nwr["highway"]({{bbox}});
                   );
                   out geom;
                   )")
        .arg(timeout);
  }

  qWarning() << "Неизвестный тип запроса:" << queryType;
  return QString();
}

// Сохранение JSON в файл
QString OSMDataLoader::saveToJSONFile(const QJsonArray &data,
                                      const QString &filename) {
  QFile file(filename);

  if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
    qDebug() << "Ошибка открытия файла для записи:" << filename;
    return "";
  }

  file.write(QJsonDocument(data).toJson());
  file.close();

  return filename;
}

void OSMDataLoader::waitData() {
  if (!reply) {
    qDebug() << "Нет активного запроса";
    return;
  }

  if (!reply->isFinished()) {
    QEventLoop loop;

    connect(reply, &QNetworkReply::finished, &loop, &QEventLoop::quit);

    loop.exec();
  }

  qDebug() << "waitData(): запрос завершён";
}

double OSMDataLoader::radToLat(double radius) {
  return radius / (earthRadius * M_PI) * 180;
}
double OSMDataLoader::radToLon(double radius) {
  const double latRadians = defaultArea.lat * M_PI / 180.0;

  return radius / (earthRadius * std::cos(latRadians)) * 180.0 / M_PI;
}

double parseLength(const QJsonValue &value, bool *ok) {
  *ok = false;

  double result = 0.0;

  if (value.isDouble()) {
    result = value.toDouble();
    *ok = true;
    return result;
  }

  QString text = value.toString().trimmed().toLower();

  bool feet = text.endsWith("ft");

  if (feet)
    text.chop(2);
  else if (text.endsWith("m"))
    text.chop(1);

  result = text.trimmed().toDouble(ok);

  if (feet && *ok)
    result *= 0.3048;

  return result;
}

float parseBuildingHeight(const QJsonObject &tags) {
  bool ok = false;

  double height = parseLength(tags.value("height"), &ok);

  if (ok && height > 0.0)
    return static_cast<float>(std::clamp(height, 2.0, 300.0));

  const double levels = tags.value("building:levels").toString().toDouble(&ok);

  if (ok && levels > 0.0) {
    return static_cast<float>(std::clamp(levels * 3.0, 2.0, 300.0));
  }

  return 9.0f;
}

float parseRoadWidth(const QJsonObject &tags) {
  bool ok = false;

  double width = parseLength(tags.value("width"), &ok);

  if (ok && width > 0.0)
    return static_cast<float>(width);

  const double lanes = tags.value("lanes").toString().toDouble(&ok);

  if (ok && lanes > 0.0)
    return static_cast<float>(lanes * 3.5);

  const QString highway = tags.value("highway").toString();

  if (highway == "motorway" || highway == "trunk")
    return 12.0f;

  if (highway == "primary" || highway == "secondary")
    return 9.0f;

  if (highway == "residential")
    return 6.0f;

  if (highway == "footway" || highway == "path" || highway == "cycleway")
    return 2.0f;

  return 5.0f;
}

Colors getBuildingColors(const QJsonObject &tags) {
  const QString type = tags.value("building").toString();

  if (type == "industrial") {
    return Colors(0.55f, 0.55f, 0.55f, 0.35f, 0.35f, 0.35f);
  }

  if (type == "commercial" || type == "retail") {
    return Colors(0.72f, 0.62f, 0.48f, 0.35f, 0.32f, 0.28f);
  }

  if (type == "house") {
    return Colors(0.75f, 0.65f, 0.52f, 0.45f, 0.25f, 0.20f);
  }

  return Colors(0.58f, 0.63f, 0.70f, 0.28f, 0.36f, 0.28f);
}

bool samePoint(const GeoPoint &a, const GeoPoint &b) {
  constexpr double eps = 1e-9;

  return std::abs(a.lat - b.lat) < eps && std::abs(a.lon - b.lon) < eps;
}

bool pointInRing(const GeoPoint &point, const QVector<GeoPoint> &ring) {
  bool inside = false;

  if (ring.size() < 3)
    return false;

  for (int i = 0, j = ring.size() - 1; i < ring.size(); j = i++) {

    const GeoPoint &a = ring[i];
    const GeoPoint &b = ring[j];

    const bool intersects =
        ((a.lat > point.lat) != (b.lat > point.lat)) &&
        (point.lon <
         (b.lon - a.lon) * (point.lat - a.lat) / (b.lat - a.lat) + a.lon);

    if (intersects)
      inside = !inside;
  }

  return inside;
}

// Читает координаты одного way из участника relation.
QVector<GeoPoint> readMemberGeometry(const QJsonObject &member) {
  QVector<GeoPoint> points;

  const QJsonArray geometry = member.value("geometry").toArray();

  for (const QJsonValue &value : geometry) {
    const QJsonObject point = value.toObject();

    if (!point.contains("lat") || !point.contains("lon"))
      continue;

    points.push_back(
        {point.value("lat").toDouble(), point.value("lon").toDouble()});
  }

  return points;
}

QVector<QVector<GeoPoint>> readRings(const QJsonObject &object,
                                     const QString &role) {
  QVector<QVector<GeoPoint>> segments;

  const QJsonArray members = object.value("members").toArray();

  for (const QJsonValue &value : members) {
    const QJsonObject member = value.toObject();

    if (member.value("type").toString() != "way")
      continue;

    if (member.value("role").toString() != role)
      continue;

    QVector<GeoPoint> points = readMemberGeometry(member);

    if (points.size() >= 2)
      segments.push_back(std::move(points));
  }

  QVector<QVector<GeoPoint>> rings;

  while (!segments.isEmpty()) {
    QVector<GeoPoint> ring = segments.takeFirst();

    bool changed = true;

    while (changed && !segments.isEmpty()) {
      changed = false;

      for (int i = 0; i < segments.size(); ++i) {
        QVector<GeoPoint> segment = segments[i];

        if (samePoint(ring.back(), segment.front())) {
          for (int j = 1; j < segment.size(); ++j)
            ring.push_back(segment[j]);

        } else if (samePoint(ring.back(), segment.back())) {
          std::reverse(segment.begin(), segment.end());

          for (int j = 1; j < segment.size(); ++j)
            ring.push_back(segment[j]);

        } else if (samePoint(ring.front(), segment.back())) {
          // Вставляем в начало в обратном порядке.
          for (int j = segment.size() - 2; j >= 0; --j)
            ring.prepend(segment[j]);

        } else if (samePoint(ring.front(), segment.front())) {
          std::reverse(segment.begin(), segment.end());

          for (int j = segment.size() - 2; j >= 0; --j)
            ring.prepend(segment[j]);

        } else {
          continue;
        }

        segments.removeAt(i);
        changed = true;
        break;
      }
    }

    if (ring.size() < 4 || !samePoint(ring.front(), ring.back()))
      continue;

    ring.removeLast();

    if (ring.size() >= 3)
      rings.push_back(std::move(ring));
  }

  return rings;
}

bool readGeometry(const QJsonObject &object, OSMFeature &feature) {
  const QString osmType = object.value("type").toString();

  if (osmType == "node") {

    if (!object.contains("lat") || !object.contains("lon")) {
      return false;
    }

    feature.geometryType = OSMGeometryType::Point;

    feature.geometry.push_back(
        {object.value("lat").toDouble(), object.value("lon").toDouble()});

    return true;
  }

  if (osmType == "way") {

    const QJsonArray geometry = object.value("geometry").toArray();

    for (const QJsonValue &value : geometry) {

      const QJsonObject point = value.toObject();

      if (!point.contains("lat") || !point.contains("lon")) {
        continue;
      }

      feature.geometry.push_back(
          {point.value("lat").toDouble(), point.value("lon").toDouble()});
    }

    if (feature.geometry.isEmpty())
      return false;

    const bool closed =
        feature.geometry.size() >= 4 &&
        std::abs(feature.geometry.front().lat - feature.geometry.back().lat) <
            1e-10 &&
        std::abs(feature.geometry.front().lon - feature.geometry.back().lon) <
            1e-10;

    const bool area = feature.tags.value("area").toString() == "yes" ||
                      (feature.tags.contains("building") && closed) ||
                      (feature.tags.contains("landuse") && closed) ||
                      (feature.tags.contains("leisure") && closed) ||
                      (feature.tags.contains("natural") && closed) ||
                      (feature.tags.contains("water") && closed);

    if (area) {
      feature.geometryType = OSMGeometryType::Polygon;

      if (closed)
        feature.geometry.pop_back();
    } else {
      feature.geometryType = OSMGeometryType::LineString;
    }

    return true;
  }

  // Relation пока не обрабатываем
  return false;
}

QVector<OSMFeature> OSMDataParser::parse(const QJsonArray &elements) {
  QVector<OSMFeature> result;

  result.reserve(elements.size());

  // ID way, которые уже войдут в геометрию relation.
  QSet<qint64> relationWayIds;

  // Первый проход: собираем ID внешних way поддерживаемых relation.
  for (const QJsonValue &value : elements) {
    if (!value.isObject())
      continue;

    const QJsonObject object = value.toObject();

    if (object.value("type").toString() != "relation")
      continue;

    const QJsonObject tags = object.value("tags").toObject();
    const QString relationType = tags.value("type").toString();

    if (relationType != "multipolygon" && relationType != "boundary" &&
        relationType != "building") {
      continue;
    }

    const QJsonArray members = object.value("members").toArray();

    for (const QJsonValue &memberValue : members) {
      const QJsonObject member = memberValue.toObject();

      if (member.value("type").toString() != "way")
        continue;

      const QString role = member.value("role").toString();

      const bool isOuter = role == "outer";
      const bool isBuildingInner =
          OSMClassifier::classify(tags) == OSMFeatureType::Building &&
          role == "inner";

      if (!isOuter && !isBuildingInner)
        continue;
      relationWayIds.insert(
          static_cast<qint64>(member.value("ref").toDouble()));
    }
  }

  // Второй проход: обрабатываем объекты.
  for (const QJsonValue &value : elements) {
    if (!value.isObject())
      continue;

    const QJsonObject object = value.toObject();

    const QString osmType = object.value("type").toString();

    // Не обрабатываем way повторно, если он уже вошёл в relation.
    if (osmType == "way") {
      const qint64 wayId = static_cast<qint64>(object.value("id").toDouble());

      if (relationWayIds.contains(wayId))
        continue;
    }

    OSMFeature feature;

    feature.id = static_cast<qint64>(object.value("id").toDouble());

    feature.tags = object.value("tags").toObject();

    feature.type = OSMClassifier::classify(feature.tags);

    if (feature.type == OSMFeatureType::Unknown)
      continue;

    // Relation обрабатываем отдельно.
    if (osmType == "relation") {
      const QString relationType = feature.tags.value("type").toString();

      if (relationType != "multipolygon" && relationType != "boundary" &&
          relationType != "building") {
        continue;
      }

      // 1. Собираем внешние кольца relation.
      const auto outerRings = readRings(object, "outer");

      // 2. Внутренние кольца нужны только зданиям.
      QVector<QVector<GeoPoint>> innerRings;

      if (feature.type == OSMFeatureType::Building) {
        innerRings = readRings(object, "inner");
      }

      // 3. Создаём отдельный OSMFeature для каждого внешнего кольца.
      for (const auto &outer : outerRings) {
        OSMFeature relationFeature = feature;

        relationFeature.geometryType = OSMGeometryType::Polygon;
        relationFeature.geometry = outer;

        if (relationFeature.type == OSMFeatureType::Building) {
          relationFeature.heightMeters =
              parseBuildingHeight(relationFeature.tags);
          relationFeature.minHeightMeters = 0.0f;

          const QString minLevel =
              relationFeature.tags.value("building:min_level").toString();

          if (!minLevel.isEmpty()) {
            bool ok = false;
            const float levels = minLevel.toFloat(&ok);

            if (ok)
              relationFeature.minHeightMeters = levels * 3.0f;
          }

          relationFeature.colors = getBuildingColors(relationFeature.tags);

          // 4. Прикрепляем к зданию только те inner,
          // которые находятся внутри текущего outer.
          for (const auto &inner : innerRings) {
            if (inner.isEmpty())
              continue;

            if (pointInRing(inner.first(), outer)) {
              relationFeature.innerRings.push_back(inner);
            }
          }
        }

        result.push_back(std::move(relationFeature));
      }

      continue;
    }

    // Обычные node и way.
    if (!readGeometry(object, feature))
      continue;

    if (feature.type == OSMFeatureType::Building) {
      feature.heightMeters = parseBuildingHeight(feature.tags);
      feature.minHeightMeters = 0.0f;

      const QString minLevel =
          feature.tags.value("building:min_level").toString();

      if (!minLevel.isEmpty()) {
        bool ok = false;
        const float levels = minLevel.toFloat(&ok);

        if (ok)
          feature.minHeightMeters = levels * 3.0f;
      }
      feature.colors = getBuildingColors(feature.tags);
    }

    if (feature.type == OSMFeatureType::Road) {
      feature.widthMeters = parseRoadWidth(feature.tags);

      const QString highway = feature.tags.value("highway").toString();

      if (highway == "footway" || highway == "path" || highway == "cycleway")
        feature.colors = Colors(0, 0.5f, 0, 0, 0.5f, 0);
      if (highway == "steps")
        feature.colors = Colors(0.5f, 0, 0, 0.5f, 0, 0);
    }

    result.push_back(std::move(feature));
  }

  return result;
}
